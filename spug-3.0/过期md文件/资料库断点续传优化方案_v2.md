### 2.1 步骤1: 新增分片检查API调用方法（含边界处理）

**文件**: `spug_web/src/pages/document/stores/upload/core/transfer.js`

```javascript
/**
 * 检查已上传分片（断点续传）- 含边界处理和格式校验
 * @param {string} fileHash - 文件MD5哈希
 * @param {number} fileSize - 文件大小
 * @param {number} totalChunks - 总分片数
 * @param {boolean} isPublic - 是否公共空间
 * @returns {Promise<{exists: boolean, uploaded_chunks: number[], count: number}>}
 */
async checkUploadedChunks(fileHash, fileSize, totalChunks, isPublic = false) {
  try {
    const { http } = await import('libs');
    const result = await http.post(API_ENDPOINTS.CHECK_UPLOADED_CHUNKS, {
      file_hash: fileHash,
      file_size: fileSize,
      total_chunks: totalChunks,
      is_public: isPublic
    });
    
    // 【边界处理】校验后端返回格式，避免异常导致前端崩溃
    if (!result || typeof result !== 'object') {
      console.warn('[TransferStore] 分片检查返回格式异常，默认从头开始');
      return { exists: false, uploaded_chunks: [], count: 0 };
    }
    
    // 【边界处理】确保 uploaded_chunks 是数组
    if (!Array.isArray(result.uploaded_chunks)) {
      console.warn('[TransferStore] uploaded_chunks 非数组格式，默认从头开始');
      return { exists: false, uploaded_chunks: [], count: 0 };
    }
    
    // 【边界处理】过滤无效分片索引（超出范围或负数）
    const validChunks = result.uploaded_chunks.filter(idx => 
      Number.isInteger(idx) && idx >= 0 && idx < totalChunks
    );
    
    if (validChunks.length !== result.uploaded_chunks.length) {
      console.warn(`[TransferStore] 过滤了 ${result.uploaded_chunks.length - validChunks.length} 个无效分片索引`);
    }
    
    return {
      exists: result.exists === true,
      uploaded_chunks: validChunks,
      count: validChunks.length,
      merge_status: result.merge_status,
      merge_task_id: result.merge_task_id
    };
    
  } catch (error) {
    console.error('[TransferStore] 检查已上传分片失败:', error);
    // 【降级处理】失败时返回空列表，从头开始上传
    return { exists: false, uploaded_chunks: [], count: 0 };
  }
}
```

```javascript
@action
async resumeItem(itemId) {
  const item = this.queueStore.findUploadItemInCurrentTenant(itemId);
  if (!item) return;

  if (!['paused', 'error', 'waiting'].includes(item.status)) {
    return;
  }

  // 检查后端状态
  if (item.transferId) {
    try {
      const transfers = await this.transferStore.fetchTransfers(
        this.rootStore.navigationStore?.isPublic
      );
      const currentTransfer = transfers.find(t => t.id === item.transferId);
      
      if (currentTransfer) {
        if (currentTransfer.status === 'COMPLETED') {
          item.status = 'completed';
          item.percent = 100;
          item.error = null;
          return;
        }
        
        if (currentTransfer.status === 'CANCELED' || currentTransfer.status === 'FAILED') {
          item.status = 'error';
          item.error = currentTransfer.status === 'CANCELED' ? '已取消' : '失败';
          return;
        }
        
        // 同步后端的 file_hash
        if (currentTransfer.file_hash && !item.fileHash) {
          item.fileHash = currentTransfer.file_hash;
        }
      }
    } catch (error) {
      console.error('[resumeItem] 获取传输状态失败:', error);
    }
  }

  // 检查是否有文件对象
  if (!item.file || !(item.file instanceof File)) {
    message.warning(`"${item.name}" 需要重新选择文件才能继续上传`);
    item.error = '请重新选择文件';
    item.status = 'error';
    this.triggerFileReSelection(itemId);
    return;
  }

  // 【核心修复】大文件断点续传检查
  const chunkCount = Math.ceil(item.fileSize / UPLOAD_CONSTANTS.CHUNK_SIZE);
  
  if (item.fileSize > UPLOAD_CONSTANTS.NORMAL_UPLOAD_THRESHOLD && item.fileHash) {
    try {
      message.loading('正在检查已上传分片...', 0);
      
      const checkResult = await this.transferStore.checkUploadedChunks(
        item.fileHash,
        item.fileSize,
        chunkCount,
        item.isPublic
      );
      
      message.destroy();
      
      if (checkResult.exists && checkResult.uploaded_chunks.length > 0) {
        const uploadedCount = checkResult.uploaded_chunks.length;
        const uploadedBytes = uploadedCount * UPLOAD_CONSTANTS.CHUNK_SIZE;
        const percent = Math.min(Math.round((uploadedBytes / item.fileSize) * 100), 99);
        
        item.currentChunk = uploadedCount;
        item.percent = percent;
        item.chunks = checkResult.uploaded_chunks.map(idx => ({ index: idx, uploaded: true }));
        
        message.success(`已找到 ${uploadedCount} 个已上传分片，将从 ${percent}% 处继续`);
        
        // 更新后端 uploaded_chunks 计数
        if (item.transferId) {
          await this.transferStore.updateTransferProgress(item.transferId, percent);
        }
      }
    } catch (error) {
      message.destroy();
      console.error('[resumeItem] 检查已上传分片失败:', error);
      // 【降级处理】继续上传，从头开始
    }
  }

  // 恢复上传
  item.status = 'uploading';
  item.error = null;
  item.canAbort = true;
  item.isPausedByUser = false;
  item.isCancelledByUser = false;
  item.abortController = new AbortController();

  const resumeFolderId = item.folderId !== null ? item.folderId : this.rootStore.navigationStore?.currentFolderId;
  
  // 使用改进的 resumeChunkedUpload 方法，支持从指定分片开始
  await this.resumeChunkedUpload(item.file, resumeFolderId, item, chunkCount);
}
```

### 2.3 步骤3: 新增断点续传专用上传方法（含重试机制）

**文件**: `spug_web/src/pages/document/stores/upload/core/index.js`

```javascript
/**
 * 断点续传专用方法 - 支持从指定分片开始上传
 * 【性能优化】支持单个分片失败重试（最多3次）
 * @param {File} file - 文件对象
 * @param {number|null} folderId - 文件夹ID
 * @param {Object} item - 上传队列项
 * @param {number} chunkCount - 总分片数
 */
@action
async resumeChunkedUpload(file, folderId, item, chunkCount) {
  const uploadId = item.id;
  const startChunkIndex = item.currentChunk || 0;
  
  try {
    // 如果还没计算MD5，需要重新计算
    if (!item.fileHash) {
      this.queueStore.updateUploadItem(uploadId, {
        status: 'calculating',
        error: '正在计算文件哈希...',
      });
      
      const fileHash = await this.rootStore.md5Store?.calculateFileMD5(file, uploadId);
      item.fileHash = fileHash;
      
      // 保存到后端
      if (item.transferId) {
        await this.transferStore.updateTransferHash(item.transferId, fileHash, chunkCount);
      }
    }

    const fileSize = file.size;

    this.queueStore.updateUploadItem(uploadId, {
      status: 'uploading',
      canAbort: true,
      chunkCount: chunkCount,
      fileSize: fileSize,
    });

    // 创建AbortController
    const taskAbortController = new AbortController();
    this.queueStore.updateUploadItem(uploadId, {
      abortController: taskAbortController,
      isPausedByUser: false,
      isCancelledByUser: false,
    });

    // 【核心】从断点继续上传，支持单个分片失败重试
    for (let chunkIndex = startChunkIndex; chunkIndex < chunkCount; chunkIndex++) {
      // 重新获取uploadItem
      const currentItem = this.queueStore.findUploadItemInCurrentTenant(uploadId);
      if (!currentItem) break;

      // 检查暂停/取消状态
      if (currentItem.isPausedByUser || currentItem.status === 'paused') {
        this.queueStore.updateUploadItem(uploadId, {
          status: 'paused',
          error: '已暂停',
          currentChunk: chunkIndex,
          canAbort: false,
        });
        break;
      }

      if (this.rootStore.isCancelled || currentItem.isCancelledByUser) {
        this.queueStore.updateUploadItem(uploadId, {
          status: 'error',
          error: '已取消',
          canAbort: false,
        });
        throw new Error('用户取消');
      }

      // 【边界处理】单个分片上传重试机制（最多3次）
      let chunkRetry = 0;
      let chunkError = null;
      
      while (chunkRetry < 3) {
        try {
          await this.chunkUploadStore.uploadSingleChunk(
            file, uploadId, chunkIndex, chunkCount, 
            item.fileHash, folderId, taskAbortController, item.isPublic
          );
          chunkError = null; // 上传成功，清除错误
          break; // 跳出重试循环
        } catch (error) {
          chunkError = error;
          chunkRetry++;
          
          const errorMsg = error?.message || String(error);
          const isAbortError = errorMsg.includes('用户暂停') || errorMsg.includes('用户取消');
          
          // 用户主动操作，不重试
          if (isAbortError) {
            throw error;
          }
          
          if (chunkRetry >= 3) {
            console.error(`[resumeChunkedUpload] 分片 ${chunkIndex} 上传3次均失败`);
            throw new Error(`分片${chunkIndex}上传失败，请重试`);
          }
          
          console.warn(`[resumeChunkedUpload] 分片 ${chunkIndex} 第${chunkRetry}次上传失败，${chunkRetry < 3 ? '1秒后重试' : '放弃'}`, error);
          // 等待1秒后重试
          await new Promise(resolve => setTimeout(resolve, 1000));
        }
      }
      
      if (chunkError) {
        throw chunkError;
      }
      
      // 更新已上传分片计数
      item.currentChunk = chunkIndex + 1;
      
      // 【性能优化】每上传10个分片，让出主线程避免卡顿
      if (chunkIndex % 10 === 0) {
        await new Promise(resolve => requestAnimationFrame(resolve));
      }
    }

    // 检查是否暂停
    const finalItem = this.queueStore.findUploadItemInCurrentTenant(uploadId);
    if (finalItem && finalItem.status !== 'paused') {
      // 合并分片
      await this.chunkUploadStore.mergeChunks(
        file, uploadId, chunkCount, item.fileHash, folderId, item.isPublic
      );
    }

  } catch (error) {
    const errorMsg = `（${file.name}）上传失败`;
    
    if (item.status !== 'paused') {
      item.status = 'error';
      item.error = error.message?.includes('已取消') ? '已取消' : (error.message || errorMsg);
      item.canAbort = false;
      item.abortController = null;

      if (item.transferId && this.rootStore.transferStore) {
        await this.rootStore.transferStore.markTransferAsFailed(item.transferId, error.message || '上传失败');
      }
    }

    throw error;
  }
}
```

### 3.1 大文件MD5计算优化（Web Worker）

**文件**: `spug_web/public/md5.worker.js` (新建)

```javascript
// MD5计算Web Worker
importScripts('https://cdn.jsdelivr.net/npm/spark-md5@3.0.2/spark-md5.min.js');

self.onmessage = async function(e) {
  const { file } = e.data;
  
  try {
    const spark = new SparkMD5.ArrayBuffer();
    const chunkSize = 2 * 1024 * 1024; // 2MB分块读取
    const chunks = Math.ceil(file.size / chunkSize);
    
    for (let i = 0; i < chunks; i++) {
      const start = i * chunkSize;
      const end = Math.min(start + chunkSize, file.size);
      const chunk = file.slice(start, end);
      
      const arrayBuffer = await chunk.arrayBuffer();
      spark.append(arrayBuffer);
      
      // 报告进度
      self.postMessage({ 
        type: 'progress', 
        percent: Math.round(((i + 1) / chunks) * 100) 
      });
      
      // 让出主线程
      if (i % 5 === 0) {
        await new Promise(resolve => setTimeout(resolve, 0));
      }
    }
    
    const hash = spark.end();
    self.postMessage({ type: 'complete', hash });
  } catch (error) {
    self.postMessage({ type: 'error', message: error.message });
  }
};
```

**使用方式** (在 md5Store 中):

```javascript
// 【性能优化】使用Web Worker计算大文件MD5，避免阻塞主线程
async calculateFileMD5WithWorker(file) {
  return new Promise((resolve, reject) => {
    const worker = new Worker('/md5.worker.js');
    
    worker.onmessage = (e) => {
      const { type, hash, message, percent } = e.data;
      
      if (type === 'progress') {
        // 可以在这里更新MD5计算进度UI
        console.log(`MD5计算进度: ${percent}%`);
      } else if (type === 'complete') {
        resolve(hash);
        worker.terminate();
      } else if (type === 'error') {
        reject(new Error(message));
        worker.terminate();
      }
    };
    
    worker.onerror = (error) => {
      reject(error);
      worker.terminate();
    };
    
    worker.postMessage({ file });
  });
}
```

```javascript
resumeItemV2(itemId) { ... }

// 通过配置开关控制使用哪个版本
const USE_RESUME_V2 = localStorage.getItem('ENABLE_RESUME_V2') === 'true';

async resumeItem(itemId) {
  if (USE_RESUME_V2) {
    return this.resumeItemV2(itemId);

  }
}
```

## 六、风险评估与缓解措施

| 风险项 | 影响 | 缓解措施 |
|-------|------|---------|
| 分片检查接口异常 | 断点续传失效 | 边界处理：返回格式校验，失败时从头开始 |
| 单个分片上传失败 | 上传中断 | 重试机制：最多3次，失败后给出明确提示 |
| 文件重新选择错误 | 续传错误数据 | MD5校验：文件名+大小+内容三重校验 |
| 大文件MD5计算卡顿 | UI无响应 | Web Worker：后台线程计算，不阻塞主线程 |
| 浏览器不支持Worker | MD5计算失败 | 降级处理：回退到主线程计算 |
