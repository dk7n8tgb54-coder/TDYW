**文件**: `stores/upload/core/transfer.js`

```javascript
/**
 * 检查已上传分片（断点续传）- 含格式校验和降级处理
 * @param {string} fileHash - 文件MD5哈希
 * @param {number} fileSize - 文件大小
 * @param {number} totalChunks - 总分片数
 * @param {boolean} isPublic - 是否公共空间
 */
async checkUploadedChunks(fileHash, fileSize, totalChunks, isPublic = false) {
  try {
    const { http } = await import('libs');
    const result = await http.post(API_ENDPOINTS.CHECK_UPLOADED_CHUNKS, {
      file_hash: fileHash,
      file_size: fileSize,
      total_chunks: totalChunks,
      is_public: isPublic
    });
    
    // 【边界处理】校验后端返回格式
    if (!result || typeof result !== 'object') {
      console.warn('[TransferStore] 分片检查返回格式异常');
      return { exists: false, uploaded_chunks: [], count: 0 };
    }
    
    // 【边界处理】过滤无效分片索引
    if (Array.isArray(result.uploaded_chunks)) {
      const validChunks = result.uploaded_chunks.filter(idx => 
        Number.isInteger(idx) && idx >= 0 && idx < totalChunks
      );
      return { ...result, uploaded_chunks: validChunks, count: validChunks.length };
    }
    
    return result;
  } catch (error) {
    console.error('[TransferStore] 检查已上传分片失败:', error);
    // 【降级处理】失败时返回空列表，从头开始上传
    return { exists: false, uploaded_chunks: [], count: 0 };
  }
}
```

#### 4.1.2 修改 resumeItem 方法

**文件**: `stores/upload/core/index.js`

```javascript
@action
async resumeItem(itemId) {
  const item = this.queueStore.findUploadItemInCurrentTenant(itemId);
  if (!item) return;

  if (!['paused', 'error', 'waiting'].includes(item.status)) {
    return;
  }

  // 检查后端状态
  if (item.transferId) {
    try {
      const transfers = await this.transferStore.fetchTransfers(
        this.rootStore.navigationStore?.isPublic
      );
      const currentTransfer = transfers.find(t => t.id === item.transferId);
      
      if (currentTransfer) {
        if (currentTransfer.status === 'COMPLETED') {
          item.status = 'completed';
          item.percent = 100;
          item.error = null;
          return;
        }
        
        if (currentTransfer.status === 'CANCELED' || currentTransfer.status === 'FAILED') {
          item.status = 'error';
          item.error = currentTransfer.status === 'CANCELED' ? '已取消' : '失败';
          return;
        }
        
        // 同步后端的 file_hash
        if (currentTransfer.file_hash && !item.fileHash) {
          item.fileHash = currentTransfer.file_hash;
        }
      }
    } catch (error) {
      console.error('[resumeItem] 获取传输状态失败:', error);
    }
  }

  // 检查是否有文件对象
  if (!item.file || !(item.file instanceof File)) {
    message.warning(`"${item.name}" 需要重新选择文件才能继续上传`);
    item.error = '请重新选择文件';
    item.status = 'error';
    this.triggerFileReSelection(itemId);
    return;
  }

  // 【核心修复】大文件断点续传检查
  if (item.fileSize > UPLOAD_CONSTANTS.NORMAL_UPLOAD_THRESHOLD && item.fileHash) {
    try {
      message.loading('正在检查已上传分片...', 0);
      
      const chunkCount = Math.ceil(item.fileSize / UPLOAD_CONSTANTS.CHUNK_SIZE);
      const checkResult = await this.transferStore.checkUploadedChunks(
        item.fileHash,
        item.fileSize,
        chunkCount,
        item.isPublic
      );
      
      message.destroy();
      
      if (checkResult.exists && checkResult.uploaded_chunks.length > 0) {
        const uploadedCount = checkResult.uploaded_chunks.length;
        const uploadedBytes = uploadedCount * UPLOAD_CONSTANTS.CHUNK_SIZE;
        const percent = Math.min(Math.round((uploadedBytes / item.fileSize) * 100), 99);
        
        item.currentChunk = uploadedCount;
        item.percent = percent;
        item.chunks = checkResult.uploaded_chunks.map(idx => ({ index: idx, uploaded: true }));
        
        message.success(`已找到 ${uploadedCount} 个已上传分片，将从 ${percent}% 处继续`);
      }
    } catch (error) {
      message.destroy();
      console.error('[resumeItem] 检查已上传分片失败:', error);
      // 【降级处理】分片检查失败时，提示用户后从头上传
      message.warning('分片检查失败，将从头开始上传');
      item.currentChunk = 0;
      item.percent = 0;
      item.chunks = [];
    }
  }

  // 恢复上传
  item.status = 'uploading';
  item.error = null;
  item.canAbort = true;
  item.isPausedByUser = false;
  item.isCancelledByUser = false;
  item.abortController = new AbortController();

  const resumeFolderId = item.folderId !== null ? item.folderId : this.rootStore.navigationStore?.currentFolderId;
  await this.resumeChunkedUpload(item.file, resumeFolderId, item);
}
```

// stores/upload/core/index.js:442-481
removeAll = async () => {
  const activeStatuses = ['uploading', 'calculating', 'merging'];
  const transferIds = [];

  const uploadQueue = this.queueStore.uploadQueue;

  Object.keys(uploadQueue).forEach(tenantId => {
    const queue = uploadQueue[tenantId];
    if (!queue || !Array.isArray(queue)) return;

    const itemsToRemove = queue.filter(item => !activeStatuses.includes(item.status));

    itemsToRemove.forEach(item => {
      if (item.uniqueKey && item.file) {
        const { file, folderId } = item;
        const isPublic = item.isPublic !== undefined ? item.isPublic : this.rootStore.navigationStore?.isPublic;
        this.queueStore.removeUniqueKey(file, folderId, isPublic);
      }
      if (item.transferId) {
        transferIds.push(item.transferId);
      }
    });

    uploadQueue[tenantId] = queue.filter(item => activeStatuses.includes(item.status));
  });

  // 【优化】使用批量删除API，含loading状态
  if (transferIds.length > 0) {
    const hideLoading = message.loading(`正在清理 ${transferIds.length} 条记录...`, 0);
    try {
      const result = await this.transferStore.batchDeleteTransfers(transferIds);
      hideLoading();
      if (result) {
        message.success(`已清理 ${transferIds.length} 条传输记录`);
      } else {
        message.warning('批量删除部分失败');
      }
    } catch (error) {
      hideLoading();
      message.error(`批量删除失败: ${error.message || '请稍后重试'}`);
      // 【降级处理】补充重试建议

async batchDeleteTransfers(ids) {
  if (!ids || ids.length === 0) return { success: true, deleted_count: 0 };
  
  const hideLoading = message.loading('正在批量删除，请稍候...', 0);
  try {
    const { http } = await import('libs');
    const result = await http.post(API_ENDPOINTS.TRANSFER_BATCH_DELETE, { ids });
    hideLoading();
    return result;
  } catch (error) {
    hideLoading();
    // 【异常处理】批量删除失败后补充重试建议

class TransferListView(View):
    """获取用户的传输记录列表（支持分页）"""

    @auth('document.document.view')
    def get(self, request):
        from ...models import DocumentTransfer
        from ...constants import TransferStatus
        
        form, error = JsonParser(
            Argument('status', type=str, required=False, help='传输状态筛选'),
            Argument('transfer_type', type=str, required=False, help='传输类型筛选'),
            Argument('is_public', type=bool, required=False, help='是否公共空间'),
            Argument('page', type=int, required=False, default=1, help='页码'),
            Argument('page_size', type=int, required=False, default=20, help='每页数量'),
            Argument('keyword', type=str, required=False, help='文件名搜索关键词'),
        ).parse(request.GET)

        if error:
            return json_response(error=error)

        request_user = request.user
        page = max(1, form.page)
        page_size = min(max(1, form.page_size), 100)  # 限制1-100条
        offset = (page - 1) * page_size

        # 构建查询（【租户隔离】）
        request_tenant_id = getattr(request_user, 'tenant_id', '')
        queryset = DocumentTransfer.active_objects.filter(user=request_user)
        
        if hasattr(form, 'is_public') and form.is_public is not None:
            queryset = queryset.filter(is_public=form.is_public)
        
        if request_tenant_id:
            queryset = queryset.filter(tenant_id=request_tenant_id)

        # 筛选条件
        if form.status:
            queryset = queryset.filter(status=form.status)
        if form.transfer_type:
            queryset = queryset.filter(transfer_type=form.transfer_type)
        if form.keyword:
            queryset = queryset.filter(file_name__icontains=form.keyword)

        # 获取总数
        total = queryset.count()
        
        # 分页查询
        queryset = queryset.order_by('-created_at')[offset:offset+page_size]

        # 构建响应
        transfers = []
        for t in queryset:
            transfers.append({
                'id': t.id,
                'transfer_type': t.transfer_type,
                'status': t.status,
                'file_name': t.file_name,
                'file_size': t.file_size,
                'file_hash': t.file_hash or '',
                'progress': t.progress,
                'transferred_size': t.transferred_size,
                'speed': t.speed,
                'total_chunks': t.total_chunks,
                'uploaded_chunks': t.uploaded_chunks,
                'folder_id': t.folder_id,
                'is_public': t.is_public,
                'created_at': t.created_at.strftime('%Y-%m-%d %H:%M:%S') if t.created_at else None,
                'started_at': t.started_at.strftime('%Y-%m-%d %H:%M:%S') if t.started_at else None,
                'completed_at': t.completed_at.strftime('%Y-%m-%d %H:%M:%S') if t.completed_at else None,
                'error_message': t.error_message,
            })

        return json_response(data={
            'list': transfers,
            'total': total,
            'page': page,
            'page_size': page_size,
            'total_pages': (total + page_size - 1) // page_size
        })

  /**
   * 获取传输列表（支持分页和搜索）- 含加载状态管理
   */
  async fetchTransfers(isPublic = false, page = 1, pageSize = 20, keyword = '') {
    this.isLoading = true;  // 【用户体验】开始加载
    try {
      const { http } = await import('libs');
      const data = await http.get(API_ENDPOINTS.TRANSFER_LIST, {
        params: { 
          is_public: isPublic,
          page,
          page_size: pageSize,
          keyword
        }
      });
      
      this.transfers = data.list || [];
      this.pagination = {
        total: data.total || 0,
        page: data.page || 1,
        page_size: data.page_size || pageSize,
        total_pages: data.total_pages || 1
      };
      
      return data;
    } catch (error) {
      console.error('[TransferStore] 获取传输列表失败:', error);
      message.error('获取传输列表失败');
      return { list: [], total: 0, page: 1, page_size: pageSize };
    } finally {
      this.isLoading = false;  // 【用户体验】结束加载
    }
  }
}

### 4.7 状态幂等性处理

**文件**: `spug_api/apps/document/views/transfer/status.py`

```python
class TransferStatusUpdateView(View):
    """更新传输状态（含幂等性校验）"""

    @auth('document.document.upload')
    def post(self, request, transfer_id):
        from ...models import DocumentTransfer
        from ...constants import TransferStatus, is_valid_status_transition
        
        try:
            form, error = JsonParser(
                Argument('status', type=str, required=True, help='新状态'),
            ).parse(request.body)

            if error:
                return json_response(error=error)

            transfer = DocumentTransfer.objects.get(id=transfer_id)

            # 权限检查（【租户隔离】）
            request_tenant_id = getattr(request.user, 'tenant_id', '')
            is_supper = getattr(request.user, 'is_supper', False)

            if (transfer.user != request.user and not is_supper) or \
               (transfer.tenant_id != request_tenant_id and not is_supper):
                return json_response(error='无权更新此传输记录')

            # 【幂等性校验】如果当前状态已经是目标状态，直接返回成功
            current_status = transfer.status
            new_status = form.status
            
            if current_status == new_status:
                return json_response(data={'status': new_status, 'idempotent': True})

            # 状态流转验证
            current_status_enum = next((s for s in TransferStatus if s.value == current_status), None)
            new_status_enum = next((s for s in TransferStatus if s.value == new_status), None)

            if current_status_enum and new_status_enum:
                if not is_valid_status_transition(current_status_enum, new_status_enum):
                    return json_response(error=f'无效的状态转换：{current_status} -> {new_status}')
            else:
                return json_response(error='无效的状态值')

            transfer.status = new_status
            if new_status == TransferStatus.UPLOADING.value:
                if not transfer.started_at:
                    transfer.started_at = timezone.now()

            transfer.save()
            return json_response(data={'status': new_status})

        except DocumentTransfer.DoesNotExist:
            return json_response(error='传输记录不存在')
        except Exception as e:
            logger.error(f'[Document] Error updating transfer status: {e}')
            return json_response(error=f'更新传输状态失败: {str(e)}')
```

## 八、风险评估与缓解

| 风险点 | 影响 | 缓解措施 |
|-------|------|---------|
| 批量操作并发冲突 | 数据不一致 | 使用数据库事务 + 乐观锁 |
| 断点续传哈希不一致 | 续传失败 | 前后端哈希算法统一校验 |
| 分页切换频繁请求 | 后端压力大 | 前端缓存 + 防抖处理 |
| 状态幂等性遗漏 | 用户看到错误提示 | 前后端双保险幂等校验 |
| 硬删除数据无法恢复 | 误删后无法找回 | 定期数据库备份 |
