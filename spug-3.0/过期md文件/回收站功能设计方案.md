## 一、需求分析

### 1.1 现状问题

当前软删除机制已在后台实现，但用户无法：
- 查看已删除的文件
- 恢复误删的文件
- 主动彻底删除文件

### 1.2 业务需求

| 角色 | 需求 | 优先级 |
|------|------|--------|
| 普通用户 | 查看自己删除的文件 | P0 |
| 普通用户 | 恢复误删文件 | P0 |
| 普通用户 | 主动彻底删除文件 | P1 |
| 管理员 | 查看所有用户回收站 | P2 |
| 管理员 | 强制清理回收站 | P2 |

### 1.3 设计原则

1. **用户自助**: 减少管理员介入
2. **分层权限**: 用户只能操作自己的文件
3. **保留期限**: 30天自动清理（可配置）
4. **操作可追溯**: 记录恢复/彻底删除操作

## 三、技术方案

### 3.1 后端API设计

#### 3.1.1 统一错误码规范

遵循企业级API错误码设计规范，便于前端处理和问题定位：

| 错误码 | 类别 | 含义 | 触发场景 |
|--------|------|------|----------|
| 400001 | 参数错误 | 请求参数缺失或格式错误 | 缺少必填参数 |
| 400002 | 批量超限 | 批量操作数量超过限制 | file_ids > 50 |
| 403001 | 权限不足 | 无操作权限 | 非文件所有者/非管理员 |
| 404001 | 资源不存在 | 文件不存在或未被删除 | 文件ID错误 |
| 404002 | 目标不存在 | 目标文件夹不存在 | custom模式folder_id无效 |
| 409001 | 并发冲突 | 操作冲突（乐观锁） | 并发恢复同一文件 |
| 429001 | 请求限流 | 操作过于频繁 | 触发限流保护 |
| 500001 | 系统错误 | 内部系统错误 | 数据库/文件系统异常 |

**错误响应格式**：
```json
{
  "error": "批量操作数量超过限制",
  "code": 400002,
  "detail": "单次最多支持50个文件"
}
```

#### 3.1.2 API 列表

| 方法 | 路径 | 功能 | 权限 | 幂等性 |
|------|------|------|------|--------|
| GET | `/api/document/recycle-bin/` | 获取回收站列表 | 登录用户 | - |
| POST | `/api/document/recycle-bin/restore/` | 恢复文件 | 文件所有者 | ✅ 支持 |
| POST | `/api/document/recycle-bin/permanent/` | 彻底删除 | 文件所有者 | ✅ 支持 |
| GET | `/api/document/recycle-bin/stats/` | 回收站统计 | 登录用户 | - |

```python
class DocumentFilePrivate(models.Model):

    def restore(self):
        """
        【回收站】恢复软删除的文件
        
        Returns:
            bool: 是否成功恢复
        """
        if not self.is_deleted:
            return False
        
        self.is_deleted = False
        self.deleted_at = None
        self.save(update_fields=['is_deleted', 'deleted_at'])
        return True
    
    def delete(self, hard=False, using=None, keep_parents=False):
        """
        【回收站】删除文件
        
        Args:
            hard: 是否硬删除（默认False软删除）
        
        Returns:
            tuple: (删除记录数, 删除对象数)
        """
        if hard:
            # 硬删除：删除物理文件 + 数据库记录
            try:
                if os.path.exists(self.file_path):
                    os.remove(self.file_path)
                    logger.info(f'[RecycleBin] 物理文件已删除: {self.file_path}')
            except Exception as e:
                logger.error(f'[RecycleBin] 删除物理文件失败: {self.file_path}, error={e}')
            
            # 调用父类delete真正删除数据库记录
            return super().delete(using=using, keep_parents=keep_parents)
        else:
            # 软删除：仅标记is_deleted
            self.is_deleted = True
            self.deleted_at = timezone.now()
            self.save(update_fields=['is_deleted', 'deleted_at'])
            return (1, {self._meta.label: 1})
```

#### RecycleBinView 实现

```python
# apps/document/views/recycle_bin.py

from django.views.generic import View
from django.core.paginator import Paginator
from libs import json_response, JsonParser, Argument, auth
from ..models import DocumentFilePrivate, DocumentFilePublic


class RecycleBinView(View):
    """回收站视图（性能优化版）"""
    
    # 缓存配置
    CACHE_TTL = 60  # 缓存60秒
    
    @auth('document.document.view')
    def get(self, request):
        """获取回收站列表（数据库层面分页+缓存优化）"""
        form, error = JsonParser(
            Argument('page', type=int, required=False, default=1),
            Argument('page_size', type=int, required=False, default=20),
            Argument('keyword', required=False),
            Argument('space', required=False, default='all', 
                     filter=lambda x: x in ['private', 'public', 'all'])
        ).parse(request.GET)
        
        if error:
            return json_response(error=error)
        
        # 限制最大分页数量
        page_size = min(form.page_size, 100)
        
        # 构建缓存键
        cache_key = f'recycle_bin:{request.user.id}:{form.space}:{form.keyword or ""}:{form.page}:{page_size}'
        
        # 尝试从缓存获取
        cached_data = cache.get(cache_key)
        if cached_data:
            return json_response(data=cached_data)
        
        # 数据库层面分页查询（优化后）
        result = self._get_paginated_results(
            request.user, form.space, form.keyword, form.page, page_size
        )
        
        # 写入缓存
        cache.set(cache_key, result, self.CACHE_TTL)
        
        return json_response(data=result)
    
    def _get_paginated_results(self, user, space, keyword, page, page_size):
        """
        优化的分页查询：数据库层面分页，避免内存溢出
        支持10万+数据量
        """
        offset = (page - 1) * page_size
        limit = page_size * 2  # 各取2倍确保合并后够page_size条
        
        results = []
        total_private = 0
        total_public = 0
        
        # 私有空间查询（select_related优化N+1）
        if space in ['private', 'all']:
            private_qs = DocumentFilePrivate.all_objects.filter(
                is_deleted=True
            ).select_related('folder', 'created_by')  # 【优化】select_related避免N+1
            
            if not user.is_superuser:
                private_qs = private_qs.filter(created_by=user)
            
            if keyword:
                private_qs = private_qs.filter(
                    Q(display_name__icontains=keyword) | 
                    Q(name__icontains=keyword)
                )
            
            total_private = private_qs.count()
            # 【优化】数据库层面LIMIT，只取需要的数量
            private_files = private_qs.order_by('-deleted_at')[:limit]
            results.extend([self._format_file(f, 'private') for f in private_files])
        
        # 公共空间查询（select_related优化N+1）
        if space in ['public', 'all']:
            public_qs = DocumentFilePublic.all_objects.filter(
                is_deleted=True
            ).select_related('folder', 'created_by')  # 【优化】select_related避免N+1
            
            if not user.is_superuser:
                public_qs = public_qs.filter(created_by=user)
            
            if keyword:
                public_qs = public_qs.filter(
                    Q(display_name__icontains=keyword) | 
                    Q(name__icontains=keyword)
                )
            
            total_public = public_qs.count()
            # 【优化】数据库层面LIMIT
            public_files = public_qs.order_by('-deleted_at')[:limit]
            results.extend([self._format_file(f, 'public') for f in public_files])
        
        # 内存排序（数据量已很小，最多200条）
        results.sort(key=lambda x: x['deleted_at'], reverse=True)
        
        # 精确分页
        paginated_results = results[offset:offset + page_size]
        total_count = total_private + total_public if space == 'all' else (total_private or total_public)
        
        return {
            'items': paginated_results,
            'total': total_count,
            'page': page,
            'page_size': page_size
        }
    
    def _format_file(self, file_obj, space):
        """格式化文件信息"""
        from datetime import timedelta
        from django.conf import settings
        
        # 从配置读取保留天数，默认30天
        retention_days = getattr(settings, 'RECYCLE_BIN_RETENTION_DAYS', 30)
        days_left = retention_days - (timezone.now() - file_obj.deleted_at).days
        
        return {
            'id': file_obj.id,
            'name': file_obj.name,
            'display_name': file_obj.display_name,
            'file_size': file_obj.file_size,
            'file_type': file_obj.file_type,
            'space': space,
            'deleted_at': file_obj.deleted_at.isoformat(),
            'retention_days_left': max(0, days_left),
            'original_folder': {
                'id': file_obj.folder.id,
                'name': file_obj.folder.name
            } if file_obj.folder else None,
            'created_by': {
                'id': file_obj.created_by.id,
                'nickname': file_obj.created_by.nickname
            } if file_obj.created_by else None
        }


class RecycleBinRestoreView(View):
    """恢复文件视图"""
    
    # 限流配置：每用户每分钟最多10次恢复操作
    RATE_LIMIT = {'requests': 10, 'window': 60}
    
    @auth('document.document.edit')
    def post(self, request):
        """恢复软删除的文件（支持幂等）"""
        # 限流检查
        if not self._check_rate_limit(request.user.id):
            return json_response(error='操作过于频繁，请稍后再试', code=429001)
        
        form, error = JsonParser(
            Argument('file_ids', type=list, required=True),
            Argument('target_folder_id', type=int, required=False),
            Argument('current_folder_id', type=int, required=False),
            Argument('restore_mode', required=False, default='original',
                     filter=lambda x: x in ['original', 'current', 'custom']),
            Argument('idempotent_key', required=False)  # 幂等键
        ).parse(request.body)
        
        if error:
            return json_response(error=error, code=400001)
        
        # 限制批量操作数量
        batch_limit = getattr(settings, 'RECYCLE_BIN_BATCH_LIMIT', 50)
        if len(form.file_ids) > batch_limit:
            return json_response(
                error=f'批量恢复最多支持{batch_limit}个文件', 
                code=400002
            )
        
        # 幂等性检查（防止重复提交）
        if form.idempotent_key:
            cache_key = f'recycle_bin:restore:{form.idempotent_key}'
            if cache.get(cache_key):
                # 返回已处理的结果
                return json_response(data=cache.get(cache_key))
        
        results = []
        success_count = 0
        start_time = time.time()
        
        from django.db import transaction
        
        try:
            with transaction.atomic():
                for file_id in form.file_ids:
                    result = self._restore_file(
                        file_id, 
                        request.user,
                        form.restore_mode,
                        form.target_folder_id,
                        form.current_folder_id
                    )
                    results.append(result)
                    if result['status'] == 'success':
                        success_count += 1
        except Exception as e:
            logger.error(f'[RecycleBin] 批量恢复事务失败: {e}')
            return json_response(error=f'恢复操作失败: {str(e)}', code=500001)
        
        # 记录监控指标
        duration = time.time() - start_time
        self._record_metrics('restore', success_count, len(form.file_ids), duration)
        
        response_data = {
            'success_count': success_count,
            'failed_count': len(form.file_ids) - success_count,
            'details': results
        }
        
        # 缓存幂等结果（5分钟）
        if form.idempotent_key:
            cache.set(cache_key, response_data, 300)
        
        return json_response(data=response_data)
    
    def _check_rate_limit(self, user_id):
        """检查限流"""
        key = f'rate_limit:restore:{user_id}'
        current = cache.get(key, 0)
        if current >= self.RATE_LIMIT['requests']:
            return False
        cache.set(key, current + 1, self.RATE_LIMIT['window'])
        return True
    
    def _record_metrics(self, operation, success, total, duration):
        """记录监控指标"""
        logger.info(
            f'[RecycleBinMetrics] operation={operation}, '
            f'success={success}, total={total}, duration={duration:.3f}s'
        )
    
    def _restore_file(self, file_id, user, mode, target_folder_id, current_folder_id):
        """恢复单个文件（自动识别私有/公共空间）"""
        # 先尝试私有空间
        try:
            file_obj = DocumentFilePrivate.all_objects.get(
                id=file_id,
                is_deleted=True
            )
            return self._restore_private_file(file_obj, user, mode, target_folder_id, current_folder_id)
        except DocumentFilePrivate.DoesNotExist:
            pass
        
        # 再尝试公共空间
        try:
            file_obj = DocumentFilePublic.all_objects.get(
                id=file_id,
                is_deleted=True
            )
            return self._restore_public_file(file_obj, user, mode, target_folder_id, current_folder_id)
        except DocumentFilePublic.DoesNotExist:
            return {
                'id': file_id,
                'status': 'failed',
                'error': '文件不存在或未被删除'
            }
    
    def _restore_private_file(self, file_obj, user, mode, target_folder_id, current_folder_id):
        """恢复私有空间文件（带乐观锁）"""
        # 权限校验
        if not self._check_permission(file_obj, user):
            return {
                'id': file_obj.id,
                'status': 'failed',
                'error': '无操作权限',
                'code': 403001
            }
        
        # 乐观锁检查：确保文件未被并发恢复
        from django.db.models import F
        updated = DocumentFilePrivate.all_objects.filter(
            id=file_obj.id,
            is_deleted=True,
            version=file_obj.version if hasattr(file_obj, 'version') else 0
        ).update(is_deleted=False, version=F('version') + 1)
        
        if not updated:
            return {
                'id': file_obj.id,
                'status': 'failed',
                'error': '文件已被恢复或正在处理中',
                'code': 409001
            }
        
        # 重新加载对象（获取更新后的状态）
        file_obj.refresh_from_db()
        
        # 检查原文件夹是否存在
        original_folder_exists = True
        if file_obj.folder:
            try:
                DocumentFolderPrivate.objects.get(id=file_obj.folder.id)
            except DocumentFolderPrivate.DoesNotExist:
                original_folder_exists = False
        
        # 恢复模式处理
        if mode == 'original':
            if not original_folder_exists:
                file_obj.folder = None
                logger.warning(f'[RecycleBin] 原文件夹不存在，文件{file_obj.id}恢复到根目录')
        
        elif mode == 'current':
            if current_folder_id:
                try:
                    target_folder = DocumentFolderPrivate.objects.get(id=current_folder_id)
                    file_obj.folder = target_folder
                except DocumentFolderPrivate.DoesNotExist:
                    file_obj.folder = None
                    logger.warning(f'[RecycleBin] 当前文件夹不存在，文件{file_obj.id}恢复到根目录')
            else:
                file_obj.folder = None
        
        elif mode == 'custom' and target_folder_id:
            try:
                target_folder = DocumentFolderPrivate.objects.get(id=target_folder_id)
                file_obj.folder = target_folder
            except DocumentFolderPrivate.DoesNotExist:
                return {
                    'id': file_obj.id,
                    'status': 'failed',
                    'error': '指定文件夹不存在',
                    'code': 404002
                }
        
        # 处理同名冲突
        from ..libs.naming_utils import generate_unique_logical_name
        file_obj.name = generate_unique_logical_name(
            DocumentFilePrivate,
            file_obj.display_name or file_obj.name,
            file_obj.folder,
            user
        )
        file_obj.save(update_fields=['folder', 'name', 'deleted_at'])
        
        # 记录审计日志
        from .base import log_operation
        log_operation(
            action="FILE_RESTORE",
            user=user,
            resource_type="FILE",
            resource_id=file_obj.id,
            is_public=False,
            restore_mode=mode
        )
        
        return {
            'id': file_obj.id,
            'status': 'success',
            'restored_id': file_obj.id
        }
    
    def _restore_public_file(self, file_obj, user, mode, target_folder_id, current_folder_id):
        """恢复公共空间文件"""
        # 权限校验（公共空间只有管理员或创建者可恢复）
        if not self._check_permission(file_obj, user):
            return {
                'id': file_obj.id,
                'status': 'failed',
                'error': '无操作权限'
            }
        
        # 公共空间恢复逻辑类似，但使用DocumentFolderPublic
        original_folder_exists = True
        if file_obj.folder:
            try:
                DocumentFolderPublic.objects.get(id=file_obj.folder.id)
            except DocumentFolderPublic.DoesNotExist:
                original_folder_exists = False
        
        if mode == 'original':
            if not original_folder_exists:
                file_obj.folder = None
                logger.warning(f'[RecycleBin] 原公共文件夹不存在，文件{file_obj.id}恢复到根目录')
        
        elif mode == 'current':
            if current_folder_id:
                try:
                    target_folder = DocumentFolderPublic.objects.get(id=current_folder_id)
                    file_obj.folder = target_folder
                except DocumentFolderPublic.DoesNotExist:
                    file_obj.folder = None
                    logger.warning(f'[RecycleBin] 当前公共文件夹不存在，文件{file_obj.id}恢复到根目录')
            else:
                file_obj.folder = None
        
        elif mode == 'custom' and target_folder_id:
            try:
                target_folder = DocumentFolderPublic.objects.get(id=target_folder_id)
                file_obj.folder = target_folder
            except DocumentFolderPublic.DoesNotExist:
                return {
                    'id': file_obj.id,
                    'status': 'failed',
                    'error': '指定文件夹不存在'
                }
        
        # 恢复文件
        file_obj.restore()
        
        # 处理同名冲突
        from ..libs.naming_utils import generate_unique_logical_name
        file_obj.name = generate_unique_logical_name(
            DocumentFilePublic,
            file_obj.display_name or file_obj.name,
            file_obj.folder,
            user
        )
        file_obj.save(update_fields=['folder', 'name'])
        
        # 记录审计日志
        from .base import log_operation
        log_operation(
            action="FILE_RESTORE",
            user=user,
            resource_type="FILE",
            resource_id=file_obj.id,
            is_public=True,
            restore_mode=mode
        )
        
        return {
            'id': file_obj.id,
            'status': 'success',
            'restored_id': file_obj.id
        }
    
    def _check_permission(self, file_obj, user):
        """检查用户是否有权限操作文件"""
        # 管理员有所有权限
        if user.is_superuser:
            return True
        
        # 文件所有者有权操作
        if file_obj.created_by == user:
            return True
        
        return False


class RecycleBinPermanentDeleteView(View):
    """彻底删除视图（支持异步批量操作）"""
    
    # 大文件阈值：100MB，超过则异步删除
    LARGE_FILE_THRESHOLD = 100 * 1024 * 1024
    
    @auth('document.document.delete')
    def post(self, request):
        """彻底删除文件（支持单个、批量、异步）"""
        form, error = JsonParser(
            Argument('file_ids', type=list, required=True),
            Argument('async_mode', type=bool, required=False, default=False)  # 强制异步模式
        ).parse(request.body)
        
        if error:
            return json_response(error=error)
        
        batch_limit = getattr(settings, 'RECYCLE_BIN_BATCH_LIMIT', 50)
        if len(form.file_ids) > batch_limit:
            return json_response(error=f'批量删除最多支持{batch_limit}个文件')
        
        if len(form.file_ids) == 0:
            return json_response(error='请选择要删除的文件')
        
        # 判断是否需要异步处理（大文件或批量）
        total_size = self._calculate_total_size(form.file_ids)
        need_async = form.async_mode or total_size > self.LARGE_FILE_THRESHOLD or len(form.file_ids) > 10
        
        if need_async:
            # 【优化】异步批量删除，避免阻塞用户
            task_id = async_batch_permanent_delete.delay(form.file_ids, request.user.id)
            return json_response(data={
                'async': True,
                'task_id': str(task_id),
                'message': '删除任务已提交，请稍后查看结果'
            })
        
        # 同步删除（小文件）
        results = []
        total_freed = 0
        
        from django.db import transaction
        
        try:
            with transaction.atomic():
                for file_id in form.file_ids:
                    result = self._permanent_delete(file_id, request.user)
                    results.append(result)
                    if result['status'] == 'success':
                        total_freed += result.get('file_size', 0)
        except Exception as e:
            logger.error(f'[RecycleBin] 批量删除事务失败: {e}')
            return json_response(error=f'删除操作失败: {str(e)}')
        
        # 清除缓存
        self._invalidate_cache(request.user.id)
        
        return json_response(data={
            'async': False,
            'success_count': sum(1 for r in results if r['status'] == 'success'),
            'failed_count': sum(1 for r in results if r['status'] == 'failed'),
            'freed_space': total_freed,
            'details': results
        })
    
    def _calculate_total_size(self, file_ids):
        """计算文件总大小"""
        total = 0
        for file_id in file_ids:
            try:
                file_obj = DocumentFilePrivate.all_objects.get(id=file_id, is_deleted=True)
                total += file_obj.file_size
            except DocumentFilePrivate.DoesNotExist:
                try:
                    file_obj = DocumentFilePublic.all_objects.get(id=file_id, is_deleted=True)
                    total += file_obj.file_size
                except DocumentFilePublic.DoesNotExist:
                    pass
        return total
    
    def _invalidate_cache(self, user_id):
        """清除用户回收站缓存"""
        # 使用通配符清除该用户的所有回收站缓存
        cache.delete_pattern(f'recycle_bin:{user_id}:*')
    
    def _permanent_delete(self, file_id, user):
        """彻底删除单个文件"""
        try:
            # 先尝试私有空间
            try:
                file_obj = DocumentFilePrivate.all_objects.get(
                    id=file_id,
                    is_deleted=True
                )
            except DocumentFilePrivate.DoesNotExist:
                # 再尝试公共空间
                file_obj = DocumentFilePublic.all_objects.get(
                    id=file_id,
                    is_deleted=True
                )
            
            # 权限校验
            if not self._check_permission(file_obj, user):
                return {
                    'id': file_id,
                    'status': 'failed',
                    'error': '无操作权限'
                }
            
            file_size = file_obj.file_size
            
            # 调用硬删除（删除物理文件+数据库记录）
            file_obj.delete(hard=True)
            
            # 记录审计日志
            from .base import log_operation
            log_operation(
                action="FILE_PERMANENT_DELETE",
                user=user,
                resource_type="FILE",
                resource_id=file_id,
                is_public=isinstance(file_obj, DocumentFilePublic),
                file_size=file_size
            )
            
            return {
                'id': file_id,
                'status': 'success',
                'file_size': file_size
            }
            
        except (DocumentFilePrivate.DoesNotExist, DocumentFilePublic.DoesNotExist):
            return {
                'id': file_id,
                'status': 'failed',
                'error': '文件不存在或未被删除'
            }
        except Exception as e:
            logger.error(f'[RecycleBin] 彻底删除文件失败: file_id={file_id}, error={e}')
            return {
                'id': file_id,
                'status': 'failed',
                'error': str(e)
            }
    
    def _check_permission(self, file_obj, user):
        """检查用户是否有权限操作文件"""
        # 管理员有所有权限
        if user.is_superuser:
            return True
        
        # 文件所有者有权操作
        if file_obj.created_by == user:
            return True
        
        return False


class RecycleBinStatsView(View):
    """回收站统计视图"""
    
    @auth('document.document.view')
    def get(self, request):
        """获取回收站统计信息"""
        from django.db.models import Sum
        from datetime import timedelta
        from django.conf import settings
        
        retention_days = getattr(settings, 'RECYCLE_BIN_RETENTION_DAYS', 30)
        expiring_threshold = timezone.now() - timedelta(days=retention_days - 7)
        
        # 私有文件统计
        private_queryset = DocumentFilePrivate.all_objects.filter(is_deleted=True)
        if not request.user.is_superuser:
            private_queryset = private_queryset.filter(created_by=request.user)
        
        private_total = private_queryset.count()
        private_size = private_queryset.aggregate(
            total=Sum('file_size')
        )['total'] or 0
        
        # 即将过期（7天内）
        private_expiring = private_queryset.filter(
            deleted_at__gte=expiring_threshold
        ).count()
        
        # 公共空间统计（仅管理员）
        public_total = 0
        public_size = 0
        public_expiring = 0
        
        if request.user.is_superuser:
            public_queryset = DocumentFilePublic.all_objects.filter(is_deleted=True)
            public_total = public_queryset.count()
            public_size = public_queryset.aggregate(
                total=Sum('file_size')
            )['total'] or 0
            public_expiring = public_queryset.filter(
                deleted_at__gte=expiring_threshold
            ).count()
        else:
            # 普通用户只看自己创建的公共文件
            public_queryset = DocumentFilePublic.all_objects.filter(
                is_deleted=True,
                created_by=request.user
            )
            public_total = public_queryset.count()
            public_size = public_queryset.aggregate(
                total=Sum('file_size')
            )['total'] or 0
            public_expiring = public_queryset.filter(
                deleted_at__gte=expiring_threshold
            ).count()
        
        return json_response(data={
            'total_count': private_total + public_total,
            'total_size': private_size + public_size,
            'private_count': private_total,
            'private_size': private_size,
            'public_count': public_total,
            'public_size': public_size,
            'expiring_soon': private_expiring + public_expiring,
            'retention_days': retention_days
        })
```

### 3.6 异步批量删除任务

```python
# apps/document/tasks/cleanup.py

from celery import shared_task
from django.contrib.auth import get_user_model

User = get_user_model()


@shared_task(queue='document.cleanup', bind=True)
def async_batch_permanent_delete(self, file_ids, user_id):
    """
    异步批量彻底删除文件（用于大文件或大批量删除）
    避免阻塞前端用户操作
    """
    logger.info(f'[AsyncDelete] 开始异步删除任务: task_id={self.request.id}, files={len(file_ids)}')
    
    try:
        user = User.objects.get(id=user_id)
    except User.DoesNotExist:
        logger.error(f'[AsyncDelete] 用户不存在: user_id={user_id}')
        return {'status': 'failed', 'error': '用户不存在'}
    
    results = []
    success_count = 0
    failed_count = 0
    total_freed = 0
    
    # 分批处理，每批10个，避免单次事务过大
    batch_size = 10
    for i in range(0, len(file_ids), batch_size):
        batch = file_ids[i:i + batch_size]
        
        try:
            with transaction.atomic():
                for file_id in batch:
                    result = _permanent_delete_single(file_id, user)
                    results.append(result)
                    
                    if result['status'] == 'success':
                        success_count += 1
                        total_freed += result.get('file_size', 0)
                    else:
                        failed_count += 1
        except Exception as e:
            logger.error(f'[AsyncDelete] 批次处理失败: batch={batch}, error={e}')
            for file_id in batch:
                results.append({'id': file_id, 'status': 'failed', 'error': str(e)})
                failed_count += 1
        
        # 更新任务进度
        progress = min(100, int((i + len(batch)) / len(file_ids) * 100))
        self.update_state(
            state='PROGRESS',
            meta={'progress': progress, 'processed': i + len(batch), 'total': len(file_ids)}
        )
    
    logger.info(
        f'[AsyncDelete] 任务完成: task_id={self.request.id}, '
        f'success={success_count}, failed={failed_count}, freed={total_freed}'
    )
    
    return {
        'status': 'completed',
        'success_count': success_count,
        'failed_count': failed_count,
        'freed_space': total_freed,
        'details': results
    }


def _permanent_delete_single(file_id, user):
    """删除单个文件（供异步任务使用）"""
    try:
        # 尝试私有空间
        try:
            file_obj = DocumentFilePrivate.all_objects.get(id=file_id, is_deleted=True)
        except DocumentFilePrivate.DoesNotExist:
            file_obj = DocumentFilePublic.all_objects.get(id=file_id, is_deleted=True)
        
        # 权限校验
        if not user.is_superuser and file_obj.created_by != user:
            return {'id': file_id, 'status': 'failed', 'error': '无操作权限'}
        
        file_size = file_obj.file_size
        file_obj.delete(hard=True)
        
        return {'id': file_id, 'status': 'success', 'file_size': file_size}
        
    except Exception as e:
        logger.error(f'[AsyncDelete] 删除文件失败: file_id={file_id}, error={e}')
        return {'id': file_id, 'status': 'failed', 'error': str(e)}
```

```python
def _check_permission(self, file_obj, user):
    """检查用户是否有权限操作文件"""
    # 管理员有所有权限
```
