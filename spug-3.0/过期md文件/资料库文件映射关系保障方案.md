# 资料库文件映射关系保障方案

> **版本**: v1.0  
> **适用**: Spug 资料库模块  
> **目标**: 确保数据库与物理文件映射关系在升级、迁移、备份等场景下永久稳定

---

## 一、核心铁律（不可违背）

### 四大原则

| 原则 | 说明 | 违规后果 |
|------|------|----------|
| **物理名终身不变** | `physical_name` 生成后永不允许修改 | 映射断裂，文件无法定位 |
| **路径终身不变** | `file_path` 生成后永不允许修改 | 文件丢失，404错误 |
| **只写一次只读** | 物理标识字段写入后设为只读 | 人为误操作导致混乱 |
| **操作驱动优先** | 所有文件操作必须通过数据库记录驱动 | 直接操作磁盘导致不同步 |

### 核心公式

```
物理文件名 = 唯一主键，一旦生成，终身不变
├─ 禁止：修改、重命名、迁移目录
└─ 允许：复制（生成新物理名）、软删除
```

```python
    # ========== 业务标识字段（可修改）==========
    display_name = models.CharField(

    )

        max_length=64, 
        help_text='逻辑文件名（API交互用，可修改）',

        related_name='files'

    file_size = models.BigIntegerField(default=0, verbose_name='文件大小(字节)')
    file_type = models.CharField(max_length=100, verbose_name='文件类型')
    created_by = models.ForeignKey(User, on_delete=models.SET_NULL, null=True, verbose_name='上传人')
    tenant_id = models.CharField(max_length=50, default='', help_text='租户标识')
    
    # ========== 软删除字段（推荐）==========
    is_deleted = models.BooleanField(default=False, help_text='是否已删除（软删除）')
    deleted_at = models.DateTimeField(null=True, blank=True, help_text='删除时间')
    deleted_by = models.ForeignKey(
        User, 
        on_delete=models.SET_NULL, 
        null=True, 
        blank=True,
        related_name='deleted_files',
        help_text='删除人'
    )
    
    created_at = models.DateTimeField(auto_now_add=True, verbose_name='上传时间')
    updated_at = models.DateTimeField(auto_now=True, verbose_name='更新时间')

    class Meta:
        db_table = 'spug_document_file_private'
        verbose_name = '文档文件(私有)'
        verbose_name_plural = '文档文件(私有)'
        ordering = ['-created_at']
        
        # ========== 核心约束：确保映射唯一性 ==========
        constraints = [
            # 同一租户内物理文件名唯一（防止重复存储）
            models.UniqueConstraint(
                fields=['physical_name', 'tenant_id'],
                name='unique_physical_name_per_tenant_private'
            ),
        ]
        
        indexes = [
            # 租户+文件夹查询优化
            models.Index(fields=['tenant_id', 'folder', 'is_deleted'], 
                        name='idx_tenant_folder_active'),
            # 物理文件名查询（用于校验）
            models.Index(fields=['physical_name'], name='idx_physical_name'),
        ]

    def __str__(self):
        return self.display_name or self.name
    
    def delete(self, *args, **kwargs):
        """重写删除方法：默认软删除"""
        hard = kwargs.pop('hard', False)
        if hard:
            # 硬删除：调用父类方法
            super().delete(*args, **kwargs)
        else:
            # 软删除：标记删除状态
            self.is_deleted = True
            self.deleted_at = timezone.now()
            self.save(update_fields=['is_deleted', 'deleted_at'])
```

|------|--------|----------|--------|--------|--------|------|
| **上传** | 生成新的 | 生成新的 | 生成新的 | 复制原文件名 | 指定 | ✅ 标准流程 |
| **复制** | 生成新的 | 生成新的 | 生成新的 | 复制/修改 | 指定 | ✅ 生成全新记录 |
| **移动** | ❌ 不动 | ❌ 不动 | 可选修改 | 可选修改 | ✅ 修改 | ⚠️ 只改folder_id |

```python
def upload_file(file_obj, folder, user, **kwargs):
    """
    文件上传标准流程
    
    原则：
    1. 先生成所有命名
    2. 再写入数据库
    3. 最后写入磁盘
    4. 任何步骤失败回滚
    """
    from apps.document.libs.naming_utils import generate_file_names
    
    original_name = file_obj.name
    
    # 1. 生成三层命名（一次性完成）
    names = generate_file_names(
        original_name=original_name,
        folder=folder,
        user=user
    )
    
    # 2. 构建完整存储路径
    file_path = build_storage_path(
        tenant_id=user.tenant_id,
        folder=folder,
        physical_name=names['physical_name']
    )
    
    # 3. 创建数据库记录（先写库，确保映射关系建立）
    try:
        with transaction.atomic():
            file_record = DocumentFilePrivate.objects.create(
                # 物理标识（只写一次）
                physical_name=names['physical_name'],
                file_path=file_path,
                
                # 业务标识
                name=names['logical_name'],
                display_name=names['display_name'],
                
                # 文件信息
                folder=folder,
                file_size=file_obj.size,
                file_type=file_obj.content_type,
                created_by=user,
                tenant_id=user.tenant_id,
                
                # 软删除状态
                is_deleted=False,
            )
    except IntegrityError:
        # 物理名冲突（理论上不会发生，因有时间戳+随机数）
        raise Exception("文件生成冲突，请重试")
    
    # 4. 写入物理文件（后写磁盘）
    try:
        save_file_to_disk(file_obj, file_path)
    except Exception as e:
        # 磁盘写入失败，删除数据库记录（回滚）
        file_record.delete(hard=True)
        raise Exception(f"文件保存失败: {e}")
    
    # 5. 记录审计日志
    log_file_operation('UPLOAD', file_record, user)
    
    return file_record
```

```python
def copy_file(source_file, target_folder, user, new_display_name=None):
    """
    文件复制标准流程
    
    原则：
    1. 生成全新物理名（不共用物理文件）
    2. 创建全新数据库记录
    3. 复制物理文件到新路径
    4. 原文件完全不动
    """
    from apps.document.libs.naming_utils import generate_file_names
    
    # 1. 生成全新命名（绝不复用原physical_name）
    display_name = new_display_name or f"{source_file.display_name}_副本"
    names = generate_file_names(
        original_name=display_name,
        folder=target_folder,
        user=user
    )
    
    # 2. 构建新路径
    new_file_path = build_storage_path(
        tenant_id=user.tenant_id,
        folder=target_folder,
        physical_name=names['physical_name']
    )
    
    # 3. 创建新数据库记录
    with transaction.atomic():
        new_file = DocumentFilePrivate.objects.create(
            # 全新物理标识
            physical_name=names['physical_name'],
            file_path=new_file_path,
            
            # 业务标识可继承或修改
            name=names['logical_name'],
            display_name=display_name,
            
            # 其他信息继承
            folder=target_folder,
            file_size=source_file.file_size,
            file_type=source_file.file_type,
            created_by=user,
            tenant_id=user.tenant_id,
        )
    
    # 4. 复制物理文件
    try:
        shutil.copy2(source_file.file_path, new_file_path)
    except Exception as e:
        # 复制失败，删除新记录
        new_file.delete(hard=True)
        raise Exception(f"文件复制失败: {e}")
    
    # 5. 记录审计日志
    log_file_operation('COPY', new_file, user, source=source_file)
    
    return new_file
```

#### 删除操作（软删除推荐）

```python
def delete_file(file_record, user, hard=False):
    """
    文件删除标准流程
    
    原则：
    - 软删除：标记删除状态，物理文件保留一段时间
    - 硬删除：先删数据库，再删物理文件
    """
    if hard:
        # ====== 硬删除 ======
        file_path = file_record.file_path
        physical_name = file_record.physical_name
        
        # 1. 先删数据库记录（解除映射关系）
        file_record.delete(hard=True)
        
        # 2. 再删物理文件
        try:
            if os.path.exists(file_path):
                os.remove(file_path)
                logger.info(f"物理文件已删除: {physical_name} -> {file_path}")
        except Exception as e:
            # 物理文件删除失败，记录日志（不影响业务）
            logger.error(f"物理文件删除失败: {physical_name}, 错误: {e}")
            # 可加入待清理队列，定时任务重试
    else:
        # ====== 软删除（推荐）======
        from django.utils import timezone
        
        file_record.is_deleted = True
        file_record.deleted_at = timezone.now()
        file_record.deleted_by = user
        file_record.save(update_fields=['is_deleted', 'deleted_at', 'deleted_by'])
        
        logger.info(f"文件已软删除: {file_record.physical_name}")
    
    # 记录审计日志
    log_file_operation('DELETE', file_record, user, hard=hard)
```

## 五、一致性校验机制

### 5.1 一键校验脚本

```python
#!/usr/bin/env python3
"""
check_file_mapping.py - 数据库 ↔ 物理文件映射校验脚本

运行方式：
    python manage.py shell < check_file_mapping.py

返回码：
    0 - 校验通过
    1 - 发现错误
"""

import os
import sys
from collections import defaultdict

# Django 初始化
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'spug.settings')
import django
django.setup()

from apps.document.models import DocumentFilePrivate, DocumentFilePublic


class MappingChecker:
    """映射关系校验器"""
    
    def __init__(self):
        self.errors = []
        self.warnings = []
        self.stats = defaultdict(int)
    
    def check_model(self, model_class, model_name):
        """校验指定模型"""
        print(f"\n{'='*60}")
        print(f"校验模型: {model_name}")
        print(f"{'='*60}")
        
        # 获取所有未软删除的记录
        files = model_class.objects.filter(is_deleted=False)
        total = files.count()
        print(f"总记录数: {total}")
        
        checked = 0
        for file_record in files.iterator(chunk_size=1000):
            checked += 1
            if checked % 1000 == 0:
                print(f"  已检查: {checked}/{total}")
            
            self._check_single_record(file_record, model_name)
        
        print(f"  完成检查: {checked} 条记录")
    
    def _check_single_record(self, file_record, model_name):
        """校验单条记录"""
        physical_name = file_record.physical_name
        file_path = file_record.file_path
        record_id = file_record.id
        
        # 1. 检查字段非空
        if not physical_name:
            self.errors.append({
                'type': '空物理名',
                'model': model_name,
                'id': record_id,
                'detail': 'physical_name 为空'
            })
            self.stats['empty_physical_name'] += 1
            return
        
        if not file_path:
            self.errors.append({
                'type': '空路径',
                'model': model_name,
                'id': record_id,
                'detail': 'file_path 为空'
            })
            self.stats['empty_file_path'] += 1
            return
        
        # 2. 检查路径末尾文件名是否等于 physical_name
        basename = os.path.basename(file_path)
        if basename != physical_name:
            self.errors.append({
                'type': '路径不匹配',
                'model': model_name,
                'id': record_id,
                'detail': f'physical_name={physical_name}, path_basename={basename}'
            })
            self.stats['path_mismatch'] += 1
        
        # 3. 检查物理文件是否存在
        if not os.path.exists(file_path):
            self.errors.append({
                'type': '文件缺失',
                'model': model_name,
                'id': record_id,
                'detail': f'physical_name={physical_name}, path={file_path}'
            })
            self.stats['file_missing'] += 1
        else:
            self.stats['file_exists'] += 1
            
            # 4. 检查文件大小是否匹配（可选）
            actual_size = os.path.getsize(file_path)
            if actual_size != file_record.file_size:
                self.warnings.append({
                    'type': '大小不匹配',
                    'model': model_name,
                    'id': record_id,
                    'detail': f'数据库={file_record.file_size}, 实际={actual_size}'
                })
                self.stats['size_mismatch'] += 1
    
    def check_duplicates(self):
        """检查重复物理名"""
        print(f"\n{'='*60}")
        print("检查重复物理名...")
        print(f"{'='*60}")
        
        for model_class, model_name in [
            (DocumentFilePrivate, 'DocumentFilePrivate'),
            (DocumentFilePublic, 'DocumentFilePublic')
        ]:
            from django.db.models import Count
            
            duplicates = model_class.objects.values('physical_name', 'tenant_id') \
                .annotate(count=Count('id')) \
                .filter(count__gt=1)
            
            for dup in duplicates:
                self.errors.append({
                    'type': '重复物理名',
                    'model': model_name,
                    'detail': f"physical_name={dup['physical_name']}, tenant={dup['tenant_id']}, count={dup['count']}"
                })
                self.stats['duplicate_physical_name'] += 1
            
            print(f"  {model_name}: 发现 {len(duplicates)} 个重复")
    
    def generate_report(self):
        """生成校验报告"""
        print(f"\n{'='*60}")
        print("校验报告")
        print(f"{'='*60}")
        
        print("\n📊 统计信息:")
        for key, value in sorted(self.stats.items()):
            print(f"  {key}: {value}")
        
        print(f"\n⚠️  警告数: {len(self.warnings)}")
        for warning in self.warnings[:10]:  # 只显示前10条
            print(f"  [{warning['type']}] {warning['model']} ID={warning['id']}")
            print(f"    -> {warning['detail']}")
        
        if len(self.warnings) > 10:
            print(f"  ... 还有 {len(self.warnings) - 10} 条警告")
        
        print(f"\n❌ 错误数: {len(self.errors)}")
        for error in self.errors[:20]:  # 只显示前20条
            print(f"  [{error['type']}] {error['model']} ID={error.get('id', 'N/A')}")
            print(f"    -> {error['detail']}")
        
        if len(self.errors) > 20:
            print(f"  ... 还有 {len(self.errors) - 20} 条错误")
        
        # 返回状态
        if self.errors:
            print(f"\n❌ 校验失败，发现 {len(self.errors)} 个错误")
            return 1
        elif self.warnings:
            print(f"\n⚠️  校验通过，但有 {len(self.warnings)} 个警告")
            return 0
        else:
            print(f"\n✅ 校验通过，映射关系完全一致")
            return 0


def main():
    """主函数"""
    print("🔍 数据库 ↔ 物理文件映射关系校验")
    print(f"运行时间: {__import__('datetime').datetime.now()}")
    
    checker = MappingChecker()
    
    # 校验私有文件
    checker.check_model(DocumentFilePrivate, 'DocumentFilePrivate')
    
    # 校验公共文件
    checker.check_model(DocumentFilePublic, 'DocumentFilePublic')
    
    # 检查重复
    checker.check_duplicates()
    
    # 生成报告
    exit_code = checker.generate_report()
    
    sys.exit(exit_code)


if __name__ == '__main__':
    main()
```

```python
# apps/document/tasks/cleanup.py

from celery import shared_task
from django.utils import timezone
from datetime import timedelta

@shared_task
def cleanup_soft_deleted_files():
    """
    清理软删除文件（物理删除）
    
    策略：
    - 软删除后保留 30 天
    - 30 天后自动物理删除
    - 可通过配置调整保留时间
    """
    RETENTION_DAYS = 30
    
    cutoff_date = timezone.now() - timedelta(days=RETENTION_DAYS)
    
    # 查找需要清理的文件
    expired_files = DocumentFilePrivate.all_objects.filter(
        is_deleted=True,
        deleted_at__lt=cutoff_date
    )
    
    deleted_count = 0
    error_count = 0
    
    for file_record in expired_files:
        try:
            # 执行硬删除
            file_record.delete(hard=True)
            deleted_count += 1
            print(f"已清理: {file_record.physical_name}")
        except Exception as e:
            error_count += 1
            print(f"清理失败 {file_record.physical_name}: {e}")
    
    return {
        'deleted': deleted_count,
        'errors': error_count,
        'retention_days': RETENTION_DAYS
    }
```
