import os
import posixpath
from django.conf import settings

# 完全复用你现有的存储根目录
DOC_STORAGE_ROOT = getattr(settings, "DOC_STORAGE_ROOT", os.path.join(settings.BASE_DIR, "storage", "documents"))

def get_document_relative_path(is_public: bool, user_id: int, folder_id: int | None, unique_name: str):
    """
    统一生成文件相对路径，全平台分隔符统一为/
    完全兼容你现有的is_safe_path()安全机制
    """
    if is_public:
        base_dir = "public"
    else:
        base_dir = f"private/user-{user_id}"
    
    # 兼容你现有的文件夹路径规则
    if folder_id:
        folder_dir = f"folder_{folder_id}"
        full_relative = posixpath.join(base_dir, folder_dir, unique_name)
    else:
        full_relative = posixpath.join(base_dir, unique_name)
    
    # 路径安全兜底
    full_relative = posixpath.normpath(full_relative)
    if not full_relative.startswith(("public/", "private/")):
        raise ValueError("非法文件路径，超出存储范围")
    return full_relative

def get_document_absolute_path(relative_path: str):
    """获取绝对路径，完全复用你现有的is_safe_path()"""
    abs_path = os.path.join(DOC_STORAGE_ROOT, relative_path)
    abs_path = os.path.normpath(abs_path)
    # 直接复用你现有的is_safe_path()做安全兜底
    from apps.document.utils import is_safe_path
    if not is_safe_path(DOC_STORAGE_ROOT, abs_path):
        raise ValueError("非法文件路径，超出存储根目录")
    return abs_path

## 四、权限体系分表适配（完全兼容现有@auth装饰器）
```python
# apps/libs/document_decorators.py（基于同表方案改造）
from functools import wraps
from django.http import JsonResponse
from apps.libs.document_utils import get_folder_model, get_file_model

# ==================== 工具函数：判断用户是否为全局管理员（按你的项目实际调整）====================
def is_global_admin(user):
    if user.is_superuser:
        return True
    # 【按你的项目实际调整】：例如检查用户是否有document.document.admin权限
    return user.has_perm('document.document.admin')

# ==================== 统一权限校验装饰器（分表版）====================
def document_permission_check(permission: str):
    """
    权限类型：view/download/upload/edit/delete/move/rename/folder_create
    使用方式：在现有@auth装饰器之后叠加使用
    """
    def decorator(view_func):
        @wraps(view_func)
        def wrapper(request, *args, **kwargs):
            # 前置：完全复用你现有的@auth装饰器，这里仅做补充校验
            user = request.user
            file_id = kwargs.get("id") or kwargs.get("file_id")
            folder_id = kwargs.get("id") or kwargs.get("folder_id")
            admin_flag = is_global_admin(user)
            is_public = request.data.get("is_public", False) if request.method == "POST" else request.GET.get("is_public", "false").lower() == "true"

            # ==================== 1. 具体文件/文件夹操作校验 ====================
            target_obj = None
            if file_id:
                FileModel = get_file_model(is_public)
                target_obj = FileModel.objects.filter(id=file_id).first()
            elif folder_id:
                FolderModel = get_folder_model(is_public)
                target_obj = FolderModel.objects.filter(id=folder_id).first()

            if target_obj:
                # 资源不存在拦截
                if not target_obj:
                    return JsonResponse({"code": 404, "msg": "资源不存在"}, status=404)

                # 【公共资源权限逻辑】
                if is_public:
                    # 查看/下载：所有用户直接放行
                    if permission in ["view", "download"]:
                        return view_func(request, *args, **kwargs)
                    # 编辑/删除/移动：仅本人/全局管理员放行
                    if (target_obj.created_by_id == user.id) or admin_flag:
                        return view_func(request, *args, **kwargs)
                    else:
                        return JsonResponse({"code": 403, "msg": "仅可操作自己上传的公共文件"}, status=403)

                # 【私有资源权限逻辑】
                else:
                    # 仅本人/全局管理员可操作
                    if (target_obj.created_by_id == user.id) or admin_flag:
                        return view_func(request, *args, **kwargs)
                    else:
                        return JsonResponse({"code": 403, "msg": "无权限操作他人私有文件"}, status=403)

            # ==================== 2. 上传/文件夹创建权限校验 ====================
            if permission in ["upload", "folder_create"]:
                target_folder_id = request.data.get("folder_id")

                # 公共空间上传/创建文件夹：所有用户直接放行
                if is_public:
                    # 校验目标文件夹是公共文件夹
                    if target_folder_id:
                        FolderModel = get_folder_model(True)
                        target_folder = FolderModel.objects.filter(id=target_folder_id).first()
                        if not target_folder:
                            return JsonResponse({"code": 400, "msg": "仅可上传到公共文件夹"}, status=400)
                    return view_func(request, *args, **kwargs)

                # 私有空间上传/创建文件夹：校验目标文件夹是当前用户的私有文件夹
                if target_folder_id:
                    FolderModel = get_folder_model(False)
                    target_folder = FolderModel.objects.filter(id=target_folder_id, created_by=user).first()
                    if not target_folder:
                        return JsonResponse({"code": 400, "msg": "仅可上传到自己的私有文件夹"}, status=400)
                return view_func(request, *args, **kwargs)

            # 列表查询默认放行
            return view_func(request, *args, **kwargs)
        return wrapper
    return decorator
```
