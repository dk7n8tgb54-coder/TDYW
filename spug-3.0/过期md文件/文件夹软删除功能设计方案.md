# 资料库文件夹软删除功能设计方案

## 1. 需求背景

### 当前问题
- 删除文件夹时，文件夹本身被永久删除，仅文件夹内文件进入回收站
- 用户无法恢复完整的文件夹结构
- 文件恢复后散落在根目录，需要手动整理

### 2024-03-16 问题排查记录
**问题现象**: 删除文件夹后，文件夹内的文件未在回收站显示

**根因分析**:
1. **外键级联删除问题**: 文件模型的 `folder` 外键使用 `on_delete=models.CASCADE`
2. **删除顺序问题**: 当前删除逻辑先软删除文件，再硬删除文件夹
3. **实际执行流程**:
   ```
   1. file.delete(hard=False)  # 文件标记 is_deleted=True
   2. folder.delete()           # 由于外键是 CASCADE，文件记录被物理删除！
   ```
4. **结果**: 虽然文件被软删除，但文件夹硬删除时触发了级联删除，导致文件记录从数据库中消失

**验证方法**:
```sql
-- 删除文件夹前检查
SELECT id, name, is_deleted, folder_id FROM document_file_private WHERE folder_id = <folder_id>;

-- 删除文件夹后检查（文件记录已不存在）
SELECT id, name FROM document_file_private WHERE id IN (<file_ids>);
```

**临时解决方案**:
- 修改外键为 `on_delete=models.SET_NULL`，让文件在文件夹删除后保留外键关系
- 或者先软删除文件夹，保留文件的外键引用

**注意**: 缓存清除问题（`invalidate_cache`）是次要问题，不是根本原因

### 目标
- 实现文件夹软删除，保留完整的文件夹层级结构
- 支持一键恢复文件夹及其内容到原位置
- 保持与现有回收站功能的兼容性

        default=False,
        help_text='是否已删除（软删除）',
        verbose_name='删除标记'
    )
    deleted_at = models.DateTimeField(
        null=True, blank=True,
        help_text='删除时间',
        verbose_name='删除时间'
    )
    deleted_by = models.ForeignKey(
        User, on_delete=models.SET_NULL, null=True, blank=True,
        related_name='deleted_folders_private',
        help_text='删除人',
        verbose_name='删除人'
    )
    
    # ========== 【新增】管理器 ==========
    objects = SoftDeletedManager()  # 默认管理器，过滤已删除
    all_objects = AllObjectsManager()  # 全量管理器，包含已删除
    
    class Meta:
        db_table = 'spug_document_folder_private'
        verbose_name = '文档文件夹(私有)'
        verbose_name_plural = '文档文件夹(私有)'
        ordering = ['-created_at']
        constraints = [
            models.UniqueConstraint(
                fields=['name', 'parent'],
                name='unique_folder_name_parent_private',
                condition=models.Q(is_deleted=False)  # 只限制未删除的
            )
        ]
    
    def delete(self, hard=False, *args, **kwargs):
        """
        删除文件夹
        hard=True: 硬删除（彻底删除）
        hard=False: 软删除（移入回收站）
        """
        if hard:
            # 硬删除：级联删除子文件夹和文件
            super().delete(*args, **kwargs)
        else:
            # 软删除：标记状态
            self.is_deleted = True
            self.deleted_at = timezone.now()
            self.save(update_fields=['is_deleted', 'deleted_at'])
    
    def restore(self):
        """恢复软删除的文件夹"""
        self.is_deleted = False
        self.deleted_at = None
        self.deleted_by = None
        self.save(update_fields=['is_deleted', 'deleted_at', 'deleted_by'])
    
    def get_full_path(self):
        """获取文件夹完整路径（用于显示）"""
        if self.parent and not self.parent.is_deleted:
            return f"{self.parent.get_full_path()}/{self.name}"
        return self.name
    
    def get_descendants(self, include_self=False):
        """获取所有子孙文件夹"""
        descendants = []
        if include_self:
            descendants.append(self)
        
        children = DocumentFolderPrivate.all_objects.filter(
            parent=self, is_deleted=True
        )
        for child in children:
            descendants.extend(child.get_descendants(include_self=True))
        
        return descendants

class DocumentFilePrivate(models.Model):

    # 【修改】外键从 CASCADE 改为 SET_NULL

    folder = models.ForeignKey(
        DocumentFolderPrivate, 
        on_delete=models.SET_NULL,  # 【修改】原为 CASCADE
        null=True, blank=True,
        related_name='files',
        verbose_name='所属文件夹'
    )
    
    # 软删除字段已存在...
    is_deleted = models.BooleanField(default=False)
    deleted_at = models.DateTimeField(null=True, blank=True)
```

**重要说明**:
- 原设计使用 `CASCADE` 会导致：文件夹删除时，即使文件已软删除，也会被物理删除
- 修改为 `SET_NULL` 后：文件夹删除时，文件的 `folder_id` 设为 NULL，但记录保留
- 这样文件可以在回收站中正常显示，且保留文件元数据

# POST /api/document/folder/
# 方法: delete

def delete(self, request):
    """
    删除文件夹（软删除）
    """
    form, error = JsonParser(
        Argument('id', type=int, help='参数错误'),
        Argument('is_public', type=bool, required=False, default=False)
    ).parse(request.GET)
    
    # ... 权限校验 ...
    
    try:
        # 【修改】软删除文件夹及其内容
        self._soft_delete_folder(folder, FolderModel, FileModel, request.user)
        
        log_operation(
            action="FOLDER_SOFT_DELETE",  # 修改操作类型
            user=request.user,
            resource_type="FOLDER",
            resource_id=folder.id,
            is_public=form.is_public,
            folder_name=folder.name
        )
        return json_response()
        
    except Exception as e:
        logger.error(f'[Document] Error soft deleting folder: {e}')
        return json_response(error=f'文件夹删除失败: {str(e)}')


def _soft_delete_folder(self, folder, FolderModel, FileModel, user):
    """
    【新增】递归软删除文件夹及其内容
    """
    # 1. 软删除所有子文件夹
    sub_folders = FolderModel.objects.filter(parent=folder, is_deleted=False)
    for sub_folder in sub_folders:
        self._soft_delete_folder(sub_folder, FolderModel, FileModel, user)
    
    # 2. 软删除当前文件夹下的所有文件
    files = folder.files.filter(is_deleted=False)
    for file in files:
        file.delete(hard=False)  # 软删除文件
    
    # 3. 软删除文件夹本身
    folder.is_deleted = True
    folder.deleted_at = timezone.now()
    folder.deleted_by = user
    folder.save(update_fields=['is_deleted', 'deleted_at', 'deleted_by'])
    
    logger.info(f'[Document] Folder soft deleted: {folder.name} (id={folder.id})')
```

# GET /api/document/recycle-bin/
# 新增参数: type=folder|file|all

def get(self, request):
    """
    获取回收站列表（支持文件夹和文件）
    """
    form, error = JsonParser(
        Argument('page', type=int, required=False, default=1),
        Argument('page_size', type=int, required=False, default=20),
        Argument('keyword', required=False),
        Argument('space', required=False, default='all'),
        Argument('type', required=False, default='all',  # 【新增】
                 filter=lambda x: x in ['folder', 'file', 'all'])
    ).parse(request.GET)
    
    if form.type == 'folder':
        # 只查询文件夹
        result = self._get_folder_results(...)
    elif form.type == 'file':
        # 只查询文件
        result = self._get_file_results(...)
    else:
        # 查询文件夹和文件
        folder_results = self._get_folder_results(...)
        file_results = self._get_file_results(...)
        # 合并并排序
        result = self._merge_results(folder_results, file_results)
    
    return json_response(data=result)


def _get_folder_results(self, user, space, keyword, page, page_size):
    """【新增】查询回收站中的文件夹"""
    results = []
    total_count = 0
    
    user_tenant_id = getattr(user, 'tenant_id', None)
    
    if space in ['private', 'public']:
        FolderModel = DocumentFolderPrivate if space == 'private' else DocumentFolderPublic
        qs = FolderModel.all_objects.filter(
            is_deleted=True
        ).select_related('parent', 'created_by', 'deleted_by')
        
        if not user.is_supper:
            if space == 'private':
                qs = qs.filter(tenant_id=user_tenant_id)
            else:
                qs = qs.filter(created_by=user)
        
        if keyword:
            qs = qs.filter(name__icontains=keyword)
        
        total_count = qs.count()
        offset = (page - 1) * page_size
        folders = qs.order_by('-deleted_at')[offset:offset + page_size]
        
        results = [self._format_folder(f, space) for f in folders]
    
    return {
        'items': results,
        'total': total_count,
        'page': page,
        'page_size': page_size
    }


def _format_folder(self, folder_obj, space):
    """【新增】格式化文件夹信息"""
    retention_days = getattr(settings, 'RECYCLE_BIN_RETENTION_DAYS', 30)
    days_left = retention_days - (timezone.now() - folder_obj.deleted_at).days
    
    # 统计文件夹内文件数量和大小
    file_count = self._get_folder_file_count(folder_obj, space)
    total_size = self._get_folder_total_size(folder_obj, space)
    
    return {
        'id': folder_obj.id,
        'type': 'folder',  # 标识类型
        'name': folder_obj.name,
        'full_path': folder_obj.get_full_path(),  # 完整路径
        'space': space,
        'deleted_at': folder_obj.deleted_at.isoformat(),
        'deleted_by': {
            'id': folder_obj.deleted_by.id,
            'nickname': folder_obj.deleted_by.nickname
        } if folder_obj.deleted_by else None,
        'created_by': {
            'id': folder_obj.created_by.id,
            'nickname': folder_obj.created_by.nickname
        } if folder_obj.created_by else None,
        'retention_days_left': max(0, days_left),
        'file_count': file_count,  # 包含的文件数量
        'total_size': total_size,  # 总大小
        'can_restore': self._can_restore_folder(folder_obj, space)  # 是否可恢复
    }

### 4.3 恢复文件夹

```python
# POST /api/document/recycle-bin/folder-restore/
# 【新增】恢复文件夹及其内容

class RecycleBinFolderRestoreView(View):
    """【新增】恢复文件夹视图"""
    
    @auth('document.recycle-bin.restore')
    def post(self, request):
        """恢复软删除的文件夹"""
        form, error = JsonParser(
            Argument('folder_ids', type=list, required=True),
            Argument('restore_mode', required=False, default='original',
                     filter=lambda x: x in ['original', 'root']),
            Argument('target_parent_id', type=int, required=False)  # 自定义恢复位置
        ).parse(request.body)
        
        if error:
            return json_response(error=error)
        
        results = []
        success_count = 0
        
        for folder_id in form.folder_ids:
            result = self._restore_folder(
                folder_id, 
                request.user,
                form.restore_mode,
                form.target_parent_id
            )
            results.append(result)
            if result['status'] == 'success':
                success_count += 1
        
        # 清除缓存
        invalidate_cache(request.user.id)
        
        return json_response(data={
            'success_count': success_count,
            'failed_count': len(form.folder_ids) - success_count,
            'details': results
        })
    
    def _restore_folder(self, folder_id, user, mode, target_parent_id):
        """恢复单个文件夹及其内容"""
        try:
            # 1. 查找文件夹
            folder = self._find_deleted_folder(folder_id, user)
            if not folder:
                return {'id': folder_id, 'status': 'failed', 'error': '文件夹不存在'}
            
            # 2. 检查权限
            if not self._check_folder_permission(folder, user):
                return {'id': folder_id, 'status': 'failed', 'error': '无操作权限'}
            
            # 3. 确定恢复位置
            if mode == 'original':
                # 恢复到原位置
                if folder.parent and folder.parent.is_deleted:
                    # 父文件夹也被删除了，恢复到根目录
                    target_parent = None
                else:
                    target_parent = folder.parent
            elif mode == 'root':
                # 恢复到根目录
                target_parent = None
            elif target_parent_id:
                # 恢复到指定位置
                target_parent = self._get_target_parent(target_parent_id, user)
                if not target_parent:
                    return {'id': folder_id, 'status': 'failed', 'error': '目标位置不存在'}
            
            # 4. 检查目标位置是否有同名文件夹
            if self._has_duplicate_name(folder.name, target_parent, user):
                # 自动重命名
                folder.name = self._generate_unique_name(folder.name, target_parent, user)
            
            # 5. 恢复文件夹
            with transaction.atomic():
                self._restore_folder_recursive(folder, target_parent, user)
            
            return {
                'id': folder_id, 
                'status': 'success',
                'restored_path': folder.get_full_path(),
                'restored_file_count': self._get_restored_file_count(folder)
            }
            
        except Exception as e:
            logger.error(f'[RecycleBin] 恢复文件夹失败: folder_id={folder_id}, error={e}')
            return {'id': folder_id, 'status': 'failed', 'error': str(e)}
    
    def _restore_folder_recursive(self, folder, new_parent, user):
        """递归恢复文件夹及其内容"""
        # 1. 恢复文件夹本身
        folder.parent = new_parent
        folder.restore()  # 清除软删除标记
        
        # 2. 恢复子文件夹
        sub_folders = folder.__class__.all_objects.filter(
            parent=folder, is_deleted=True
        )
        for sub_folder in sub_folders:
            self._restore_folder_recursive(sub_folder, folder, user)
        
        # 3. 恢复文件夹内的文件
        files = folder.files.filter(is_deleted=True)
        for file in files:
            file.restore()
        
        logger.info(f'[RecycleBin] 文件夹恢复完成: {folder.name} (id={folder.id})')
```

            'failed_count': sum(1 for r in results if r['status'] == 'failed'),
            'details': results
        })
    
    def _permanent_delete_folder(self, folder_id, user):
        """递归彻底删除文件夹"""
        try:
            folder = self._find_deleted_folder(folder_id, user)
            if not folder:
                return {'id': folder_id, 'status': 'failed', 'error': '文件夹不存在'}
            
            # 递归删除子文件夹
            sub_folders = folder.__class__.all_objects.filter(
                parent=folder, is_deleted=True
            )
            for sub_folder in sub_folders:
                self._permanent_delete_folder(sub_folder.id, user)
            
            # 删除文件夹内的所有文件
            files = folder.files.filter(is_deleted=True)
            for file in files:
                file.delete(hard=True)  # 硬删除
            
            # 删除物理目录
            self._delete_physical_folder(folder)
            
            # 删除文件夹记录
            folder.delete(hard=True)
            
            return {'id': folder_id, 'status': 'success'}
            
        except Exception as e:
            logger.error(f'[RecycleBin] 彻底删除文件夹失败: {e}')
            return {'id': folder_id, 'status': 'failed', 'error': str(e)}
