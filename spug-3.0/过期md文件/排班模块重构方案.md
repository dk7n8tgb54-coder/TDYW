5. **测试友好**: 便于单元测试

```python
"""排班模块业务异常定义"""


class ScheduleException(Exception):
    """排班业务异常基类"""
    def __init__(self, message, code=None):
        super().__init__(message)
        self.message = message
        self.code = code or 'SCHEDULE_ERROR'

    def to_dict(self):
        return {
            'error': self.message,
            'code': self.code
        }


class ScheduleConflictError(ScheduleException):
    """排班冲突异常"""
    def __init__(self, message='该人员在此日期已有排班'):
        super().__init__(message, 'SCHEDULE_CONFLICT')


class SwapSelfError(ScheduleException):
    """不能与自己换班异常"""
    def __init__(self, message='不能与自己换班'):
        super().__init__(message, 'SWAP_SELF')


class AlreadyApprovedError(ScheduleException):
    """已审批记录不能修改异常"""
    def __init__(self, message='已审批的记录不能修改'):
        super().__init__(message, 'ALREADY_APPROVED')


class RecordNotFoundError(ScheduleException):
    """记录不存在异常"""
    def __init__(self, message='记录不存在或无权操作'):
        super().__init__(message, 'RECORD_NOT_FOUND')
```

```python
# services/schedule_service.py
"""排班业务逻辑服务"""

from abc import ABC, abstractmethod
from datetime import datetime, timedelta
from django.db import transaction
from ..models import Schedule, ScheduleStaff, ScheduleShift
from ..exceptions import (
    ScheduleConflictError, AlreadyApprovedError,
    RecordNotFoundError, InvalidStatusTransitionError
)
from ..constants import SwapStatus


class ScheduleServiceInterface(ABC):
    """排班服务接口抽象基类"""
    
    @abstractmethod
    def generate_auto_schedule(self, staff, shift, start_date, end_date):
        """自动生成排班"""
        pass
    
    @abstractmethod
    def swap_schedules(self, swap_record, user):
        """执行排班交换"""
        pass
    
    @abstractmethod
    def restore_swap_schedules(self, swap_record, user):
        """恢复换班前的排班状态"""
        pass


class ScheduleService(ScheduleServiceInterface):
    """排班服务实现类"""
    
    @staticmethod
    def generate_auto_schedule(staff, shift, start_date, end_date):
        """
        自动生成排班
        
        Args:
            staff: 排班人员
            shift: 班次规则
            start_date: 开始日期
            end_date: 结束日期
            
        Returns:
            list: 排班记录列表
        """
        schedules = []
        current_date = start_date
        
        if shift.shift_type == 'work_rest' and shift.work_days and shift.rest_days:
            # 上X休Y模式
            work_count = 0
            rest_count = 0
            is_work_cycle = True
            
            while current_date <= end_date:
                if is_work_cycle:
                    schedules.append({
                        'staff_id': staff.id,
                        'staff_name': staff.user_name,
                        'schedule_date': current_date.strftime('%Y-%m-%d'),
                        'shift_id': shift.id,
                        'shift_name': shift.name,
                    })
                    work_count += 1
                    if work_count >= shift.work_days:
                        work_count = 0
                        is_work_cycle = False
                else:
                    rest_count += 1
                    if rest_count >= shift.rest_days:
                        rest_count = 0
                        is_work_cycle = True
                current_date += timedelta(days=1)
        else:
            # 自定义模式 - 每天排班
            while current_date <= end_date:
                schedules.append({
                    'staff_id': staff.id,
                    'staff_name': staff.user_name,
                    'schedule_date': current_date.strftime('%Y-%m-%d'),
                    'shift_id': shift.id,
                    'shift_name': shift.name,
                })
                current_date += timedelta(days=1)
        
        return schedules
    
    @staticmethod
    @transaction.atomic
    def swap_schedules(swap_record, user):
        """
        执行排班交换
        
        Args:
            swap_record: 换班记录
            user: 操作用户
        """
        from_date = swap_record.from_date or swap_record.schedule_date
        to_date = swap_record.to_date or swap_record.schedule_date
        
        # 获取双方排班
        from_schedule = Schedule.objects.filter(
            staff_id=swap_record.from_staff_id,
            schedule_date=from_date
        ).first()
        
        to_schedule = Schedule.objects.filter(
            staff_id=swap_record.to_staff_id,
            schedule_date=to_date
        ).first()
        
        if from_schedule and to_schedule:
            # 交换排班信息
            Schedule.objects.filter(pk=from_schedule.pk).update(
                staff_id=swap_record.to_staff_id,
                staff_name=swap_record.to_staff_name,
                shift_id=swap_record.to_shift_id,
                shift_name=swap_record.to_shift_name,
            )
            Schedule.objects.filter(pk=to_schedule.pk).update(
                staff_id=swap_record.from_staff_id,
                staff_name=swap_record.from_staff_name,
                shift_id=swap_record.from_shift_id,
                shift_name=swap_record.from_shift_name,
            )
        elif from_schedule:
            from_schedule.delete()
        elif to_schedule:
            to_schedule.delete()
    
    @staticmethod
    @transaction.atomic  
    def restore_swap_schedules(swap_record, user):
        """恢复换班前的排班状态"""
        from_date = swap_record.from_date or swap_record.schedule_date
        to_date = swap_record.to_date or swap_record.schedule_date
        
        # 查找当前排班位置（已交换后）
        from_schedule = Schedule.objects.filter(
            staff_id=swap_record.from_staff_id,
            schedule_date=to_date
        ).first()
        
        to_schedule = Schedule.objects.filter(
            staff_id=swap_record.to_staff_id,
            schedule_date=from_date
        ).first()
        
        if from_schedule and to_schedule:
            # 恢复原排班
            from_schedule.schedule_date = from_date
            from_schedule.staff_name = swap_record.from_staff_name
            from_schedule.shift_id = swap_record.from_shift_id
            from_schedule.shift_name = swap_record.from_shift_name
            from_schedule.save()
            
            to_schedule.schedule_date = to_date
            to_schedule.staff_name = swap_record.to_staff_name
            to_schedule.shift_id = swap_record.to_shift_id
            to_schedule.shift_name = swap_record.to_shift_name
            to_schedule.save()
```

```python
# utils/validators.py
"""数据验证工具"""

import re
from datetime import datetime


class ScheduleValidator:
    """排班数据验证器"""
    
    DATETIME_PATTERN = re.compile(r'^\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}$')
    DATE_PATTERN = re.compile(r'^\d{4}-\d{2}-\d{2}$')
    
    @classmethod
    def validate_datetime(cls, value, field_name='时间'):
        """验证日期时间格式"""
        if not value:
            return None
        if not cls.DATETIME_PATTERN.match(value):
            return f'{field_name}格式错误，请使用：YYYY-MM-DD HH:MM:SS'
        return None
    
    @classmethod
    def validate_date(cls, value, field_name='日期'):
        """验证日期格式"""
        if not value:
            return None
        if not cls.DATE_PATTERN.match(value):
            return f'{field_name}格式错误，请使用：YYYY-MM-DD'
        return None
    
    @classmethod
    def validate_swap_status_transition(cls, current_status, new_status):
        """验证换班状态流转是否合法"""
        valid_transitions = {
            'pending': ['approved', 'rejected'],
            'approved': ['cancelled'],
            'rejected': [],
            'cancelled': []
        }
        
        if new_status not in valid_transitions.get(current_status, []):
            return f'不允许的状态流转：{current_status} -> {new_status}'
        return None
```

```python
        return json_response()
```

```python
            'updated_at': human_datetime(),
            'updated_by': user,
            'approved_by': user,
            'approved_by_name': user.username if user else '',
            'approved_at': human_datetime()
        }
```

```javascript
/**
```

```javascript
import moment from 'moment';

/**
 * 生成自动排班
 * @param {Object} staff - 人员信息
 * @param {Object} shift - 班次规则
 * @param {moment} startDate - 开始日期
 * @param {moment} endDate - 结束日期
 * @param {Array} existingSchedules - 现有排班（用于冲突检查）
 * @returns {{schedules: Array, conflicts: Array}}
 */
export const generateAutoSchedule = (staff, shift, startDate, endDate, existingSchedules = []) => {
  const schedules = [];
  const conflicts = [];
  let currentDate = startDate.clone();
  
  if (shift.shift_type === 'work_rest' && shift.work_days && shift.rest_days) {
    // 上X休Y模式
    let workCount = 0;
    let restCount = 0;
    let isWorkCycle = true;
    
    while (currentDate.isSameOrBefore(endDate)) {
      const dateStr = currentDate.format('YYYY-MM-DD');
      
      if (isWorkCycle) {
        // 检查冲突
        const conflict = checkScheduleConflict(staff.id, dateStr, existingSchedules);
        if (conflict) {
          conflicts.push(dateStr);
        } else {
          schedules.push(createScheduleItem(staff, shift, dateStr));
        }
        
        workCount++;
        if (workCount >= shift.work_days) {
          workCount = 0;
          isWorkCycle = false;
        }
      } else {
        restCount++;
        if (restCount >= shift.rest_days) {
          restCount = 0;
          isWorkCycle = true;
        }
      }
      currentDate.add(1, 'day');
    }
  } else {
    // 自定义模式 - 每天排班
    while (currentDate.isSameOrBefore(endDate)) {
      const dateStr = currentDate.format('YYYY-MM-DD');
      const conflict = checkScheduleConflict(staff.id, dateStr, existingSchedules);
      
      if (conflict) {
        conflicts.push(dateStr);
      } else {
        schedules.push(createScheduleItem(staff, shift, dateStr));
      }
      currentDate.add(1, 'day');
    }
  }
  
  return { schedules, conflicts };
};

/**
 * 检查排班冲突
 */
const checkScheduleConflict = (staffId, dateStr, existingSchedules) => {
  return existingSchedules.find(s => 
    s.schedule_date === dateStr && s.staff_id === staffId
  );
};

/**
 * 创建排班项
 */
const createScheduleItem = (staff, shift, dateStr) => ({
  staff_id: staff.id,
  staff_name: staff.user_name,
  schedule_date: dateStr,
  shift_id: shift.id,
  shift_name: shift.name,
});
```

```javascript
/**
 * 排班状态管理

import { observable, action, makeObservable } from 'mobx';
import { http } from 'libs';
import moment from 'moment';

class ScheduleStore {
  @observable scheduleList = [];
  @observable staffList = [];
  @observable shiftList = [];
  @observable isFetching = false;
  @observable currentDate = moment();

  constructor() {
    makeObservable(this);
  }

  @action
  setCurrentDate(date) {
    this.currentDate = moment(date);
  }

  @action
  async fetchSchedules(year, month, append = true) {
    this.isFetching = true;
    try {
      const res = await http.get('/api/schedule/', { params: { year, month } });
      if (append) {
        const existingIds = new Set(this.scheduleList.map(s => s.id));
        const newSchedules = res.filter(s => !existingIds.has(s.id));
        this.scheduleList = [...this.scheduleList, ...newSchedules];
      } else {
        this.scheduleList = res;
      }
    } finally {
      this.isFetching = false;
    }
  }

  @action
  async fetchStaffList() {
    this.isFetching = true;
    try {
      const res = await http.get('/api/schedule/staff/');
      this.staffList = res;
    } finally {
      this.isFetching = false;
    }
  }

  @action
  async fetchShiftList() {
    const res = await http.get('/api/schedule/shift/');
    this.shiftList = res;
  }

  @action
  async addSchedule(data) {
    await http.post('/api/schedule/', data);
    await this.fetchSchedules(this.currentDate.year(), this.currentDate.month() + 1, false);
  }

  @action
  async deleteSchedule(id) {
    await http.delete('/api/schedule/', { params: { id } });
    await this.fetchSchedules(this.currentDate.year(), this.currentDate.month() + 1, false);
  }

  @action
  async batchQuerySchedules(staffIds, startDate, endDate) {
    return http.post('/api/schedule/batch_query/', {
      staff_ids: staffIds,
      start_date: startDate,
      end_date: endDate,
    });
  }

  /**
   * 获取某日期的排班
   */
  getSchedulesByDate(dateStr) {
    return this.scheduleList.filter(s => s.schedule_date === dateStr);
  }

  /**
   * 检查排班冲突
   */
  hasConflict(staffId, dateStr) {
    return this.scheduleList.some(
      s => s.staff_id === staffId && s.schedule_date === dateStr
    );
  }
}

export default new ScheduleStore();
```

```javascript
/**
 * 排班数据管理Hook
 */
import { useCallback, useMemo } from 'react';
import { useLocalStore } from 'mobx-react';
import scheduleStore from '../store/scheduleStore';

export const useSchedule = () => {
  const store = useLocalStore(() => scheduleStore);

  // 获取某日期的排班并按班次分组
  const getShiftGroups = useCallback((schedules, dateStr, swapList, substituteList) => {
    const dateSchedules = schedules.filter(s => s.schedule_date === dateStr);
    const groupedByShift = {};

    dateSchedules.forEach(schedule => {
      const shiftId = schedule.shift_id || 0;
      const isSwap = swapList?.some(swap =>
        swap.status === 'approved' &&
        (swap.from_date === dateStr || swap.to_date === dateStr) &&
        (swap.from_staff_id === schedule.staff_id || swap.to_staff_id === schedule.staff_id)
      );
      const isSubstitute = substituteList?.some(sub =>
        sub.status === 'approved' &&
        sub.schedule_date === dateStr &&
        (sub.original_staff_id === schedule.staff_id || sub.substitute_staff_id === schedule.staff_id)
      );

      if (!groupedByShift[shiftId]) {
        groupedByShift[shiftId] = {
          shift_name: schedule.shift_name,
          shift_color: schedule.shift_color,
          schedules: []
        };
      }
      
      groupedByShift[shiftId].schedules.push({
        ...schedule,
        isSwap,
        isSubstitute
      });
    });

    return Object.values(groupedByShift);
  }, []);

  return useMemo(() => ({
    // 状态
    scheduleList: store.scheduleList,
    staffList: store.staffList,
    shiftList: store.shiftList,
    isFetching: store.isFetching,
    currentDate: store.currentDate,
    
    // Actions
    fetchSchedules: store.fetchSchedules.bind(store),
    fetchStaffList: store.fetchStaffList.bind(store),
    fetchShiftList: store.fetchShiftList.bind(store),
    addSchedule: store.addSchedule.bind(store),
    deleteSchedule: store.deleteSchedule.bind(store),
    batchQuerySchedules: store.batchQuerySchedules.bind(store),
    setCurrentDate: store.setCurrentDate.bind(store),
    
    // 计算属性
    getShiftGroups,
  }), [store, getShiftGroups]);
};
```

```python
# middleware/metrics.py
"""性能监控中间件"""

import time
import logging
from django.utils.deprecation import MiddlewareMixin

logger = logging.getLogger('performance')


class PerformanceMetricsMiddleware(MiddlewareMixin):
    """性能指标收集中间件"""
    
    def process_request(self, request):
        request._start_time = time.time()
        return None
    
    def process_response(self, request, response):
        if hasattr(request, '_start_time'):
            duration = time.time() - request._start_time
            
            # 记录慢请求
            if duration > 1.0:  # 超过1秒
                logger.warning(
                    f'Slow request: {request.path} took {duration:.2f}s',
                    extra={
                        'path': request.path,
                        'method': request.method,
                        'duration': duration,
                        'user': request.user.username if request.user.is_authenticated else 'anonymous'
                    }
                )
            
            # 添加到响应头（调试用）
            response['X-Request-Duration'] = f'{duration:.3f}s'
        
        return response
```

```javascript
/**
 * 测量组件渲染时间
 */
export const measureRenderTime = (componentName) => {
  const start = performance.now();
  
  return () => {
    const duration = performance.now() - start;
    
    if (duration > 100) {  // 超过100ms认为是慢渲染
      console.warn(`[Performance] ${componentName} rendered slowly: ${duration.toFixed(2)}ms`);
      
      // 可发送到监控平台
      if (window.gtag) {
        window.gtag('event', 'slow_render', {
          component: componentName,
          duration: Math.round(duration)
        });
      }
    }
    
    return duration;
  };
};

/**
 * 测量API响应时间
 */
export const measureApiCall = async (apiName, apiPromise) => {
  const start = performance.now();
  
  try {
    const result = await apiPromise;
    const duration = performance.now() - start;
    
    if (duration > 1000) {  // 超过1秒
      console.warn(`[Performance] Slow API ${apiName}: ${duration.toFixed(2)}ms`);
    }
    
    return result;
  } catch (error) {
    const duration = performance.now() - start;
    console.error(`[Performance] API ${apiName} failed after ${duration.toFixed(2)}ms`);
    throw error;
  }
};
```

```python
def with_log_context(trace_id=None, tenant_id=None, user_id=None, operation=None):
    """日志上下文装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            logger = logging.getLogger(func.__module__)
            
            extra = {
                'trace_id': trace_id or str(uuid.uuid4())[:8],
                'tenant_id': tenant_id or 'system',
                'user_id': user_id or 'anonymous',
                'operation': operation or func.__name__
            }
            
            logger.info(f'Start {func.__name__}', extra=extra)
            try:
                result = func(*args, **kwargs)
                logger.info(f'End {func.__name__} - Success', extra=extra)
                return result
            except Exception as e:
                logger.error(f'End {func.__name__} - Error: {str(e)}', extra=extra)
                raise
        return wrapper
    return decorator
```

```python
import logging
import time
from functools import wraps
from django.conf import settings

logger = logging.getLogger(__name__)


def retry_on_failure(max_retries=3, delay=1):
    """失败重试装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_retries):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_retries - 1:
                        logger.error(f'Failed after {max_retries} attempts: {e}')
                        raise
                    logger.warning(f'Attempt {attempt + 1} failed, retrying...')
                    time.sleep(delay * (attempt + 1))
        return wrapper
    return decorator


class NotificationServiceV2:
    """增强版通知服务"""
    
    @staticmethod
    @retry_on_failure(max_retries=3, delay=1)
    def notify_swap_created(swap_record):
        """通知换班申请已创建 - 带重试"""
        try:
            # 尝试发送钉钉通知
            if settings.DINGTALK_WEBHOOK:
                NotificationServiceV2._send_dingtalk(swap_record)
            
            # 尝试发送邮件通知
            if settings.EMAIL_ENABLED:
                NotificationServiceV2._send_email(swap_record)
                
        except Exception as e:
            # 降级：仅记录日志，不阻塞主流程
            logger.warning(
                f'Notification failed for swap {swap_record.id}, '
                f'but main process continues. Error: {e}'
            )
    
    @staticmethod
    def _send_dingtalk(swap_record):
        """发送钉钉通知"""
```

```python
# middleware/rate_limit.py
"""批量操作限流中间件"""

import time
from django.http import JsonResponse
from django.core.cache import cache


class BatchOperationRateLimit:
    """批量操作限流"""
    
    # 限制配置
    LIMITS = {
        'batch_delete': {
            'max_items': 100,      # 单次最多删除100条
            'max_per_minute': 5,   # 每分钟最多5次
            'max_per_hour': 50     # 每小时最多50次
        },
        'batch_create': {
            'max_items': 50,       # 单次最多创建50条
            'max_per_minute': 10,
            'max_per_hour': 100
        }
    }
    
    def __init__(self, get_response):
        self.get_response = get_response
    
    def __call__(self, request):
        # 只处理批量操作
        if 'batch' not in request.path:
            return self.get_response(request)
        
        user_id = request.user.id if request.user.is_authenticated else 'anonymous'
        operation = self._get_operation_type(request.path)
        
        # 检查频率限制
        if not self._check_rate_limit(user_id, operation):
            return JsonResponse({
                'error': '操作过于频繁，请稍后再试'
            }, status=429)
        
        # 检查数量限制
        if request.body:
            import json
            try:
                data = json.loads(request.body)
                item_count = self._get_item_count(data)
                limit = self.LIMITS.get(operation, {}).get('max_items', 50)
                
                if item_count > limit:
                    return JsonResponse({
                        'error': f'单次操作数量超过限制（最多{limit}条）'
                    }, status=400)
            except json.JSONDecodeError:
                pass
        
        return self.get_response(request)
    
    def _get_operation_type(self, path):
        """获取操作类型"""
        if 'batch_delete' in path:
            return 'batch_delete'
        if 'batch' in path and ('swap' in path or 'substitute' in path):
            return 'batch_create'
        return 'default'
    
    def _get_item_count(self, data):
        """获取操作项数量"""
        if isinstance(data, dict):
            if 'adjustments' in data:
                return len(data['adjustments'])
            if 'records' in data:
                return len(data['records'])
        return 1
    
    def _check_rate_limit(self, user_id, operation):
        """检查频率限制"""
        limits = self.LIMITS.get(operation, {})
        
        # 每分钟限制
        minute_key = f'rate_limit:{operation}:{user_id}:minute'
        minute_count = cache.get(minute_key, 0)
        if minute_count >= limits.get('max_per_minute', 10):
            return False
        cache.set(minute_key, minute_count + 1, 60)
        
        # 每小时限制
        hour_key = f'rate_limit:{operation}:{user_id}:hour'
        hour_count = cache.get(hour_key, 0)
        if hour_count >= limits.get('max_per_hour', 100):
            return False
        cache.set(hour_key, hour_count + 1, 3600)
        
        return True
```

```python
# middleware/audit_log.py
"""操作审计日志中间件"""

import json
import logging
from django.utils.deprecation import MiddlewareMixin

audit_logger = logging.getLogger('audit')


class AuditLogMiddleware(MiddlewareMixin):
    """审计日志中间件"""
    
    SENSITIVE_FIELDS = ['password', 'token', 'secret', 'credit_card']
    
    def process_response(self, request, response):
        # 只记录修改操作
        if request.method not in ['POST', 'PUT', 'PATCH', 'DELETE']:
            return response
        
        # 跳过健康检查等非业务请求
        if '/health' in request.path or '/api/' not in request.path:
            return response
        
        # 构建审计日志
        audit_entry = {
            'timestamp': timezone.now().isoformat(),
            'user_id': request.user.id if request.user.is_authenticated else None,
            'username': request.user.username if request.user.is_authenticated else 'anonymous',
            'tenant_id': getattr(request.user, 'tenant_id', None),
            'action': request.method,
            'path': request.path,
            'ip': self._get_client_ip(request),
            'user_agent': request.META.get('HTTP_USER_AGENT', ''),
        }
        
        # 记录请求数据（脱敏）
        if request.body:
            try:
                body = json.loads(request.body)
                audit_entry['request_data'] = self._sanitize_data(body)
            except:
                audit_entry['request_data'] = 'parse_error'
        
        # 记录响应状态
        audit_entry['response_status'] = response.status_code
        
        # 记录资源ID（如果响应包含）
        if hasattr(response, 'data') and isinstance(response.data, dict):
            audit_entry['resource_id'] = response.data.get('id')
        
        audit_logger.info(json.dumps(audit_entry, ensure_ascii=False))
        
        return response
    
    def _get_client_ip(self, request):
        """获取客户端IP"""
        x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
        if x_forwarded_for:
            return x_forwarded_for.split(',')[0].strip()
        return request.META.get('REMOTE_ADDR')
    
    def _sanitize_data(self, data):
        """脱敏处理"""
        if not isinstance(data, dict):
            return data
        
        sanitized = {}
        for key, value in data.items():
            if any(sensitive in key.lower() for sensitive in self.SENSITIVE_FIELDS):
                sanitized[key] = '***'
            else:
                sanitized[key] = value
        return sanitized
```
