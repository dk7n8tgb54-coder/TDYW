### 3.3 防抖机制 (P2)

**问题**: 用户快速连续点击"全部暂停"/"全部开始"可能导致竞态条件

**修复代码**:
```javascript
// 新增防抖控制属性
_pauseAllDebounceTimer = null;
_resumeAllDebounceTimer = null;
_isPauseAllRunning = false;
_isResumeAllRunning = false;

// pauseAll 防抖实现
pauseAll = async () => {
  if (this._pauseAllDebounceTimer) {
    clearTimeout(this._pauseAllDebounceTimer);
  }
  
  if (this._isPauseAllRunning) {
    this._pauseAllDebounceTimer = setTimeout(() => {
      this._pauseAllDebounceTimer = null;
      this.pauseAll();
    }, 300);
    return;
  }
  
  this._isPauseAllRunning = true;
  
  try {
    await this._doPauseAll();
  } finally {
    setTimeout(() => {
      this._isPauseAllRunning = false;
    }, 300);
  }
}
```

### 3.4 状态校对机制 (P2)

**问题**: 前后端状态不一致时无自动校正机制

**新增方法**:
```javascript
/**
 * 【P2修复】状态校对机制 - 同步前后端传输状态
 */
@action
async syncTransferStatus(isPublic = false) {
  try {
    const transfers = await this.transferStore.fetchTransfers(isPublic);
    const uploadQueue = this.queueStore.currentUploadQueue;
    
    let syncCount = 0;
    uploadQueue.forEach(item => {
      if (item.transferId) {
        const backendTransfer = transfers.find(t => t.id === item.transferId);
        if (backendTransfer) {
          const backendStatus = this.mapBackendStatus(backendTransfer.status);
          if (backendStatus && backendStatus !== item.status) {
            console.warn(`[syncTransferStatus] 状态不一致: ${item.name}`);
            item.status = backendStatus;
            if (backendStatus === 'completed') {
              item.percent = 100;
              item.error = null;
            }
            syncCount++;
          }
        }
      }
    });
    
    this.failedSyncTransfers = [];
  } catch (error) {
    console.error('[syncTransferStatus] 状态校对失败:', error);
  }
}

/**
 * 后端状态映射到前端状态
 */
mapBackendStatus(backendStatus) {
  const statusMap = {
    'UPLOADING': 'uploading',
    'PAUSED': 'paused',
    'MERGING': 'merging',
    'COMPLETED': 'completed',
    'FAILED': 'error',
    'CANCELED': 'error',
    'PENDING': 'waiting',
  };
  return statusMap[backendStatus] || null;
}
```
