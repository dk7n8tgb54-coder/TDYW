## 修复项 #9：Token改用sessionStorage

### 涉及文件
- `spug_web/src/libs/functools.js`
- `spug_web/src/pages/login/index.js`

### 修改方案

#### 文件：`spug_web/src/libs/functools.js`

**修改前（第17-24行）：**
```javascript
export function updatePermissions() {
  X_TOKEN = localStorage.getItem('token');
  Permission.isReady = true;
  Permission.isSuper = localStorage.getItem('is_supper') === 'true';
  try {
    Permission.permissions = JSON.parse(localStorage.getItem('permissions') || '[]');
  } catch (e) {

  }
}
```

**修改后：**
```javascript
// Use sessionStorage for security - cleared when browser tab closes
// This reduces the window of token exposure compared to localStorage
const tokenStorage = sessionStorage;

export function updatePermissions() {
  X_TOKEN = tokenStorage.getItem('token');
  Permission.isReady = true;
  Permission.isSuper = tokenStorage.getItem('is_supper') === 'true';
  try {
    Permission.permissions = JSON.parse(tokenStorage.getItem('permissions') || '[]');
  } catch (e) {
    Permission.permissions = [];
  }
}
```

#### 文件：`spug_web/src/pages/login/index.js`

**修改前（第63-70行）：**
```javascript
function doLogin(data) {
  localStorage.setItem('id', data['id']);
  localStorage.setItem('token', data['access_token']);
  localStorage.setItem('nickname', data['nickname']);
  localStorage.setItem('is_supper', data['is_supper']);
  localStorage.setItem('tenant_id', data['tenant_id'] || '');
  localStorage.setItem('permissions', JSON.stringify(data['permissions']));
  localStorage.setItem('login_type', loginType);
  updatePermissions();
```

**修改后：**
```javascript
function doLogin(data) {
  sessionStorage.setItem('id', data['id']);
  sessionStorage.setItem('token', data['access_token']);
  sessionStorage.setItem('nickname', data['nickname']);
  sessionStorage.setItem('is_supper', data['is_supper']);
  sessionStorage.setItem('tenant_id', data['tenant_id'] || '');
  sessionStorage.setItem('permissions', JSON.stringify(data['permissions']));
  sessionStorage.setItem('login_type', loginType);
  updatePermissions();
```

**注意：** 还需要搜索项目中所有使用 `localStorage` 读取token相关数据的地方，统一改为 `sessionStorage`。

## 修复项 #10：修复会话固定攻击

### 涉及文件
- `spug_api/apps/account/views.py`

### 修改方案

**修改前（第340-341行）：**
```python
token_isvalid = user.access_token and len(user.access_token) == 32 and user.token_expired >= time.time()
user.access_token = user.access_token if token_isvalid else uuid.uuid4().hex
```

**修改后：**
```python
# SECURITY: Always generate new token on login to prevent session fixation
# This ensures that if an old token was compromised, re-login invalidates it
user.access_token = uuid.uuid4().hex
```

### 说明
每次登录都生成新Token，确保：
- 旧Token立即失效
- 用户可通过重新登录"踢掉"攻击者
- 符合安全最佳实践

## 修复项 #20：优化登录失败锁定策略

### 涉及文件
- `spug_api/apps/account/views.py`

### 修改方案

**修改前（login函数中，第305-314行）：**
```python
value = cache.get_or_set(form.username, 0, 86400)
if value >= 3:
    if user and user.is_active:
        user.is_active = False
        user.save()
    return handle_response(error='账户已被系统禁用')
cache.set(form.username, value + 1, 86400)
return handle_response(error="用户名或密码错误，连续多次错误账户将会被禁用")
```

**修改后：**
```python
# Rate limiting: IP-based (prevent distributed brute force)
x_real_ip = get_request_real_ip(request.headers)
ip_key = f'login_fail:ip:{x_real_ip}'
ip_fails = cache.get(ip_key, 0)
if ip_fails >= 30:
    return handle_response(error='登录尝试过于频繁，请稍后再试')

# Rate limiting: User-based (prevent targeted attacks)
user_key = f'login_fail:user:{form.username}'
user_fails = cache.get(user_key, 0)
if user_fails >= 5:
    remaining_seconds = cache.ttl(user_key) if hasattr(cache, 'ttl') else 900
    minutes = max(1, remaining_seconds // 60)
    return handle_response(error=f'账户已临时锁定，请{minutes}分钟后重试')

# Increment failure counters
cache.set(ip_key, ip_fails + 1, 3600)       # IP: 1 hour window
cache.set(user_key, user_fails + 1, 900)    # User: 15 min window

remaining = 5 - user_fails - 1
if remaining > 0:
    return handle_response(error=f'用户名或密码错误，还剩{remaining}次尝试机会')
else:
    return handle_response(error='账户已临时锁定，请15分钟后重试')
```

### 关键改进

| 改进 | 说明 |
|------|------|
| IP级别限流 | 防止分布式暴力破解（30次/小时/IP） |
| 用户级别限流 | 防止针对性攻击（5次/15分钟/用户） |
| 临时锁定替代永久禁用 | 不再设置 `is_active=False`，15分钟后自动解锁 |
| 登录成功清除计数 | 正确登录后重置失败计数器（已有 `cache.delete(user.username)`） |
| 友好提示 | 告知用户剩余尝试次数和锁定时间 |
