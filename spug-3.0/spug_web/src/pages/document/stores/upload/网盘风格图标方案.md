# 资料库传输列表 - 主流网盘风格设计方案

## 一、主流网盘设计调研

### 百度网盘、阿里云盘、夸克网盘的共同特点

```
┌─────────────────────────────────────────────────────────────┐
│ 📄 文件名.pdf                    45%  |=====>       |  [⏸] [✕]  │
│    2.5 MB / 10 MB    上传中  2.5 MB/s                         │
├─────────────────────────────────────────────────────────────┤
│ 📁 文件夹名称                    80%  |=========>   |  [⏸] [✕]  │
│    文件 8/10    上传中                                       │
├─────────────────────────────────────────────────────────────┤
│ 📄 文档.docx                    100% |===========|  [🗑]       │
│    已完成                                                   │
├─────────────────────────────────────────────────────────────┤
│ 🖼️ 图片.jpg                      30%  |==>          |  [⏵] [✕]  │
│    ⚠️ 上传失败 - 点击重试                                    │
└─────────────────────────────────────────────────────────────┘
```

### 核心设计元素
1. **左侧文件类型图标** - 根据文件类型显示不同图标
2. **双行信息展示** - 文件名 + 状态详情（大小、速度、进度）
3. **进度条整合** - 进度条放在右侧或下方
4. **状态文字明确** - "上传中"、"已完成"、"已暂停"等
5. **操作按钮精简** - 只显示当前可用的操作

## 三、组件代码实现

### 1. 文件类型图标组件

```javascript
// components/FileTypeIcon.js
import React from 'react';
import {
  FilePdfOutlined,
  FileImageOutlined,
  FileWordOutlined,
  FileExcelOutlined,
  FilePptOutlined,
  FileZipOutlined,
  FileTextOutlined,
  VideoCameraOutlined,
  SoundOutlined,
  FolderOutlined,
  FileOutlined,
} from '@ant-design/icons';

// 文件类型图标映射
const FILE_ICONS = {
  // 图片
  'image': { icon: FileImageOutlined, color: '#52c41a' },
  
  // 视频
  'video': { icon: VideoCameraOutlined, color: '#722ed1' },
  
  // 音频
  'audio': { icon: SoundOutlined, color: '#eb2f96' },
  
  // 文档
  'pdf': { icon: FilePdfOutlined, color: '#ff4d4f' },
  'word': { icon: FileWordOutlined, color: '#1890ff' },
  'excel': { icon: FileExcelOutlined, color: '#52c41a' },
  'ppt': { icon: FilePptOutlined, color: '#fa8c16' },
  'text': { icon: FileTextOutlined, color: '#8c8c8c' },
  
  // 压缩包
  'zip': { icon: FileZipOutlined, color: '#faad14' },
  
  // 文件夹
  'folder': { icon: FolderOutlined, color: '#faad14' },
  
  // 默认
  'default': { icon: FileOutlined, color: '#8c8c8c' },
};

/**
 * 获取文件类型配置
 * @param {string} fileName - 文件名
 * @param {string} mimeType - MIME类型
 * @param {boolean} isFolder - 是否文件夹
 */
const getFileTypeConfig = (fileName, mimeType, isFolder) => {
  if (isFolder) return FILE_ICONS['folder'];
  
  const ext = fileName?.split('.').pop()?.toLowerCase() || '';
  
  // 根据扩展名判断
  if (['pdf'].includes(ext)) return FILE_ICONS['pdf'];
  if (['doc', 'docx'].includes(ext)) return FILE_ICONS['word'];
  if (['xls', 'xlsx', 'csv'].includes(ext)) return FILE_ICONS['excel'];
  if (['ppt', 'pptx'].includes(ext)) return FILE_ICONS['ppt'];
  if (['txt', 'md', 'log'].includes(ext)) return FILE_ICONS['text'];
  if (['zip', 'rar', '7z', 'tar', 'gz'].includes(ext)) return FILE_ICONS['zip'];
  
  // 根据 MIME 类型判断
  if (mimeType) {
    if (mimeType.startsWith('image/')) return FILE_ICONS['image'];
    if (mimeType.startsWith('video/')) return FILE_ICONS['video'];
    if (mimeType.startsWith('audio/')) return FILE_ICONS['audio'];
    if (mimeType.includes('pdf')) return FILE_ICONS['pdf'];
    if (mimeType.includes('word') || mimeType.includes('document')) return FILE_ICONS['word'];
    if (mimeType.includes('excel') || mimeType.includes('spreadsheet')) return FILE_ICONS['excel'];
    if (mimeType.includes('powerpoint') || mimeType.includes('presentation')) return FILE_ICONS['ppt'];
    if (mimeType.startsWith('text/')) return FILE_ICONS['text'];
  }
  
  return FILE_ICONS['default'];
};

/**
 * 文件类型图标组件
 */
const FileTypeIcon = ({ fileName, mimeType, isFolder, size = 40 }) => {
  const config = getFileTypeConfig(fileName, mimeType, isFolder);
  const IconComponent = config.icon;
  
  return (
    <div
      style={{
        width: size,
        height: size,
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        backgroundColor: `${config.color}15`,  // 15 = 10% 透明度
        borderRadius: 8,
        flexShrink: 0,
      }}
    >
      <IconComponent style={{ fontSize: size * 0.5, color: config.color }} />
    </div>
  );
};

export default FileTypeIcon;
```

### 2. 传输列表项组件

```javascript
// components/TransferItem.js
import React from 'react';
import { Progress, Button, Tooltip } from 'antd';
import {
  PauseOutlined,
  CaretRightOutlined,
  CloseOutlined,
  DeleteOutlined,
  RedoOutlined,
  LoadingOutlined,
} from '@ant-design/icons';
import FileTypeIcon from './FileTypeIcon';

// 状态配置
const STATUS_CONFIG = {
  waiting: {
    text: '等待中',
    color: '#8c8c8c',
    showProgress: false,
    showSpeed: false,
  },
  calculating: {
    text: '计算中',
    color: '#fa8c16',
    showProgress: true,
    showSpeed: false,
    icon: <LoadingOutlined spin />,
  },
  uploading: {
    text: '上传中',
    color: '#1890ff',
    showProgress: true,
    showSpeed: true,
    icon: <LoadingOutlined spin />,
  },
  paused: {
    text: '已暂停',
    color: '#faad14',
    showProgress: true,
    showSpeed: false,
  },
  completed: {
    text: '已完成',
    color: '#52c41a',
    showProgress: true,
    showSpeed: false,
  },
  error: {
    text: '上传失败',
    color: '#ff4d4f',
    showProgress: true,
    showSpeed: false,
  },
  merging: {
    text: '合并中',
    color: '#722ed1',
    showProgress: true,
    showSpeed: false,
    icon: <LoadingOutlined spin />,
  },
};

/**
 * 格式化文件大小
 */
const formatSize = (bytes) => {
  if (bytes === 0) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
};

/**
 * 格式化速度
 */
const formatSpeed = (bytesPerSecond) => {
  if (!bytesPerSecond) return '';
  return formatSize(bytesPerSecond) + '/s';
};

/**
 * 传输列表项组件（网盘风格）
 */
const TransferItem = ({
  item,
  onPause,
  onResume,
  onCancel,
  onRemove,
  onRetry,
}) => {
  const status = item.status;
  const config = STATUS_CONFIG[status];
  
  // 计算进度显示
  const percent = Math.round(item.percent || 0);
  const uploadedSize = item.fileSize ? Math.round(item.fileSize * (percent / 100)) : 0;
  
  // 判断显示哪些操作按钮
  const canPause = ['waiting', 'calculating', 'uploading', 'merging'].includes(status);
  const canResume = ['paused', 'error', 'waiting'].includes(status);
  const canCancel = ['waiting', 'calculating', 'uploading', 'paused', 'merging'].includes(status);
  const canRemove = ['completed', 'error', 'paused'].includes(status);
  const canRetry = status === 'error';
  
  return (
    <div
      style={{
        padding: '12px 16px',
        borderBottom: '1px solid #f0f0f0',
        backgroundColor: status === 'error' ? '#fff2f0' : 'transparent',
        transition: 'background-color 0.3s',
      }}
    >
      {/* 第一行：图标 + 文件名 + 百分比 + 操作按钮 */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 12 }}>
        {/* 文件图标 */}
        <FileTypeIcon 
          fileName={item.name} 
          isFolder={item.isFolder}
          size={40} 
        />
        
        {/* 文件名和状态信息 */}
        <div style={{ flex: 1, minWidth: 0 }}>
          <div
            style={{
              fontSize: 14,
              fontWeight: 500,
              color: status === 'error' ? '#ff4d4f' : '#262626',
              overflow: 'hidden',
              textOverflow: 'ellipsis',
              whiteSpace: 'nowrap',
              marginBottom: 4,
            }}
            title={item.name}
          >
            {item.name}
          </div>
          
          {/* 状态详情：已传输 / 总大小 • 状态文字 • 速度 */}
          <div style={{ fontSize: 12, color: '#8c8c8c', display: 'flex', gap: 8 }}>
            {/* 大小信息 */}
            {item.fileSize ? (
              <span>
                {formatSize(uploadedSize)} / {formatSize(item.fileSize)}
              </span>
            ) : item.isFolder ? (
              <span>文件 {item.currentFile || 0}/{item.totalFiles || 0}</span>
            ) : null}
            
            <span>•</span>
            
            {/* 状态文字 */}
            <span style={{ color: config.color, display: 'flex', alignItems: 'center', gap: 4 }}>
              {config.icon}
              {config.text}
            </span>
            
            {/* 上传速度 */}
            {config.showSpeed && item.speed > 0 && (
              <>
                <span>•</span>
                <span style={{ color: '#1890ff' }}>{formatSpeed(item.speed)}</span>
              </>
            )}
          </div>
        </div>
        
        {/* 百分比 */}
        {config.showProgress && (
          <div
            style={{
              fontSize: 14,
              fontWeight: 500,
              color: config.color,
              minWidth: 45,
              textAlign: 'right',
            }}
          >
            {percent}%
          </div>
        )}
        
        {/* 操作按钮组 */}
        <div style={{ display: 'flex', gap: 4 }}>
          {canPause && (
            <Tooltip title="暂停">
              <Button
                type="text"
                size="small"
                icon={<PauseOutlined />}
                onClick={() => onPause(item.id)}
                style={{ color: '#595959' }}
              />
            </Tooltip>
          )}
          
          {canResume && (
            <Tooltip title={status === 'error' ? '重试' : '开始'}>
              <Button
                type="text"
                size="small"
                icon={<CaretRightOutlined />}
                onClick={() => onResume(item.id)}
                style={{ color: '#52c41a' }}
              />
            </Tooltip>
          )}
          
          {canCancel && (
            <Tooltip title="取消">
              <Button
                type="text"
                size="small"
                danger
                icon={<CloseOutlined />}
                onClick={() => onCancel(item.id)}
              />
            </Tooltip>
          )}
          
          {canRemove && (
            <Tooltip title="删除">
              <Button
                type="text"
                size="small"
                danger
                icon={<DeleteOutlined />}
                onClick={() => onRemove(item.id)}
              />
            </Tooltip>
          )}
        </div>
      </div>
      
      {/* 第二行：进度条 */}
      {config.showProgress && (
        <div style={{ marginTop: 8, marginLeft: 52 }}>
          <Progress
            percent={percent}
            size={{ height: 4 }}
            strokeColor={status === 'error' ? '#ff4d4f' : config.color}
            trailColor="#f0f0f0"
            showInfo={false}
            status={status === 'error' ? 'exception' : undefined}
          />
        </div>
      )}
      
      {/* 错误提示 */}
      {status === 'error' && item.error && (
        <div
          style={{
            marginTop: 8,
            marginLeft: 52,
            padding: '6px 12px',
            backgroundColor: '#fff',
            borderRadius: 4,
            border: '1px solid #ffccc7',
            display: 'flex',
            alignItems: 'center',
            gap: 8,
          }}
        >
          <span style={{ fontSize: 12, color: '#ff4d4f', flex: 1 }}>
            ⚠️ {item.error}
          </span>
          {canRetry && (
            <Button
              type="link"
              size="small"
              icon={<RedoOutlined />}
              onClick={() => onRetry(item.id)}
              style={{ padding: 0, height: 'auto', fontSize: 12 }}
            >
              重试
            </Button>
          )}
        </div>
      )}
    </div>
  );
};

export default TransferItem;
```

### 3. 分组传输列表组件

```javascript
// components/TransferList.js
import React from 'react';
import { Collapse, Empty, Button, Tooltip, Space } from 'antd';
import {
  DeleteOutlined,
  ClearOutlined,
  PauseCircleOutlined,
  PlayCircleOutlined,
} from '@ant-design/icons';
import TransferItem from './TransferItem';

const { Panel } = Collapse;

/**
 * 分组传输列表组件
 */
const TransferList = ({
  uploadingItems = [],
  completedItems = [],
  errorItems = [],
  onPause,
  onResume,
  onCancel,
  onRemove,
  onPauseAll,
  onResumeAll,
  onClearCompleted,
}) => {
  // 渲染列表项
  const renderItems = (items) => {
    if (items.length === 0) return null;
    
    return items.map(item => (
      <TransferItem
        key={item.id}
        item={item}
        onPause={onPause}
        onResume={onResume}
        onCancel={onCancel}
        onRemove={onRemove}
        onRetry={onResume}
      />
    ));
  };
  
  // 全部为空时的空状态
  if (uploadingItems.length === 0 && completedItems.length === 0 && errorItems.length === 0) {
    return (
      <Empty
        image={Empty.PRESENTED_IMAGE_SIMPLE}
        description="暂无传输任务"
        style={{ padding: '60px 0' }}
      />
    );
  }
  
  return (
    <div>
      {/* 全局操作栏 */}
      {uploadingItems.length > 0 && (
        <div
          style={{
            padding: '8px 16px',
            borderBottom: '1px solid #f0f0f0',
            backgroundColor: '#fafafa',
            display: 'flex',
            justifyContent: 'space-between',
            alignItems: 'center',
          }}
        >
          <span style={{ fontSize: 13, color: '#262626', fontWeight: 500 }}>
            正在传输 {uploadingItems.length} 个文件
          </span>
          <Space>
            <Tooltip title="全部暂停">
              <Button
                type="text"
                size="small"
                icon={<PauseCircleOutlined />}
                onClick={onPauseAll}
              >
                全部暂停
              </Button>
            </Tooltip>
          </Space>
        </div>
      )}
      
      {/* 传输中列表 */}
      {uploadingItems.length > 0 && (
        <div style={{ backgroundColor: '#fff' }}>
          {renderItems(uploadingItems)}
        </div>
      )}
      
      {/* 已暂停列表 */}
      {uploadingItems.filter(i => i.status === 'paused').length > 0 && (
        <div
          style={{
            padding: '8px 16px',
            borderTop: '1px solid #f0f0f0',
            backgroundColor: '#fafafa',
            display: 'flex',
            justifyContent: 'space-between',
            alignItems: 'center',
          }}
        >
          <span style={{ fontSize: 13, color: '#262626' }}>
            已暂停 {uploadingItems.filter(i => i.status === 'paused').length} 个文件
          </span>
          <Button
            type="link"
            size="small"
            icon={<PlayCircleOutlined />}
            onClick={onResumeAll}
          >
            全部开始
          </Button>
        </div>
      )}
      
      {/* 已完成 - 可折叠 */}
      {completedItems.length > 0 && (
        <Collapse
          defaultActiveKey={[]}
          ghost
          style={{ borderTop: '1px solid #f0f0f0' }}
        >
          <Panel
            header={
              <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                <span style={{ fontSize: 13, color: '#262626', fontWeight: 500 }}>
                  已完成 ({completedItems.length})
                </span>
                <Tooltip title="清空已完成">
                  <Button
                    type="text"
                    size="small"
                    icon={<ClearOutlined />}
                    onClick={(e) => {
                      e.stopPropagation();
                      onClearCompleted();
                    }}
                  >
                    清空
                  </Button>
                </Tooltip>
              </div>
            }
            key="completed"
            style={{ backgroundColor: '#fafafa' }}
          >
            {renderItems(completedItems)}
          </Panel>
        </Collapse>
      )}
      
      {/* 传输失败 */}
      {errorItems.length > 0 && (
        <>
          <div
            style={{
              padding: '8px 16px',
              borderTop: '1px solid #f0f0f0',
              backgroundColor: '#fff2f0',
            }}
          >
            <span style={{ fontSize: 13, color: '#ff4d4f', fontWeight: 500 }}>
              传输失败 ({errorItems.length})
            </span>
          </div>
          <div style={{ backgroundColor: '#fff' }}>
            {renderItems(errorItems)}
          </div>
        </>
      )}
    </div>
  );
};

export default TransferList;
```

### 4. 主面板整合

```javascript
// UploadPanel.netdisk.js - 网盘风格版本
import React from 'react';
import { observer } from 'mobx-react';
import { Card, Tag } from 'antd';
import { CloudUploadOutlined, CloudOutlined } from '@ant-design/icons';
import { uploadCoreStore } from './stores';
import navigationStore from './stores/navigation';
import TransferList from './components/TransferList';

@observer
class UploadPanel extends React.Component {
  // ... 原有的事件处理方法

  render() {
    const { currentUploadQueue, isPaused } = uploadCoreStore;
    
    // 按状态分组
    const uploadingItems = currentUploadQueue.filter(
      item => ['waiting', 'calculating', 'uploading', 'paused', 'merging'].includes(item.status)
    );
    const completedItems = currentUploadQueue.filter(
      item => item.status === 'completed'
    );
    const errorItems = currentUploadQueue.filter(
      item => item.status === 'error'
    );
    
    // 空间信息
    const spaceType = navigationStore.isPublic ? '公共共享库' : '我的文件';
    const spaceColor = navigationStore.isPublic ? 'gold' : 'blue';
    const spaceIcon = navigationStore.isPublic ? <CloudOutlined /> : <CloudUploadOutlined />;
    
    return (
      <Card
        size="small"
        style={{ 
          width: 500,  // 加宽以容纳更多信息
          height: 600, 
          display: 'flex', 
          flexDirection: 'column' 
        }}
        bodyStyle={{ 
          padding: 0, 
          flex: 1, 
          display: 'flex', 
          flexDirection: 'column', 
          overflow: 'hidden' 
        }}
        title={
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            <Tag 
              icon={spaceIcon}
              color={spaceColor} 
              style={{ margin: 0, fontSize: 12 }}
            >
              {spaceType}
            </Tag>
            <span style={{ fontSize: 12, color: '#8c8c8c' }}>
              共 {currentUploadQueue.length} 个任务
            </span>
          </div>
        }
      >
        <TransferList
          uploadingItems={uploadingItems}
          completedItems={completedItems}
          errorItems={errorItems}
          onPause={(id) => uploadCoreStore.pauseItem(id)}
          onResume={(id) => uploadCoreStore.resumeItem(id)}
          onCancel={(id) => uploadCoreStore.cancelItem(id)}
          onRemove={(id) => uploadCoreStore.removeItem(id)}
          onPauseAll={() => uploadCoreStore.pauseAll()}
          onResumeAll={() => uploadCoreStore.resumeAll()}
          onClearCompleted={() => uploadCoreStore.removeCompleted()}
        />
      </Card>
    );
  }
}

export default UploadPanel;
```
