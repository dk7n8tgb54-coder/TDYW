```python
# apps/document/services/merge_status_handlers.py
from abc import ABC, abstractmethod

class StatusHandler(ABC):
    @abstractmethod
    def handle(self, result, response):
        pass

class PendingHandler(StatusHandler):
    def handle(self, result, response):
        response['message'] = '任务等待中'

class StartedHandler(StatusHandler):
    def handle(self, result, response):
        response['message'] = '任务开始执行'
        response['progress'] = 5

class ProgressHandler(StatusHandler):
    def handle(self, result, response):
        meta = result.info or {}
        response.update(meta)

class SuccessHandler(StatusHandler):
    def handle(self, result, response):
        result_data = result.result or {}
        response['status'] = result_data.get('status', 'completed').lower()
        response['file_id'] = result_data.get('file_id')
        response['file_name'] = result_data.get('file_name')

class FailureHandler(StatusHandler):
    def handle(self, result, response):
        response['status'] = 'failed'
        response['error'] = str(result.result)
```

#### 步骤2: 创建状态处理工厂
```python
class StatusHandlerFactory:
    _handlers = {
        'PENDING': PendingHandler,
        'STARTED': StartedHandler,
        'PROGRESS': ProgressHandler,
        'SUCCESS': SuccessHandler,
        'FAILURE': FailureHandler,
    }
    
    @classmethod
    def get_handler(cls, state):
        handler_class = cls._handlers.get(state)
        return handler_class() if handler_class else None
```

#### 步骤3: 重构 get 方法
```python
def get(self, request):
    task_id = self._resolve_task_id(request)
    if not task_id:

    result = AsyncResult(task_id)
    response = {'task_id': task_id, 'status': result.state.lower()}

```

```python
class PermissionChecker:
    """权限检查器"""
    
    def __init__(self, user, is_private):
        self.user = user
        self.is_private = is_private
        self.user_tenant_id = getattr(user, 'tenant_id', None)
    
    def can_delete_file(self, file_obj):
        """检查是否有权限删除文件"""
        if self.is_private:
            return self.user_tenant_id is not None and file_obj.tenant_id == self.user_tenant_id
        else:
            return self.user.is_supper or file_obj.created_by == self.user
    
    def can_delete_folder(self, folder):
        """检查是否有权限删除文件夹"""
        if self.is_private:
            return self.user_tenant_id is not None and folder.tenant_id == self.user_tenant_id
        else:
            return self.user.is_supper or folder.created_by == self.user
```

#### 步骤2: 创建 FolderCollector 类

```python
class FolderCollector:
    """文件夹收集器 - BFS遍历"""
    
    @staticmethod
    def collect_all_subfolders(folder, FolderModel):
        """收集所有子文件夹"""
        folder_queue = [folder]
        all_folders = []
        
        while folder_queue:
            current = folder_queue.pop(0)
            all_folders.append(current)
            sub_folders = FolderModel.all_objects.filter(parent=current, is_deleted=True)
            folder_queue.extend(sub_folders)
        
        return all_folders
```

```python
def _delete_folder_contents_iterative(folder, FolderModel, FileModel, user):
    is_private = FolderModel == DocumentFolderPrivate
    permission_checker = PermissionChecker(user, is_private)
    
    # 收集所有文件夹
    all_folders = FolderCollector.collect_all_subfolders(folder, FolderModel)
    
    # 删除内容
    deleter = ContentDeleter(permission_checker)
    for current_folder in reversed(all_folders):
        deleter.delete_files_in_folder(current_folder, FileModel)
        if current_folder.id != folder.id:
            deleter.delete_folder(current_folder)
    
    return deleter.freed_size, deleter.deleted_count
```

```python
# apps/document/services/directory_cleaner.py
class DirectoryCleaner:
    """目录清理器基类"""
    
    def __init__(self, max_age):
        self.max_age = max_age
        self.current_time = time.time()
        self.cleaned_count = 0
        self.errors = []
    
    def should_clean(self, path):
        """判断是否应该清理"""
        dir_age = self.current_time - os.path.getmtime(path)
        return dir_age > self.max_age
    
    def clean(self, path):
        """执行清理"""
        raise NotImplementedError
    
    def get_stats(self):
        """获取清理统计"""
        return {
            'cleaned_count': self.cleaned_count,
            'errors': self.errors
        }
```

#### 步骤2: 创建具体清理器
```python
class ChunkDirectoryCleaner(DirectoryCleaner):
    """分片目录清理器"""
    
    def clean(self, path):
        try:
            shutil.rmtree(path, ignore_errors=True)
            self.cleaned_count += 1
        except Exception as e:
            self.errors.append(f'Failed to cleanup {path}: {str(e)}')

class TaskFileCleaner(DirectoryCleaner):
    """任务文件清理器"""
    
    def clean(self, path):
        try:
            os.remove(path)
            self.cleaned_count += 1
        except Exception as e:
            self.errors.append(f'Failed to delete task {path}: {str(e)}')
```

#### 步骤3: 创建 DirectoryTraverser 类
```python
class DirectoryTraverser:
    """目录遍历器"""
    
    @staticmethod
    def traverse_two_level(base_dir, cleaner, file_pattern=None):
        """遍历两层目录结构"""
        if not os.path.exists(base_dir):
            return
        
        for first_level in os.listdir(base_dir):
            first_path = os.path.join(base_dir, first_level)
            if not os.path.isdir(first_path):
                continue
            
            for second_level in os.listdir(first_path):
                second_path = os.path.join(first_path, second_level)
                
                if file_pattern and not second_level.endswith(file_pattern):
                    continue
                if not os.path.exists(second_path):
                    continue
                
                if cleaner.should_clean(second_path):
                    cleaner.clean(second_path)
```

#### 步骤4: 重构 cleanup_old_chunks
```python
@shared_task(bind=True, soft_time_limit=1800, time_limit=3600, queue='document.cleanup')
def cleanup_old_chunks(self, days=1):
    max_age = days * 24 * 60 * 60
    
    # 清理分片目录
    chunk_cleaner = ChunkDirectoryCleaner(max_age)
    DirectoryTraverser.traverse_two_level(
        os.path.join(settings.BASE_DIR, 'storage', 'document_chunks'),
        chunk_cleaner
    )
    
    # 清理任务文件
    task_cleaner = TaskFileCleaner(max_age)
    chunk_base_dir = os.path.join(settings.BASE_DIR, 'storage', 'document_merge_tasks')
    if os.path.exists(chunk_base_dir):
        for task_file in os.listdir(chunk_base_dir):
            if not task_file.endswith('.task'):
                continue
            task_path = os.path.join(chunk_base_dir, task_file)
            if task_cleaner.should_clean(task_path):
                task_cleaner.clean(task_path)
    
    result = {
        'status': 'success',
        'cleaned_dirs': chunk_cleaner.cleaned_count,
        'cleaned_tasks': task_cleaner.cleaned_count,
        'errors': chunk_cleaner.errors + task_cleaner.errors
    }
    
    logger.info(f'[Celery] Cleanup old chunks completed: {result}')
    return result
```
