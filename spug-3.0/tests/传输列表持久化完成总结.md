**当前状态**: 未开始

**目标**:

- 确保页面刷新时正确恢复对应租户的传输记录

**修改方案**:

```javascript
// 修改前
@observable uploadQueue = [];

// 修改后
@observable uploadQueue = {};  // { [tenantId]: [uploadItems...] }
@observable currentTenantId = 'default';

// 获取当前租户的队列
@computed
get currentUploadQueue() {
  return this.uploadQueue[this.currentTenantId] || [];
}

// 添加到队列
@action.bound
addToQueue(item) {
  const tenantId = item.tenantId || this.currentTenantId;
  if (!this.uploadQueue[tenantId]) {
    this.uploadQueue[tenantId] = [];
  }
  this.uploadQueue[tenantId].push(item);
}

// 从队列删除
@action.bound
removeFromQueue(uploadId, tenantId = null) {
  const targetTenantId = tenantId || this.currentTenantId;
  if (this.uploadQueue[targetTenantId]) {
    this.uploadQueue[targetTenantId] = this.uploadQueue[targetTenantId]
      .filter(item => item.id !== uploadId);
  }
}
```

- `UploadCoreStore.js` - 核心上传逻辑
