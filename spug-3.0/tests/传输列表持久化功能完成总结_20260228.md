# 传输列表持久化功能完成总结

## 一、项目背景

资料库模块的传输列表（上传/下载任务）存在以下问题：
1. ❌ 页面刷新后传输列表丢失
2. ❌ 无后端传输记录，无法查询历史
3. ❌ 前端队列未按租户组织，违反多租户隔离原则
4. ❌ 分片存储未按租户隔离，相同哈希的分片会冲突

// Explorer.js - 页面加载时恢复传输列表
@action.bound
async loadTransferList() {
  try {
    const response = await http.get('/api/document/transfers/');
    const transfers = response.data || [];

    // 按租户分组
    const transfersByTenant = {};
    transfers.forEach(t => {
      if (!transfersByTenant[t.tenant_id]) {
        transfersByTenant[t.tenant_id] = [];
      }
      transfersByTenant[t.tenant_id].push(t);
    });

    // 恢复到 uploadQueue
    Object.keys(transfersByTenant).forEach(tenantId => {
      if (!uploadCoreStore.uploadQueue[tenantId]) {
        uploadCoreStore.uploadQueue[tenantId] = [];
      }
      // 将 API 返回的传输记录转换为上传任务格式
      const uploadTasks = transfersByTenant[tenantId].map(t => ({
        id: t.id,
        file: { name: t.file_name, size: t.file_size },
        name: t.file_name,
        percent: t.progress,
        status: t.status,
        tenantId: t.tenant_id,
        // ... 其他字段
      }));
      uploadCoreStore.uploadQueue[tenantId].push(...uploadTasks);
    });
  } catch (error) {
    console.error('[传输] 恢复传输列表失败:', error);
  }
}

// UploadCoreStore.js - 上传过程中更新进度
async uploadChunk(uploadId, chunkIndex, totalChunks) {
  // ... 上传逻辑 ...

  // 计算进度
  const progress = Math.round((chunkIndex + 1) / totalChunks * 100);

  // 调用 API 更新进度
  try {
    await http.post(`/api/document/transfers/${uploadId}/progress/`, {
      progress: progress,
      uploaded_chunks: chunkIndex + 1,
      transferred_size: (chunkIndex + 1) * this.CHUNK_SIZE,
      speed: calculateSpeed(),
    });
  } catch (error) {
    console.error('[传输] 更新进度失败:', error);
  }
}

// UploadCoreStore.js - 完成传输
async completeUpload(uploadId) {
  try {
    await http.post(`/api/document/transfers/${uploadId}/complete/`);
    console.log('[传输] 传输已完成:', uploadId);
  } catch (error) {
    console.error('[传输] 完成传输失败:', error);
  }
}

// UploadCoreStore.js - 取消传输
async cancelUpload(uploadId) {
  try {
    await http.post(`/api/document/transfers/${uploadId}/cancel/`);
    console.log('[传输] 传输已取消:', uploadId);
  } catch (error) {
    console.error('[传输] 取消传输失败:', error);
  }
}
