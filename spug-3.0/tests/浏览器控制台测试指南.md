# 浏览器控制台测试指南

## 一、准备工作

### 1. 打开浏览器
- 访问 Spug 系统: http://localhost/#/document
- 登录系统
- 进入文档管理页面

### 2. 打开开发者工具
- **Chrome/Edge**: 按 `F12` 或 `Ctrl+Shift+I`
- **Firefox**: 按 `F12` 或 `Ctrl+Shift+K`
- 切换到 **Console（控制台）** 标签

### 3. 验证 store 是否可用
在控制台输入：
```javascript
window.documentStore
```

如果返回对象则表示已加载，如果返回 `undefined` 需要等待页面完全加载。

## 四、多文件并发上传测试

### 测试4: 并发上传5个文件
```javascript
// 创建多个测试文件
function createMultipleFiles(count = 5, size = 100 * 1024) {
  const files = [];
  for (let i = 0; i < count; i++) {
    const buffer = new Uint8Array(size);
    for (let j = 0; j < size; j++) {
      buffer[j] = Math.floor(Math.random() * 256);
    }
    files.push(new File([buffer], `multi-test-${Date.now()}-${i}.bin`, {
      type: 'application/octet-stream'
    }));
  }
  return files;
}

// 并发上传测试
async function testConcurrentUpload() {
  console.log('🚀 开始并发上传测试');
  console.time('总耗时');
  
  const files = createMultipleFiles(5, 100 * 1024); // 5个100KB文件
  console.log(`📦 准备上传 ${files.length} 个文件`);
  
  // 触发上传
  documentStore.handleFilesUpload(files);
  
  // 监控上传进度
  const monitor = setInterval(() => {
    const queue = documentStore.uploadQueue;
    const uploading = documentStore.uploadingFiles;
    const uploaded = documentStore.uploadedFiles;
    
    // 显示每个上传文件的进度
    if (uploading.length > 0) {
      console.log(`\n📊 正在上传 (${uploading.length}):`);
      uploading.forEach((file, idx) => {
        const progress = ((file.uploadedSize / file.totalSize) * 100).toFixed(2);
        console.log(`  [${idx + 1}] ${file.name}: ${progress}% (${file.uploadedSize}/${file.totalSize})`);
      });
    }
    
    console.log(`\n⏳ 队列: ${queue.length} | 上传中: ${uploading.length} | 已完成: ${uploaded.length}`);
    
    // 检查是否全部完成
    if (queue.length === 0 && uploading.length === 0) {
      clearInterval(monitor);
      console.timeEnd('总耗时');
      console.log('✅ 所有文件上传完成!');
      console.log('已上传列表:', uploaded.map(f => ({
        name: f.name,
        size: f.totalSize,
        status: f.status
      })));
    }
  }, 500);
}

// 执行测试
testConcurrentUpload();
```

```javascript
// 完整测试套件
class UploadTestSuite {
  constructor() {
    this.results = [];
  }
  
  log(message, type = 'info') {
    const icons = {
      info: 'ℹ️',
      success: '✅',
      error: '❌',
      warning: '⚠️',
      test: '🧪'
    };
    console.log(`${icons[type]} ${message}`);
  }
  
  createTestFile(size, name) {
    const buffer = new Uint8Array(size);
    for (let i = 0; i < size; i++) {
      buffer[i] = Math.floor(Math.random() * 256);
    }
    return new File([buffer], name, { type: 'application/octet-stream' });
  }
  
  async waitForUploadComplete(timeout = 60000) {
    const startTime = Date.now();
    return new Promise((resolve, reject) => {
      const check = () => {
        if (documentStore.uploadQueue.length === 0 && 
            documentStore.uploadingFiles.length === 0) {
          resolve(true);
        } else if (Date.now() - startTime > timeout) {
          reject(new Error('上传超时'));
        } else {
          setTimeout(check, 500);
        }
      };
      check();
    });
  }
  
  async test1_SingleFile() {
    this.log('测试1: 单文件上传', 'test');
    try {
      const file = this.createTestFile(50 * 1024, 'single-test.bin');
      documentStore.handleFilesUpload([file]);
      await this.waitForUploadComplete();
      this.log('单文件上传成功', 'success');
      this.results.push({ name: '单文件上传', status: '通过' });
    } catch (e) {
      this.log(`单文件上传失败: ${e.message}`, 'error');
      this.results.push({ name: '单文件上传', status: '失败', error: e.message });
    }
  }
  
  async test2_MultipleFiles() {
    this.log('测试2: 多文件并发上传', 'test');
    try {
      const files = [
        this.createTestFile(100 * 1024, 'multi-1.bin'),
        this.createTestFile(100 * 1024, 'multi-2.bin'),
        this.createTestFile(100 * 1024, 'multi-3.bin')
      ];
      documentStore.handleFilesUpload(files);
      await this.waitForUploadComplete(120000);
      this.log('多文件上传成功', 'success');
      this.results.push({ name: '多文件上传', status: '通过' });
    } catch (e) {
      this.log(`多文件上传失败: ${e.message}`, 'error');
      this.results.push({ name: '多文件上传', status: '失败', error: e.message });
    }
  }
  
  async test3_LargeFile() {
    this.log('测试3: 大文件上传 (1MB)', 'test');
    try {
      const file = this.createTestFile(1024 * 1024, 'large-file.bin');
      documentStore.handleFilesUpload([file]);
      await this.waitForUploadComplete(120000);
      this.log('大文件上传成功', 'success');
      this.results.push({ name: '大文件上传', status: '通过' });
    } catch (e) {
      this.log(`大文件上传失败: ${e.message}`, 'error');
      this.results.push({ name: '大文件上传', status: '失败', error: e.message });
    }
  }
  
  async test4_ConcurrencyControl() {
    this.log('测试4: 并发控制 (10个文件)', 'test');
    try {
      const files = [];
      for (let i = 0; i < 10; i++) {
        files.push(this.createTestFile(50 * 1024, `concurrent-${i}.bin`));
      }
      documentStore.handleFilesUpload(files);
      
      // 检查并发数
      await new Promise(resolve => setTimeout(resolve, 1000));
      const concurrentCount = documentStore.uploadingFiles.length;
      this.log(`并发上传数: ${concurrentCount} (限制: ${documentStore.maxConcurrentUploads})`, 'info');
      
      await this.waitForUploadComplete(180000);
      this.log('并发控制测试成功', 'success');
      this.results.push({ name: '并发控制', status: '通过' });
    } catch (e) {
      this.log(`并发控制测试失败: ${e.message}`, 'error');
      this.results.push({ name: '并发控制', status: '失败', error: e.message });
    }
  }
  
  async runAll() {
    this.log('========================================', 'info');
    this.log('开始运行完整测试套件', 'info');
    this.log('========================================', 'info');
    
    await this.test1_SingleFile();
    await new Promise(r => setTimeout(r, 2000)); // 等待间隔
    
    await this.test2_MultipleFiles();
    await new Promise(r => setTimeout(r, 2000));
    
    await this.test3_LargeFile();
    await new Promise(r => setTimeout(r, 2000));
    
    await this.test4_ConcurrencyControl();
    
    this.log('========================================', 'info');
    this.log('测试套件完成', 'info');
    this.log('========================================', 'info');
    
    // 输出测试报告
    console.table(this.results);
    
    const passed = this.results.filter(r => r.status === '通过').length;
    const total = this.results.length;
    this.log(`通过率: ${passed}/${total} (${(passed/total*100).toFixed(1)}%)`, 
              passed === total ? 'success' : 'warning');
  }
}

// 运行完整测试套件
const suite = new UploadTestSuite();
suite.runAll();
```

## 十、常见问题

### Q1: 控制台提示 documentStore 未定义
**A**: 确保页面已完全加载，并且你已经在文档管理页面。可以尝试:
```javascript
// 等待 store 加载
function waitForStore() {
  if (window.documentStore) {
    console.log('✅ store 已加载');
    return true;
  }
  setTimeout(waitForStore, 500);
}
waitForStore();
```

### Q2: 上传文件后看不到进度
**A**: 可能是页面未刷新或未监听到状态变化。尝试:
```javascript
// 手动触发 MobX 响应
documentStore.forceUpdate && documentStore.forceUpdate();
```

### Q3: 如何清理测试上传的文件?
**A**: 直接在页面上删除上传的文件，或者使用后端 API 删除:
```javascript
// 获取已上传的文件 ID
const uploadedFileIds = documentStore.uploadedFiles.map(f => f.id);
console.log('待删除文件 ID:', uploadedFileIds);
// 然后通过界面或 API 删除
```

1. **直接访问内部状态** - 可以查看所有 MobX store 的数据
