# 文件夹上传架构重构实施方案 v5（生产就绪版）

## 设计原则

1. **单一职责** (SRP) - 每个类只做一件事
2. **代码精简** - 文件<1000行，函数<200行
3. **解耦设计** - 减少相互调用，依赖注入
4. **消除重复** - 公共逻辑抽取为工具函数
5. **性能优化** - 减少遍历、缓存结果、并发控制
6. **纯函数工具** - 工具函数与 UI 层解耦
7. **内存安全** - 避免内存泄漏，及时清理引用
8. **容错兜底** - 空值校验、异常清理、降级处理

import { UPLOAD_CONSTANTS } from '../constants';

/**
 * 检查是否为非空文件列表
 * @param {*} files 
 * @returns {boolean}
 */
export function isValidFileList(files) {
  return files && Array.isArray(files) && files.length > 0;
}

/**
 * 安全获取文件数组
 * @param {*} files 
 * @returns {Array}
 */
export function safeGetFiles(files) {
  return isValidFileList(files) ? files : [];
}

/**
 * 安全清理路径
 * 循环移除危险字符，直到无路径遍历风险
 * 修复：处理多级嵌套危险字符（如 ....// 会被替换为 ../）
 * @param {string} path 
 * @returns {string}
 */
export function sanitizePath(path) {
  if (!path || typeof path !== 'string') {
    return '';
  }
  
  // 统一路径分隔符，移除空字符
  let sanitized = path.replace(/\\/g, '/').replace(/\x00/g, '');
  
  // 循环移除，直到无../和开头的.
  const dangerousPattern = /\.\.\/|^\.+/g;
  while (dangerousPattern.test(sanitized)) {
    sanitized = sanitized.replace(dangerousPattern, '');
  }
  
  return sanitized;
}

/**
 * 路径安全校验
 * 对每一级路径单独校验，彻底防止路径遍历
 * 修复：拆分后逐层校验，避免绕过风险
 * @param {string} path - 路径或文件夹名
 * @returns {{valid: boolean, error?: string, sanitizedPath?: string}}
 */
export function validatePathSecurity(path) {
  if (!path || typeof path !== 'string') {
    return { valid: false, error: '路径不能为空' };
  }

  // 先清理路径，再拆分校验
  const sanitizedPath = sanitizePath(path);
  const pathParts = sanitizedPath.split('/').filter(p => p !== '');

  // 校验每一级文件夹/文件名
  const invalidNamePattern = /[<>:"|?*]/; // 系统禁止的文件名字符
  for (const part of pathParts) {
    if (invalidNamePattern.test(part)) {
      return { valid: false, error: `路径包含非法字符: ${part}` };
    }
    if (part.length > 255) {
      return { valid: false, error: `文件名/文件夹名长度超过限制: ${part}` };
    }
  }

  // 校验全路径长度
  if (sanitizedPath.length > UPLOAD_CONSTANTS.MAX_PATH_LENGTH) {
    return { valid: false, error: '路径总长度超过限制' };
  }

  return { valid: true, sanitizedPath };
}

/**
 * 批量路径安全校验
 * 分离有效/无效文件，不过滤则继续处理
 * @param {File[]} files - 文件列表
 * @returns {{valid: File[], invalid: {file: File, reason: string}[], hasInvalid: boolean}}
 */
export function validatePathsBatch(files) {
  const valid = [];
  const invalid = [];
  
  for (const file of files) {
    const relativePath = file.webkitRelativePath || file.name;
    const result = validatePathSecurity(relativePath);
    
    if (!result.valid) {
      invalid.push({ file, reason: result.error });
    } else {
      valid.push(file);
    }
  }
  
  return { valid, invalid, hasInvalid: invalid.length > 0 };
}

/**
 * 文件夹解析工具
 * 纯函数，无副作用，可测试
 * 修复：接受配置参数，不再直接依赖UPLOAD_CONSTANTS
 */

import { UPLOAD_CONSTANTS } from '../constants';

/**
 * 解析文件列表，按文件夹分组
 * @param {FileList} files - 原始文件列表
 * @returns {Map<string, File[]>} 文件夹路径 -> 文件列表
 */
export function groupFilesByFolder(files) {
  const groups = new Map();
  
  for (const file of files) {
    const path = file.webkitRelativePath || '';
    const folderPath = path.split('/').slice(0, -1).join('/');
    
    if (!groups.has(folderPath)) {
      groups.set(folderPath, []);
    }
    groups.get(folderPath).push(file);
  }
  
  return groups;
}

/**
 * 提取所有需要创建的文件夹路径（按深度排序）
 * @param {Map} fileGroups - 分组后的文件
 * @returns {string[]} 排序后的文件夹路径
 */
export function extractFolderPaths(fileGroups) {
  const paths = Array.from(fileGroups.keys()).filter(p => p !== '');
  
  // 按深度排序（父文件夹在前）
  return paths.sort((a, b) => {
    const depthA = a.split('/').length;
    const depthB = b.split('/').length;
    return depthA - depthB;
  });
}

/**
 * 计算字符串简单哈希
 * @param {string} str 
 * @returns {number}
 */
function simpleHash(str) {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash = hash & hash;
  }
  return Math.abs(hash);
}

/**
 * 生成文件夹唯一标识（同步函数）
 * 改进：增加文件修改时间采样、整体哈希，大幅降低误判概率
 * 
 * @param {File[]} files - 文件列表
 * @param {string|number} folderId - 目标文件夹ID
 * @param {boolean} isPublic - 是否公共空间
 * @returns {string} 唯一标识
 */
export function generateFolderKey(files, folderId, isPublic) {
  // 空列表兜底
  if (!files || files.length === 0) {
    return `folder-empty-${Date.now()}-${folderId}-${isPublic ? '1' : '0'}`;
  }
  
  const topFolderName = files[0]?.webkitRelativePath?.split('/')[0] || 'unknown';
  const count = files.length;
  const totalSize = files.reduce((sum, f) => sum + f.size, 0);
  
  // 改进采样策略：增加文件修改时间和哈希，减少误判
  const sampleCount = Math.min(files.length, 5); // 增加到前5个文件
  const samples = files.slice(0, sampleCount).map(f => {
    const lastModified = f.lastModified || 0;
    return `${f.name}-${f.size}-${lastModified}`;
  }).join('|');
  
  // 增加整体哈希采样（基于文件名和大小）
  const allNames = files.slice(0, 10).map(f => f.name).join('');
  const nameHash = simpleHash(allNames);
  
  return `folder-${topFolderName}-${count}-${totalSize}-${samples}-${nameHash}-${folderId}-${isPublic ? '1' : '0'}`;
}

/**
 * 分批处理大文件列表，避免阻塞主线程
 * @param {Array} items - 待处理项目
 * @param {Function} processor - 处理函数
 * @param {number} batchSize - 每批大小
 * @param {Object} config - 配置对象（可选，用于动态配置）
 * @returns {Promise<Array>} 处理结果
 */
export async function processInBatches(items, processor, batchSize = UPLOAD_CONSTANTS.BATCH_PROCESS_SIZE, config = null) {
  // 修复：如果传入配置，优先使用配置中的BATCH_PROCESS_SIZE
  const actualBatchSize = (config && config.BATCH_PROCESS_SIZE) || batchSize;
  const results = [];
  
  for (let i = 0; i < items.length; i += actualBatchSize) {
    const batch = items.slice(i, i + actualBatchSize);
    const batchResults = await Promise.all(batch.map(processor));
    results.push(...batchResults);
    
    // 让出主线程，避免阻塞UI
    if (i + actualBatchSize < items.length) {
      await new Promise(resolve => setTimeout(resolve, 0));
    }
  }
  
  return results;
}

/**
 * 构建文件信息对象
 */
export function buildFileInfo(file, folderId, relativePath) {
  return {
    file,
    name: file.name,
    size: file.size,
    folderId,
    relativePath: relativePath || file.name,
  };
}

/**
 * 检查文件是否重复
 * @param {Array} fileInfos - 文件信息对象数组
 * @param {Function} generateKeyFn - 生成唯一键的函数
 * @returns {{unique: Array, duplicates: Array}} 
 */
export function checkDuplicates(fileInfos, generateKeyFn) {
  const seen = new Set();
  const unique = [];
  const duplicates = [];
  
  for (const fileInfo of fileInfos) {
    const key = generateKeyFn(fileInfo);
    if (seen.has(key)) {
      duplicates.push(fileInfo);
    } else {
      seen.add(key);
      unique.push(fileInfo);
    }
  }
  
  return { unique, duplicates };
}

    console.warn('[validators] 外部upload-utils不可用，使用默认校验');

 * 批量校验文件名

  const invalid = [];

### 6. 服务 - FolderCreator.js (200行)

```javascript
/**
 * 文件夹创建服务
 * 职责：在服务器创建文件夹结构，支持失败回滚、幂等性检查
 * 关键修复：区分「本次创建」和「历史复用」的文件夹ID，避免误删用户历史数据
 * 性能优化：同深度文件夹并发创建
 */
import { UPLOAD_CONSTANTS, UploadErrorCode } from '../constants';
import pLimit from 'p-limit';

export class FolderCreator {
  constructor(httpClient, options = {}) {
    this.http = httpClient;
    this.createdFolders = new Map(); // path -> id (所有已处理的文件夹)
    this.createdByThisInstance = new Set(); // 仅本次POST创建的文件夹ID，回滚仅删此集合
    this.reusedFolderIds = new Set(); // 历史复用的文件夹ID，禁止回滚删除
    this.failedRollbackIds = [];
    this.enableServerCheck = options.enableServerCheck !== false;
    this.serverFolderCache = new Map();
    // 修复：支持注入配置，优先使用注入的配置
    this.config = options.config || UPLOAD_CONSTANTS;
  }
  
  /**
   * 重置实例状态，避免跨任务污染
   */
  resetState() {
    this.createdFolders.clear();
    this.createdByThisInstance.clear();
    this.reusedFolderIds.clear();
    this.failedRollbackIds = [];
    this.serverFolderCache.clear();
  }
  
  /**
   * 批量创建文件夹（优化：同深度并发创建）
   * 修复：按深度分组，同深度并发，不同深度串行，既保证父子顺序又提升性能
   */
  async createFolders(folderPaths, baseFolderId, isPublic) {
    this.resetState();
    
    // 按深度分组
    const depthGroups = new Map();
    folderPaths.forEach(path => {
      const depth = path.split('/').length;
      if (!depthGroups.has(depth)) depthGroups.set(depth, []);
      depthGroups.get(depth).push(path);
    });
    
    // 按深度从小到大排序
    const sortedDepths = Array.from(depthGroups.keys()).sort((a, b) => a - b);
    
    try {
      for (const depth of sortedDepths) {
        const paths = depthGroups.get(depth);
        // 修复：同深度并发创建，并发数可配置
        const limit = pLimit(this.config.FOLDER_CREATE_CONCURRENCY);
        await Promise.all(
          paths.map(path => limit(() => this.createSingleFolder(path, baseFolderId, isPublic)))
        );
      }
      return this.createdFolders;
    } catch (error) {
      await this.rollback();
      throw error;
    }
  }
  
  /**
   * 检查服务端是否已存在同名文件夹
   */
  async checkExistingFolder(name, parentId, isPublic) {
    if (!this.enableServerCheck) return null;
    
    const cacheKey = `${parentId}_${name}`;
    if (this.serverFolderCache.has(cacheKey)) {
      return this.serverFolderCache.get(cacheKey);
    }
    
    try {
      const response = await this.http.get('/api/document/folder/', {
        params: { parent_id: parentId, name, is_public: isPublic }
      });
      
      if (response.results && response.results.length > 0) {
        const existingId = response.results[0].id;
        this.serverFolderCache.set(cacheKey, existingId);
        return existingId;
      }
    } catch (e) {
      console.warn('[FolderCreator] 查询现有文件夹失败:', e);
    }
    
    return null;
  }
  
  /**
   * 创建单个文件夹（带缓存和服务端幂等检查）
   * 修复：区分本次创建和历史复用
   */
  async createSingleFolder(folderPath, baseFolderId, isPublic) {
    if (this.createdFolders.has(folderPath)) {
      return this.createdFolders.get(folderPath);
    }
    
    const parts = folderPath.split('/');
    let currentPath = '';
    let parentId = baseFolderId;
    
    for (const name of parts) {
      currentPath = currentPath ? `${currentPath}/${name}` : name;
      
      // 检查本地缓存
      if (this.createdFolders.has(currentPath)) {
        parentId = this.createdFolders.get(currentPath);
        continue;
      }
      
      // 服务端幂等检查
      const existingId = await this.checkExistingFolder(name, parentId, isPublic);
      if (existingId) {
        // 修复：复用历史文件夹时，仅加入复用集合，不加入回滚集合
        this.createdFolders.set(currentPath, existingId);
        this.reusedFolderIds.add(existingId);
        parentId = existingId;
        continue;
      }
      
      // 创建文件夹
      const result = await this.http.post('/api/document/folder/', {
        name,
        parent_id: parentId,
        is_public: isPublic,
      });
      
      // 修复：本次创建的文件夹，加入专属回滚集合
      this.createdFolders.set(currentPath, result.id);
      this.createdByThisInstance.add(result.id);
      parentId = result.id;
    }
    
    return parentId;
  }
  
  /**
   * 获取文件所属文件夹ID
   */
  getFolderId(relativePath, baseFolderId) {
    const folderPath = relativePath.split('/').slice(0, -1).join('/');
    return this.createdFolders.get(folderPath) || baseFolderId;
  }
  
  /**
   * 获取回滚失败的文件夹ID
   */
  getFailedRollbackIds() {
    return [...this.failedRollbackIds];
  }
  
  /**
   * 失败回滚 - 仅删除本次创建的文件夹
   * 修复：遍历 createdByThisInstance 而非全部ID，使用注入的配置
   */
  async rollback() {
    console.warn('[FolderCreator] 回滚本次创建的文件夹:', this.createdByThisInstance.size);
    this.failedRollbackIds = [];
    
    // 修复：仅遍历本次创建的文件夹ID
    for (const id of [...this.createdByThisInstance].reverse()) {
      let success = false;
      let attempts = 0;
      
      while (!success && attempts < this.config.ROLLBACK_RETRY_ATTEMPTS) {
        try {
          await this.http.delete(`/api/document/folder/${id}/`);
          success = true;
        } catch (e) {
          attempts++;
          console.error(`回滚文件夹 ${id} 失败(尝试${attempts}):`, e);
          
          if (attempts < this.config.ROLLBACK_RETRY_ATTEMPTS) {
            const delay = this.config.ROLLBACK_RETRY_DELAY * Math.pow(2, attempts - 1);
            await new Promise(resolve => setTimeout(resolve, delay));
          }
        }
      }
      
      if (!success) {
        this.failedRollbackIds.push(id);
        if (typeof window !== 'undefined' && window.monitor) {
          window.monitor.reportError('FOLDER_ROLLBACK_FAILED', { folderId: id });
        }
      }
    }
    
    if (this.failedRollbackIds.length > 0) {
      console.error('[FolderCreator] 以下文件夹回滚失败，需手动清理:', this.failedRollbackIds);
    }
    
    this.createdFolders.clear();
    this.createdByThisInstance.clear();
    this.reusedFolderIds.clear();
  }
}

  constructor(coreStore, options = {}) {

   * 预处理文件夹上传
   * 修复：批量路径校验，过滤无效文件而非全量拦截
   * @returns {Promise<{fileQueue: FileInfo[], folderKey: string}>} 
   */
  @action
  async preprocess(files, targetFolderId, isPublic) {
    // 1. 空文件列表校验
    if (!isValidFileList(files)) {
      return { fileQueue: [], folderKey: null };
    }
    
    // 2. 文件名合法性校验
    const { valid: validFilesByName, invalid: invalidFilesByName, hasInvalid: hasInvalidName } = validateFiles(files);
    
    if (hasInvalidName) {
      const errorMsg = getValidationErrorMessage(invalidFilesByName);
      if (errorMsg) this.showMessage('error', errorMsg);
    }
    
    if (validFilesByName.length === 0) {
      return { fileQueue: [], folderKey: null };
    }
    
    // 3. 路径安全校验（修复：批量校验，过滤无效文件而非全量拦截）
    const { valid: validFiles, invalid: invalidPaths, hasInvalid: hasInvalidPath } = validatePathsBatch(validFilesByName);
    
    if (hasInvalidPath) {
      // 仅提示无效文件，不拦截全量
      const invalidNames = invalidPaths.map(item => item.file.name).slice(0, 3);
      const moreCount = invalidPaths.length - invalidNames.length;
      let msg = `以下文件路径非法：${invalidNames.join('、')}`;
      if (moreCount > 0) msg += ` 等${moreCount}个文件`;
      this.showMessage('warning', msg);
    }
    
    if (validFiles.length === 0) {
      this.showMessage('error', '没有合法的文件可上传');
      return { fileQueue: [], folderKey: null };
    }
    
    // 4. 解析文件结构
    const fileGroups = groupFilesByFolder(validFiles);
    const folderPaths = extractFolderPaths(fileGroups);
    
    // 5. 生成文件夹唯一标识（同步函数）
    const folderKey = generateFolderKey(validFiles, targetFolderId, isPublic);
    
    if (this.core.queueStore.uploadingUniqueKeys.has(folderKey)) {
      throw new Error(UploadErrorCode.FOLDER_ALREADY_IN_QUEUE);
    }
    
    this.core.queueStore.uploadingUniqueKeys.add(folderKey);
    
    // 6. 创建服务器文件夹结构
    const httpClient = this.core.http || this.core.httpClient;
    if (!httpClient) {
      this.core.queueStore.uploadingUniqueKeys.delete(folderKey);
      throw new Error(UploadErrorCode.HTTP_CLIENT_NOT_FOUND);
    }
    
    // 修复：传递配置实例给FolderCreator
    const folderCreator = new FolderCreator(httpClient, { config: this.config });
    
    try {
      await folderCreator.createFolders(folderPaths, targetFolderId, isPublic);
      
      // 检查回滚失败
      const failedIds = folderCreator.getFailedRollbackIds();
      if (failedIds.length > 0) {
        console.warn('[FolderPreprocessor] 部分文件夹回滚失败:', failedIds);
      }
    } catch (error) {
      this.core.queueStore.uploadingUniqueKeys.delete(folderKey);
      
      const failedIds = folderCreator.getFailedRollbackIds();
      if (failedIds.length > 0) {
        console.error('[FolderPreprocessor] 回滚失败的文件夹ID:', failedIds);
      }
      
      throw error;
    }
    
    // 7. 构建标准化文件队列
    const fileQueue = [];
    for (const file of validFiles) {
      const folderId = folderCreator.getFolderId(
        file.webkitRelativePath || file.name,
        targetFolderId
      );
      
      fileQueue.push(buildFileInfo(file, folderId, file.webkitRelativePath));
    }
    
    return { fileQueue, folderKey };
  }
}

export class GlobalConcurrencyLimiter {
  constructor(maxConcurrency = UPLOAD_CONSTANTS.GLOBAL_CONCURRENCY_LIMIT) {
    this.limit = pLimit(maxConcurrency);
    this.activeCount = 0;
    this.maxConcurrency = maxConcurrency;
  }
  
  async acquire(fn) {
    this.activeCount++;
    try {
      return await this.limit(fn);
    } finally {
      this.activeCount--;
    }
  }
  
  getActiveCount() {
    return this.activeCount;
  }

/**
 * 带重试的异步操作
 * @param {Function} fn - 执行函数
 * @param {number} maxRetries - 最大重试次数
 * @param {string} operationName - 操作名称（用于日志）
 * @param {number} retryDelay - 基础重试延迟
 * @returns {Promise<any>}
 */
async function withRetry(fn, maxRetries, operationName = 'operation', retryDelay = 1000) {
  let lastError;
  
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error;
      
      if (attempt < maxRetries) {
        const delay = retryDelay * Math.pow(2, attempt);
        console.warn(`[withRetry] ${operationName} 失败，${delay}ms后重试(尝试${attempt + 1}/${maxRetries}):`, error.message);
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
  }
  
  throw lastError;
}

        const duplicateNames = duplicates.map(d => d.name);
        const warningMsg = getDuplicateWarningMessage(duplicateNames);
        if (warningMsg) this.showMessage('warning', warningMsg);
      }
      
      if (unique.length === 0) {
        if (folderKey) {
          this.queueManager.cleanupFolderKey(folderKey);
        }
        return;
      }
      
      // 4. 显示上传面板
      this.showUploadPanel();
      this.core.isPaused = false;
      this.core.isCancelled = false;
      
      // 修复：pendingFiles仅存储文件元数据，不存储完整File对象，避免内存暴涨
      this.core.pendingFiles = unique.map(f => ({
        name: f.name,
        size: f.size,
        uploadId: f.id || null,
      }));
      
      // 5. 加入队列
      const { items, failedItems } = await this.queueManager.addToQueue(unique, isPublic, folderKey);
      
      // 注册队列项状态变更监听，用于清理唯一键
      this._setupQueueItemListeners(items, folderKey);
      
      // 提示失败项（如有）
      if (failedItems.length > 0) {
        console.warn('[UploadCoordinator] 部分文件创建传输记录失败:', failedItems);
        if (failedItems.length <= 3) {
          const names = failedItems.map(f => f.fileInfo.name).join('、');
          this.showMessage('warning', `以下文件未能加入队列：${names}`);
        } else {
          this.showMessage('warning', `${failedItems.length} 个文件未能加入队列`);
        }
      }
      
    } catch (error) {
      if (folderKey) {
        this.queueManager.cleanupFolderKey(folderKey);
      }
      
      // 特定错误码处理
      switch (error.message) {
        case UploadErrorCode.FOLDER_ALREADY_IN_QUEUE:
          this.showMessage('warning', '该文件夹已在队列中');
          break;
        case UploadErrorCode.HTTP_CLIENT_NOT_FOUND:
          this.showMessage('error', '上传服务初始化失败，请刷新页面重试');
          break;
        case UploadErrorCode.INVALID_TENANT_ID:
          this.showMessage('error', '租户信息无效，请重新登录');
          break;
        case UploadErrorCode.INVALID_PATH:
          this.showMessage('error', '文件路径包含非法字符');
          break;
        case UploadErrorCode.STATE_MACHINE_CREATE_FAILED:
          this.showMessage('error', '上传状态初始化失败，请重试');
          break;
        case UploadErrorCode.STATE_MACHINE_MANAGER_NOT_FOUND:
          this.showMessage('error', '上传管理器未初始化，请刷新页面');
          break;
        default:
          this.showMessage('error', `上传失败: ${error.message}`);
      }
      console.error('[UploadCoordinator] 上传失败:', error);
    }
  }
  
  /**
   * 注册队列项状态监听，在成功/失败/取消时清理唯一键
   * 修复：收集销毁方法，支持全量清理，增加超时机制
   */
  _setupQueueItemListeners(items, folderKey) {
    // 最大上传时长：24小时
    const MAX_UPLOAD_DURATION = 24 * 60 * 60 * 1000;
    
    items.forEach(item => {
      // 修复：超时自动清理，避免极端场景内存泄漏
      const timeoutTimer = setTimeout(() => {
        this.queueManager.cleanupUniqueKey(item);
        this._removeDisposer(dispose);
        dispose?.();
        console.warn(`[UploadCoordinator] 任务${item.id}超时，自动清理`);
      }, MAX_UPLOAD_DURATION);

      const dispose = this.core.queueStore.watchItemStatus?.(item.id, (status) => {
        if (['completed', 'error', 'cancelled'].includes(status)) {
          clearTimeout(timeoutTimer); // 终态清除超时定时器
          this.queueManager.cleanupUniqueKey(item);
          if (folderKey && this._areAllItemsFinished(items)) {
            this.queueManager.cleanupFolderKey(folderKey);
          }
          // 执行后从列表移除
          this._removeDisposer(dispose);
          dispose?.();
        }
      });

      if (dispose) {
        this._listenerDisposers.push(dispose);
      }
    });
  }
  
  /**
   * 辅助方法：移除已执行的销毁方法
   */
  _removeDisposer(dispose) {
    const index = this._listenerDisposers.indexOf(dispose);
    if (index > -1) {
      this._listenerDisposers.splice(index, 1);
    }
  }
  
  /**
   * 检查所有队列项是否已完成
   */
  _areAllItemsFinished(items) {
    return items.every(item => {
      const currentItem = this.core.queueStore.getItem?.(item.id);
      return currentItem && ['completed', 'error', 'cancelled'].includes(currentItem.status);
    });
  }
  
  showUploadPanel() {
    const uploadUIStore = this.core.rootStore?.uploadUIStore;
    if (!uploadUIStore?.showUploadPanel) {
      console.warn('[UploadCoordinator] uploadUIStore.showUploadPanel 未定义');
      return;
    }
    uploadUIStore.showUploadPanel();
  }
  
  clearPendingFiles() {
    this.core.pendingFiles = [];
  }
}

export default UploadCoordinator;
```
