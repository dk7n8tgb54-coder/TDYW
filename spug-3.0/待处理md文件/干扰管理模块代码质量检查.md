```python
from functools import wraps
from django.db import DatabaseError

def handle_exceptions(view_func):
    """统一异常处理装饰器"""
    @wraps(view_func)
    def wrapper(*args, **kwargs):
        try:
            return view_func(*args, **kwargs)
        except ValueError as e:
            logger.warning(f'[Interference] 参数错误: {e}')
            return json_response(error='参数格式错误')
        except DatabaseError as e:
            logger.error(f'[Interference] 数据库错误: {e}')
            return json_response(error='数据库操作失败')
        except Exception as e:
            logger.exception(f'[Interference] 未知错误: {e}')
            return json_response(error='服务器内部错误')
    return wrapper

class InterferenceView(View):
    @auth('interference.interference.view')
    @handle_exceptions
    def get(self, request):
        # 原有逻辑
        pass
```
