## 一、现状分析

### 1.1 当前问题

| 文件 | 大小 | 行数 | 问题 | 复杂度 |
|------|------|------|------|--------|
| `views/file.py` | ~29KB | 800+ | 包含删除、上传、下载、预览、复制、移动、重命名 | 高 |
| `views/folder.py` | ~35KB | 900+ | 包含CRUD、列表、复制、移动、下载、重命名 | 高 |
| `views/upload.py` | ~33KB | 850+ | 包含分片上传、合并、断点续传、锁管理 | 高 |
| `views/recycle_bin.py` | ~34KB | 850+ | 包含列表、恢复、永久删除、统计 | 中高 |
| `views/transfer.py` | ~25KB | 600+ | 包含10+个视图类 | 中高 |
| `views/base.py` | ~8KB | 200+ | 包含过多工具函数 | 中 |

### 1.2 过大文件的影响

1. **代码冲突**: 多人协作时容易产生合并冲突
2. **测试困难**: 难以编写单元测试，覆盖率难以提升
3. **阅读成本高**: 新成员难以快速理解代码
4. **热更新慢**: 开发模式下文件变更检测慢
5. **代码复用难**: 功能耦合，难以复用

## 二、优化目标

### 2.1 核心目标

1. **单一职责**: 每个文件只负责一个功能点
2. **可测试性**: 便于编写单元测试
3. **可维护性**: 降低代码阅读成本
4. **可扩展性**: 新增功能不影响现有代码

### 2.2 量化指标

| 指标 | 当前 | 目标 | 说明 |
|------|------|------|------|
| 单文件最大行数 | 2000+ | <500 | 后端视图文件 |
| 单文件最大KB | 35KB | <20KB | 后端视图文件 |
| 文件数量 | 6 | 30+ | views/目录 |

from functools import wraps
from django.http import JsonResponse

def handle_view_errors(func):
    """视图错误处理装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            logger.error(f'[Document] View error: {e}', exc_info=True)

**视图层简化示例**：
```python
# views/file/copy.py
from django.views import View
from django.http import JsonResponse
from ..services.file_service import FileService
from ..serializers.file_serializers import FileCopySerializer

class FileCopyView(View):
    """文件复制视图"""
    
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.service = FileService()
    
    @handle_view_errors
    def post(self, request):
        # 1. 数据验证
        serializer = FileCopySerializer(data=request.POST)
        if not serializer.is_valid():
            return JsonResponse({'error': serializer.errors}, status=400)
        
        # 2. 调用Service层
        try:
            new_file = self.service.copy_file(
                file_id=serializer.validated_data['file_id'],
                target_folder_id=serializer.validated_data.get('target_folder_id'),
                user=request.user
            )
        except FileNotFoundError as e:
            return JsonResponse({'error': str(e)}, status=404)
        except PermissionError as e:
            return JsonResponse({'error': str(e)}, status=403)
        
        # 3. 返回响应
        return JsonResponse({
            'id': new_file.id,
            'name': new_file.display_name,
            'message': '复制成功'
        })
```

### 11.2 序列化器（Serializers）

**使用Django REST Framework或自定义序列化器**：
```python
# serializers/file_serializers.py
from rest_framework import serializers
from ..models import DocumentFile

class FileCopySerializer(serializers.Serializer):
    """文件复制序列化器"""
    file_id = serializers.IntegerField(required=True, min_value=1)
    target_folder_id = serializers.IntegerField(required=False, allow_null=True)
    
    def validate_file_id(self, value):
        """验证文件是否存在"""
        if not DocumentFile.objects.filter(id=value).exists():
            raise serializers.ValidationError('文件不存在')
        return value

class FileUploadSerializer(serializers.Serializer):
    """文件上传序列化器"""
    file_name = serializers.CharField(max_length=255)
    file_size = serializers.IntegerField(min_value=1, max_value=10*1024*1024*1024)
    folder_id = serializers.IntegerField(required=False, allow_null=True)
    
    def validate_file_name(self, value):
        """验证文件名合法性"""
        if '/' in value or '\\' in value:
            raise serializers.ValidationError('文件名包含非法字符')
        return value
```

### 11.3 Repository模式（数据访问层）

```python
# repositories/file_repository.py
from typing import Optional, List, QuerySet
from ..models import DocumentFile

class FileRepository:
    """文件数据访问仓库"""
    
    def get_by_id(self, file_id: int, user) -> Optional[DocumentFile]:
        """根据ID获取文件"""
        try:
            return DocumentFile.objects.get(
                id=file_id,
                tenant_id=user.tenant_id if not user.is_public else None,
                is_deleted=False
            )
        except DocumentFile.DoesNotExist:
            return None
    
    def get_by_folder(self, folder_id: Optional[int], user) -> QuerySet:
        """获取文件夹下的文件列表"""
        queryset = DocumentFile.objects.filter(
            folder_id=folder_id,
            is_deleted=False
        )
        if not user.is_public:
            queryset = queryset.filter(tenant_id=user.tenant_id)
        return queryset.order_by('-created_at')
    
    def create(self, **kwargs) -> DocumentFile:
        """创建文件记录"""
        return DocumentFile.objects.create(**kwargs)
    
    def update(self, file_id: int, **kwargs) -> int:
        """更新文件记录"""
        return DocumentFile.objects.filter(id=file_id).update(**kwargs)
```

# services/cache_service.py
from django.core.cache import cache
from functools import wraps
import hashlib
import json

def cache_view_result(timeout: int = 60):
    """视图结果缓存装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(self, request, *args, **kwargs):
            # 生成缓存key
            cache_key = f"{func.__name__}:{request.user.id}:{hashlib.md5(str(request.GET).encode()).hexdigest()}"
            
            # 尝试从缓存获取
            result = cache.get(cache_key)
            if result is not None:
                return JsonResponse(result)
            
            # 执行视图函数
            response = func(self, request, *args, **kwargs)
            
            # 缓存结果（仅限200响应）
            if response.status_code == 200:
                cache.set(cache_key, json.loads(response.content), timeout)
            
            return response
        return wrapper
    return decorator

# 使用示例
class FileView(View):
    @cache_view_result(timeout=300)  # 缓存5分钟
    def get(self, request):
        # 列表查询逻辑...

                    f'用户: {user.username} | '
                    f'资源: {resource_type} | '
                    f'耗时: {(datetime.now() - start_time).total_seconds():.3f}s'

                    f'[操作失败] {action} | '
                    f'用户: {user.username} | '
                    f'资源: {resource_type} | '
                    f'错误: {str(e)}',
                    exc_info=True
                )
                raise
        return wrapper
    return decorator

### 11.9 异步视图（Django 3.1+）

**对于IO密集型操作，使用异步视图提升性能**：
```python
# views/async_file.py
from django.http import JsonResponse
from asgiref.sync import sync_to_async
import asyncio

class AsyncFileDownloadView(View):
    """异步文件下载视图"""
    
    async def get(self, request):
        file_id = request.GET.get('id')
        
        # 异步获取文件信息
        file_info = await self.get_file_info_async(file_id)
        
        if not file_info:
            return JsonResponse({'error': '文件不存在'}, status=404)
        
        # 异步读取文件
        content = await self.read_file_async(file_info['path'])
        
        response = HttpResponse(content, content_type=file_info['mime_type'])
        response['Content-Disposition'] = f'attachment; filename="{file_info["name"]}"'
        return response
    
    @sync_to_async
    def get_file_info_async(self, file_id):
        """异步包装的数据库查询"""
        try:
            file_obj = DocumentFile.objects.get(id=file_id, is_deleted=False)
            return {
                'id': file_obj.id,
                'name': file_obj.display_name,
                'path': file_obj.physical_path,
                'mime_type': file_obj.mime_type,
            }
        except DocumentFile.DoesNotExist:
            return None
    
    async def read_file_async(self, file_path):
        """异步读取文件"""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(None, self._read_file, file_path)
    
    def _read_file(self, file_path):
        """同步读取文件"""
        with open(file_path, 'rb') as f:
            return f.read()
```
