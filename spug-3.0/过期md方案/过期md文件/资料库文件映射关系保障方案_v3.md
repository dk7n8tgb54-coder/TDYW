# 资料库文件映射关系保障方案 V3（最终版）

> **版本**: v3.0（审计优化后）  
> **适用**: Spug 资料库模块  
> **目标**: 生产级零风险落地，保障数据库与物理文件映射关系长期稳定

---

## 一、审计优化说明

基于专业审计报告，V3版本针对以下7项整改项进行了优化：

| 整改项 | 优先级 | 状态 | 说明 |
|--------|--------|------|------|
| 模型字段添加 `editable=False` | 高 | ✅ 已采纳 | 从根源防止物理名/路径被修改 |
| 软删除查询全覆盖 | 中 | ✅ 已采纳 | 所有接口默认过滤已删除文件 |
| 移动操作异常捕获 | 高 | ✅ 已采纳 | 高并发下移动功能稳定 |
| 定时清理任务 | 中 | ✅ 已采纳 | 防止磁盘空间耗尽 |
| 迁移脚本健壮性 | 低 | ✅ 已采纳 | 支持环境变量自动配置 |
| 校验脚本分批处理 | 低 | ✅ 已采纳 | 提升大数据量校验效率 |
| 唯一索引兼容说明 | 低 | ✅ 已采纳 | 明确MySQL null值处理行为 |

## 二、核心铁律（必须遵守）

```
┌─────────────────────────────────────────────────────────────┐
│                    映射关系保障核心                          │
├─────────────────────────────────────────────────────────────┤
│  1. physical_name 一旦生成，终身不允许修改                   │
│  2. file_path 生成后应保持稳定（尽量只改 folder_id）         │
│  3. 业务名（display_name）可以随意修改                       │
│  4. 所有文件操作必须通过数据库记录驱动                       │
│  5. 删除使用软删除，物理文件延迟清理                         │
│  6. 任何操作异常必须捕获并记录，禁止静默失败                 │
└─────────────────────────────────────────────────────────────┘
```

```python
class SoftDeletedManager(models.Manager):
    """默认管理器：只返回未删除的记录"""
    def get_queryset(self):
        return super().get_queryset().filter(is_deleted=False)


class AllObjectsManager(models.Manager):
    """全量管理器：返回所有记录（包括已删除）"""
    def get_queryset(self):
        return super().get_queryset()

        null=True,
        blank=True,
        help_text='删除时间',
        verbose_name='删除时间'
    )

    created_at = models.DateTimeField(auto_now_add=True, verbose_name='上传时间')
    updated_at = models.DateTimeField(auto_now=True, verbose_name='更新时间')
    
    # ========== 自定义管理器（审计优化：双管理器）==========
    objects = SoftDeletedManager()      # 默认：不包含已删除
    all_objects = AllObjectsManager()   # 全量：包含已删除
    
    class Meta:
        db_table = 'spug_document_file_private'
        verbose_name = '文档文件(私有)'
        verbose_name_plural = '文档文件(私有)'
        ordering = ['-created_at']
    
    def __str__(self):
        return self.display_name or self.name
    
    def delete(self, *args, **kwargs):
        """
        重写删除方法：默认软删除
        
        Args:
            hard: 是否硬删除（默认False软删除）
        """
        hard = kwargs.pop('hard', False)
        
        if hard:
            # 硬删除：调用父类方法
            super().delete(*args, **kwargs)
        else:
            # 软删除：标记删除状态
            self.is_deleted = True
            self.deleted_at = timezone.now()
            self.save(update_fields=['is_deleted', 'deleted_at'])
    
    def restore(self):
        """恢复软删除的文件"""
        self.is_deleted = False
        self.deleted_at = None
        self.save(update_fields=['is_deleted', 'deleted_at'])
```

```python
# apps/document/views/file.py - FileMoveView

class FileMoveView(View):
    @auth('document.document.move')
    def post(self, request):
        import json
        try:
            data = json.loads(request.body)
            file_id = data.get('id')
            target_id = data.get('target_id')
            is_public = data.get('is_public', False)
        except:
            return json_response(error='参数错误')

        # 获取模型
        FileModel = get_file_model(is_public=is_public)
        FolderModel = get_folder_model(is_public=is_public)

        # 查询文件
        file_query = FileModel.objects.filter(pk=file_id)
        if not is_public:
            file_query = apply_tenant_filter(file_query, request.user, strict_mode=True)
        file = file_query.select_related('created_by').first()
        
        if not file:
            return json_response(error='文件不存在')

        # 公共空间权限校验
        if is_public and not check_public_space_permission(request.user, file, 'file', '移动'):
            return json_response(error='公共空间中只能移动自己创建的文件')

        # 查询目标文件夹
        if target_id:
            target_query = FolderModel.objects.filter(pk=target_id)
            if not is_public:
                target_query = apply_tenant_filter(target_query, request.user, strict_mode=True)
            target = target_query.first()
            if not target:
                return json_response(error='目标文件夹不存在')
        else:
            target = None

        # 如果已经在目标文件夹，无需操作
        current_folder_id = file.folder.id if file.folder else None
        if current_folder_id == target_id:
            return json_response()

        # 【审计优化】移动操作：只改 folder_id，不移动物理文件
        try:
            # 1. 更新文件夹关联
            file.folder = target
            
            # 2. 生成新的逻辑名（处理同名冲突）
            from ..libs.naming_utils import generate_unique_logical_name
            try:
                file.name = generate_unique_logical_name(
                    FileModel,
                    file.display_name or file.name,
                    target,
                    request.user
                )
            except Exception as e:
                logger.error(f'[Document] 生成唯一逻辑名失败：{e}')
                return json_response(error='文件移动失败：生成文件名失败')
            
            # 3. 【核心原则】physical_name 和 file_path 终身不变
            # 不修改 physical_name
            # 不修改 file_path
            # 不移动物理文件
            
            file.save(update_fields=['folder', 'name', 'updated_at'])
            
            log_operation(
                action="FILE_MOVE",
                user=request.user,
                resource_type="FILE",
                resource_id=file.id,
                is_public=is_public,
                target_folder_id=target_id
            )
            
            logger.info(f'[Document] 文件移动成功：id={file.id}, target_folder={target_id}')
            return json_response()
            
        except Exception as e:
            logger.error(f'[Document] 文件移动失败：{e}')
            return json_response(error=f'文件移动失败：{str(e)}')
```

### 4.2 删除操作修复（软删除）

```python
# apps/document/views/file.py - FileView.delete

class FileView(View):
    @auth('document.document.delete')
    def delete(self, request):
        form, error = JsonParser(
            Argument('id', type=int, help='参数错误'),
            Argument('is_public', type=bool, required=False, default=False),
            Argument('hard', type=bool, required=False, default=False)  # 【新增】支持硬删除
        ).parse(request.GET)
        
        if error is not None:
            return json_response(error=error)
            
        FileModel = get_file_model(is_public=form.is_public)
        
        file_query = FileModel.objects.filter(pk=form.id)
        if not form.is_public:
            file_query = apply_tenant_filter(file_query, request.user, strict_mode=True)
        file = file_query.select_related('created_by').first()
        
        if not file:
            return json_response(error='文件不存在')

        # 公共空间权限校验
        if form.is_public and not check_public_space_permission(request.user, file, 'file', '删除'):
            return json_response(error='公共空间中只能删除自己创建的文件')

        try:
            if form.hard:
                # 硬删除（管理员使用）
                self._hard_delete(file, form.is_public)
            else:
                # 软删除（默认）
                file.delete(hard=False)  # 调用模型的软删除方法
                logger.info(f'[Document] 文件已软删除：id={file.id}, physical_name={file.physical_name}')
            
            log_operation(
                action="FILE_DELETE",
                user=request.user,
                resource_type="FILE",
                resource_id=file.id,
                is_public=form.is_public,
                hard=form.hard
            )
            return json_response()
            
        except Exception as e:
            logger.error(f'[Document] 文件删除失败：{e}')
            return json_response(error=f'文件删除失败：{str(e)}')
    
    def _hard_delete(self, file_obj, is_public=False):
        """
        硬删除：删除数据库记录和物理文件
        注意：物理文件删除失败不影响数据库删除
        """
        file_path = file_obj.file_path
        physical_name = file_obj.physical_name or file_obj.name
        
        # 1. 先删除数据库记录
        file_obj.delete(hard=True)
        logger.info(f'[Document] 数据库记录已删除：physical_name={physical_name}')
        
        # 2. 再删除物理文件（失败不报错，由定时任务清理）
        if os.path.exists(file_path):
            try:
                os.remove(file_path)
                logger.info(f'[Document] 物理文件已删除：{file_path}')
            except Exception as e:
                logger.warning(f'[Document] 物理文件删除失败：{file_path}, error={e}')
```

```python
# apps/document/tasks/cleanup.py

import os
import logging
from datetime import timedelta
from celery import shared_task
from django.utils import timezone
from django.conf import settings

logger = logging.getLogger(__name__)


@shared_task
def clean_soft_deleted_files(retention_days=30, dry_run=False):
    """
    清理软删除文件的物理文件
    
    Args:
        retention_days: 保留天数（默认30天）
        dry_run: 仅模拟运行，不实际删除（用于测试）
    
    Returns:
        dict: 清理统计
    """
    from apps.document.models import DocumentFilePrivate, DocumentFilePublic
    
    cutoff_time = timezone.now() - timedelta(days=retention_days)
    
    logger.info(f'[Cleanup] 开始清理软删除文件，保留天数：{retention_days}，截止时间：{cutoff_time}')
    
    stats = {
        'private': {'checked': 0, 'deleted': 0, 'errors': 0},
        'public': {'checked': 0, 'deleted': 0, 'errors': 0}
    }
    
    # 清理私有文件
    private_files = DocumentFilePrivate.all_objects.filter(
        is_deleted=True,
        deleted_at__lte=cutoff_time
    )
    
    for file in private_files:
        stats['private']['checked'] += 1
        
        if dry_run:
            logger.info(f'[Cleanup][DryRun] 将清理：{file.file_path}')
            continue
        
        try:
            if os.path.exists(file.file_path):
                os.remove(file.file_path)
                logger.info(f'[Cleanup] 已清理：{file.file_path}')
            else:
                logger.warning(f'[Cleanup] 文件不存在：{file.file_path}')
            
            stats['private']['deleted'] += 1
            
        except Exception as e:
            stats['private']['errors'] += 1
            logger.error(f'[Cleanup] 清理失败：{file.file_path}, error={e}')
    
    # 清理公共文件
    public_files = DocumentFilePublic.all_objects.filter(
        is_deleted=True,
        deleted_at__lte=cutoff_time
    )
    
    for file in public_files:
        stats['public']['checked'] += 1
        
        if dry_run:
```

```python
#!/usr/bin/env python3
"""
check_file_mapping.py - 数据库 ↔ 物理文件映射校验（审计优化版）

特性：
- 支持分批处理（大数据量友好）
- 支持指定租户/文件夹校验
- 详细的错误报告

使用方式：
    python manage.py shell < check_file_mapping.py
    
    # 或带参数
    python manage.py shell -c "
import sys
sys.argv = ['check_file_mapping.py', '--batch-size', '500', '--tenant', 'tenant_001']
exec(open('check_file_mapping.py').read())
"
"""

import os
import sys
import argparse
from datetime import datetime

os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'spug.settings')
import django
django.setup()

from apps.document.models import DocumentFilePrivate, DocumentFilePublic


def check_file_batch(queryset, model_name, batch_size=1000):
    """
    分批检查文件记录
    
    Args:
        queryset: 文件查询集
        model_name: 模型名称（用于日志）
        batch_size: 每批处理数量
    """
    errors = []
    total = queryset.count()
    processed = 0
    
    print(f"  开始检查 {model_name}：共 {total} 条记录，批大小 {batch_size}")
    
    # 分批处理
    for offset in range(0, total, batch_size):
        batch = queryset[offset:offset+batch_size]
        
        for file_record in batch:
            processed += 1
            error = check_single_file(file_record, model_name)
            if error:
                errors.append(error)
        
        # 进度报告
        if (offset // batch_size + 1) % 10 == 0:
            print(f"    进度: {processed}/{total} ({processed*100//total}%)")
    
    print(f"  检查完成：{processed} 条记录，发现 {len(errors)} 个错误")
    return errors


def check_single_file(file_record, model_name):
    """检查单个文件记录"""
    record_id = file_record.id
    
    # 获取物理文件名（兼容新旧数据）
    physical_name = file_record.physical_name or file_record.name
    file_path = file_record.file_path
    
    # 1. 检查字段非空
    if not physical_name:
        return {
            'type': '空物理名',
            'model': model_name,
            'id': record_id,
            'detail': 'physical_name 和 name 均为空'
        }
    
    if not file_path:
        return {
            'type': '空路径',
            'model': model_name,
            'id': record_id,
            'detail': 'file_path 为空'
        }
    
    # 2. 检查路径是否包含物理文件名
    basename = os.path.basename(file_path)
    if basename != physical_name:
        return {
            'type': '路径不匹配',
            'model': model_name,
            'id': record_id,
            'detail': f'physical_name={physical_name}, path_basename={basename}'
        }
    
    # 3. 检查物理文件是否存在（仅未软删除的）
    if not getattr(file_record, 'is_deleted', False):
        if not os.path.exists(file_path):
            return {
                'type': '文件缺失',
                'model': model_name,
                'id': record_id,
                'detail': f'path={file_path}'
            }
    
    return None


def main():
    parser = argparse.ArgumentParser(description='资料库文件映射关系校验')
    parser.add_argument('--batch-size', type=int, default=1000, help='批处理大小')
    parser.add_argument('--tenant', type=str, help='指定租户ID校验')
    parser.add_argument('--model', type=str, choices=['private', 'public', 'all'], default='all', help='校验的模型')
    
    # Django shell 环境下可能无法解析参数，使用默认值
    try:
        args = parser.parse_args()
    except:
        args = parser.parse_args([])
    
    print("="*70)
    print("资料库文件映射关系校验（审计优化版）")
    print(f"开始时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"批处理大小: {args.batch_size}")
    if args.tenant:
        print(f"指定租户: {args.tenant}")
    print("="*70)
    
    all_errors = []
    
    # 校验私有文件
    if args.model in ['private', 'all']:
        print("\n[1/2] 校验私有文件...")
        queryset = DocumentFilePrivate.all_objects.filter(is_deleted=False)
        if args.tenant:
            queryset = queryset.filter(tenant_id=args.tenant)
        
        errors = check_file_batch(queryset, 'DocumentFilePrivate', args.batch_size)
        all_errors.extend(errors)
    
    # 校验公共文件
    if args.model in ['public', 'all']:
        print("\n[2/2] 校验公共文件...")
        queryset = DocumentFilePublic.all_objects.filter(is_deleted=False)
        
        errors = check_file_batch(queryset, 'DocumentFilePublic', args.batch_size)
        all_errors.extend(errors)
    
    # 输出结果
    print("\n" + "="*70)
    print("校验结果汇总")
    print("="*70)
    
    error_types = {}
    for error in all_errors:
        error_type = error['type']
        error_types[error_type] = error_types.get(error_type, 0) + 1
    
    print(f"\n错误类型统计:")
    for error_type, count in sorted(error_types.items()):
        print(f"  {error_type}: {count} 个")
    
    print(f"\n总错误数: {len(all_errors)}")
    
    if all_errors:
        print("\n详细错误（前20条）:")
        for i, error in enumerate(all_errors[:20], 1):
            print(f"  {i}. [{error['type']}] {error['model']} ID={error['id']}")
            print(f"      {error['detail']}")
        
        if len(all_errors) > 20:
            print(f"  ... 还有 {len(all_errors) - 20} 条错误")
        
        print("\n" + "="*70)
        print("❌ 校验失败，请修复上述问题")
        print("="*70)
        sys.exit(1)
    else:
        print("\n" + "="*70)
        print("✅ 校验通过，映射关系正常")
        print("="*70)
        sys.exit(0)


if __name__ == '__main__':
    main()
```
