### 4.1 断点续传修复 (P0-1)

#### 4.1.1 新增分片检查方法

**文件**: `stores/upload/core/transfer.js`

```javascript
/**
 * 检查已上传分片（断点续传）
 * @param {string} fileHash - 文件MD5哈希
 * @param {number} fileSize - 文件大小
 * @param {number} totalChunks - 总分片数
 * @param {boolean} isPublic - 是否公共空间
 */
async checkUploadedChunks(fileHash, fileSize, totalChunks, isPublic = false) {
  try {
    const { http } = await import('libs');
    const result = await http.post(API_ENDPOINTS.CHECK_UPLOADED_CHUNKS, {
      file_hash: fileHash,
      file_size: fileSize,
      total_chunks: totalChunks,
      is_public: isPublic
    });
    return result;
  } catch (error) {
    console.error('[TransferStore] 检查已上传分片失败:', error);
    return { exists: false, uploaded_chunks: [], count: 0 };
  }
}
```

#### 4.1.2 修改 resumeItem 方法

**文件**: `stores/upload/core/index.js`

```javascript
@action
async resumeItem(itemId) {
  const item = this.queueStore.findUploadItemInCurrentTenant(itemId);
  if (!item) return;

  if (!['paused', 'error', 'waiting'].includes(item.status)) {
    return;
  }

  // 检查后端状态
  if (item.transferId) {
    try {
      const transfers = await this.transferStore.fetchTransfers(
        this.rootStore.navigationStore?.isPublic
      );
      const currentTransfer = transfers.find(t => t.id === item.transferId);
      
      if (currentTransfer) {
        if (currentTransfer.status === 'COMPLETED') {
          item.status = 'completed';
          item.percent = 100;
          item.error = null;
          return;
        }
        
        if (currentTransfer.status === 'CANCELED' || currentTransfer.status === 'FAILED') {
          item.status = 'error';
          item.error = currentTransfer.status === 'CANCELED' ? '已取消' : '失败';
          return;
        }
        
        // 同步后端的 file_hash
        if (currentTransfer.file_hash && !item.fileHash) {
          item.fileHash = currentTransfer.file_hash;
        }
      }
    } catch (error) {
      console.error('[resumeItem] 获取传输状态失败:', error);
    }
  }

  // 检查是否有文件对象
  if (!item.file || !(item.file instanceof File)) {
    message.warning(`"${item.name}" 需要重新选择文件才能继续上传`);
    item.error = '请重新选择文件';
    item.status = 'error';
    this.triggerFileReSelection(itemId);
    return;
  }

  // 【核心修复】大文件断点续传检查
  if (item.fileSize > UPLOAD_CONSTANTS.NORMAL_UPLOAD_THRESHOLD && item.fileHash) {
    try {
      message.loading('正在检查已上传分片...', 0);
      
      const chunkCount = Math.ceil(item.fileSize / UPLOAD_CONSTANTS.CHUNK_SIZE);
      const checkResult = await this.transferStore.checkUploadedChunks(
        item.fileHash,
        item.fileSize,
        chunkCount,
        item.isPublic
      );
      
      message.destroy();
      
      if (checkResult.exists && checkResult.uploaded_chunks.length > 0) {
        const uploadedCount = checkResult.uploaded_chunks.length;
        const uploadedBytes = uploadedCount * UPLOAD_CONSTANTS.CHUNK_SIZE;
        const percent = Math.min(Math.round((uploadedBytes / item.fileSize) * 100), 99);
        
        item.currentChunk = uploadedCount;
        item.percent = percent;
        item.chunks = checkResult.uploaded_chunks.map(idx => ({ index: idx, uploaded: true }));
        
        message.success(`已找到 ${uploadedCount} 个已上传分片，将从 ${percent}% 处继续`);
      }
    } catch (error) {
      message.destroy();
      console.error('[resumeItem] 检查已上传分片失败:', error);
    }
  }

  // 恢复上传
  item.status = 'uploading';
  item.error = null;
  item.canAbort = true;
  item.isPausedByUser = false;
  item.isCancelledByUser = false;
  item.abortController = new AbortController();

  const resumeFolderId = item.folderId !== null ? item.folderId : this.rootStore.navigationStore?.currentFolderId;
  await this.resumeChunkedUpload(item.file, resumeFolderId, item);
}
```

### 4.2 批量删除优化 (P0-2)

**修改 removeAll 方法：**

```javascript
// stores/upload/core/index.js:442-481
removeAll = async () => {
  const activeStatuses = ['uploading', 'calculating', 'merging'];
  const transferIds = [];

  const uploadQueue = this.queueStore.uploadQueue;

  Object.keys(uploadQueue).forEach(tenantId => {
    const queue = uploadQueue[tenantId];
    if (!queue || !Array.isArray(queue)) return;

    const itemsToRemove = queue.filter(item => !activeStatuses.includes(item.status));

    itemsToRemove.forEach(item => {
      if (item.uniqueKey && item.file) {
        const { file, folderId } = item;
        const isPublic = item.isPublic !== undefined ? item.isPublic : this.rootStore.navigationStore?.isPublic;
        this.queueStore.removeUniqueKey(file, folderId, isPublic);
      }
      if (item.transferId) {
        transferIds.push(item.transferId);
      }
    });

    uploadQueue[tenantId] = queue.filter(item => activeStatuses.includes(item.status));
  });

  // 【优化】使用批量删除API
  if (transferIds.length > 0) {
    try {
      const result = await this.transferStore.batchDeleteTransfers(transferIds);
      if (result) {
        message.success(`已清理 ${transferIds.length} 条传输记录`);
      } else {
        message.warning('批量删除部分失败');
      }
    } catch (error) {
      message.error('批量删除失败，请稍后重试');
    }
  }
}
```

#### 4.4.1 后端修改

**文件**: `spug_api/apps/document/views/transfer/list.py`

```python
class TransferListView(View):
    """获取用户的传输记录列表（支持分页）"""

    @auth('document.document.view')
    def get(self, request):
        from ...models import DocumentTransfer
        from ...constants import TransferStatus
        
        form, error = JsonParser(
            Argument('status', type=str, required=False, help='传输状态筛选'),
            Argument('transfer_type', type=str, required=False, help='传输类型筛选'),
            Argument('is_public', type=bool, required=False, help='是否公共空间'),
            Argument('page', type=int, required=False, default=1, help='页码'),
            Argument('page_size', type=int, required=False, default=20, help='每页数量'),
        ).parse(request.GET)

        if error:
            return json_response(error=error)

        request_user = request.user
        page = form.page
        page_size = min(form.page_size, 100)  # 最大100条
        offset = (page - 1) * page_size

        # 构建查询
        if hasattr(form, 'is_public') and form.is_public is not None:
            if form.is_public:
                queryset = DocumentTransfer.objects.filter(
                    user=request_user,
                    is_public=True
                )
            else:
                queryset = DocumentTransfer.objects.filter(
                    user=request_user,
                    is_public=False
                )
                if hasattr(request_user, 'tenant_id') and request_user.tenant_id:
                    queryset = queryset.filter(tenant_id=request_user.tenant_id)
        else:
            queryset = DocumentTransfer.objects.filter(user=request_user)
            if hasattr(request_user, 'tenant_id') and request_user.tenant_id:
                queryset = queryset.filter(tenant_id=request_user.tenant_id)

        # 筛选条件
        if form.status:
            queryset = queryset.filter(status=form.status)
        if form.transfer_type:
            queryset = queryset.filter(transfer_type=form.transfer_type)

        # 获取总数
        total = queryset.count()
        
        # 分页查询
        queryset = queryset.order_by('-created_at')[offset:offset+page_size]

        # 构建响应
        transfers = []
        for t in queryset:
            transfers.append({
                'id': t.id,
                'transfer_type': t.transfer_type,
                'status': t.status,
                'file_name': t.file_name,
                'file_size': t.file_size,
                'file_hash': t.file_hash or '',
                'progress': t.progress,
                'transferred_size': t.transferred_size,
                'speed': t.speed,
                'total_chunks': t.total_chunks,
                'uploaded_chunks': t.uploaded_chunks,
                'folder_id': t.folder_id,
                'is_public': t.is_public,
                'created_at': t.created_at.strftime('%Y-%m-%d %H:%M:%S') if t.created_at else None,
                'started_at': t.started_at.strftime('%Y-%m-%d %H:%M:%S') if t.started_at else None,
                'completed_at': t.completed_at.strftime('%Y-%m-%d %H:%M:%S') if t.completed_at else None,
                'error_message': t.error_message,
            })

        return json_response(data={
            'list': transfers,
            'total': total,
            'page': page,
            'page_size': page_size
        })
```

```python
from celery.result import AsyncResult

class TransferBatchTaskStatusView(View):
    """查询批量任务状态"""

    @auth('document.document.view')
    def get(self, request, task_id):
        try:
            result = AsyncResult(task_id)
            
            if result.ready():
                task_result = result.result
                return json_response(data={
                    'status': 'completed',
                    'result': task_result,
                    'task_id': task_id
                })
            else:
                return json_response(data={
                    'status': 'pending',
                    'task_id': task_id
                })
        except Exception as e:
            return json_response(error=f'查询任务状态失败: {str(e)}')
```

```javascript
// 轮询直到完成
async pollBatchTaskStatus(taskId, onComplete) {
  const check = async () => {
    const result = await this.getBatchTaskStatus(taskId);
    if (result && result.status === 'completed') {
      onComplete && onComplete(result);
      return;
    }
    setTimeout(check, 1000);
  };
```
