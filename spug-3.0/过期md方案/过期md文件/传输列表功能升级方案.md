### 3.2 问题详细说明

#### 1) 资源清理检查点不完整

**现状**: 部分清理，缺少系统性策略

**建议补充**:
```javascript
// pauseItem 完成后
item.abortController = null;
item.abortToken = null;

// completed 状态后
if (item.status === 'completed') {
  item.file = null;  // 释放大文件引用
}
```

#### 2) 状态检查方法不统一

**现状**: 多处重复代码

```javascript
// 当前：多处重复
if (item.status === 'paused' || item.isPausedByUser) { ... }

// 建议：统一方法
isPaused(uploadId) {
  const item = this.findUploadItem(uploadId);
  return item && (item.status === 'paused' || item.isPausedByUser);
}
```

#### 3) MD5计算无法中断

**现状**: 大文件(>1GB)计算时用户体验不佳

**推荐方案**: Promise.race + 取消监听

#### 4) 全局/单任务状态冲突

**现状**: `rootStore.isPaused` 和 `item.isPausedByUser` 并存

**建议**: 逐步移除全局状态，完全使用单任务状态

## 四、推荐实施方案

### 方案A：保守优化（推荐）

**适用场景**: 当前功能稳定，希望低风险渐进改进

**实施步骤**:
1. **本周**: 补充资源清理检查点
2. **下周**: 统一状态检查方法
3. **下月**: 实施MD5 Promise.race方案
4. **持续**: 补充单元测试

**优点**:
- 风险低，逐步验证
- 每步都有明确收益
- 不影响现有功能

**缺点**:
- 时间较长
- 无法一次性解决架构问题

### 方案B：状态机重构

**适用场景**: 计划进行较大功能迭代，顺便重构

**实施步骤**:
1. 引入轻量级状态机
2. 迁移现有逻辑到状态机
3. 利用状态机能力实现其他优化

**优点**:
- 架构清晰，长期收益大
- 状态管理规范化

**缺点**:
- 工作量大（2-3天）
- 需要充分测试
- 有一定风险

### 方案C：混合方案（平衡推荐）

**实施步骤**:
1. **阶段1**（本周）: 资源清理 + 统一状态检查
2. **阶段2**（2周后）: 引入轻量级状态机
3. **阶段3**（状态机稳定后）: MD5可中断化
4. **阶段4**（持续）: 完善测试

### 阶段2：状态机引入（2周后）

#### 5.2.1 轻量级状态机设计

**核心思想**: 不引入外部库，使用简单的状态表实现

**文件**: 新建 `uploadStateMachine.js`

**实现**:
```javascript
/**
 * 上传任务状态机
 * 轻量级实现，不依赖外部库
 */
export class UploadStateMachine {
  constructor() {
    // 状态定义
    this.states = {
      waiting: {
        on: { START: 'calculating' },
        onEntry: this.onWaitingEntry,
        onExit: this.onWaitingExit
      },
      calculating: {
        on: { 
          COMPLETE: 'uploading',
          PAUSE: 'paused',
          ERROR: 'error'
        },
        onEntry: this.onCalculatingEntry,
        onExit: this.onCalculatingExit
      },
      uploading: {
        on: {
          COMPLETE: 'merging',
          PAUSE: 'paused',
          ERROR: 'error'
        },
        onEntry: this.onUploadingEntry,
        onExit: this.onUploadingExit
      },
      paused: {
        on: {
          RESUME: 'uploading',
          CANCEL: 'error'
        },
        onEntry: this.onPausedEntry,
        onExit: this.onPausedExit
      },
      merging: {
        on: {
          SUCCESS: 'completed',
          ERROR: 'error'
        },
        onEntry: this.onMergingEntry,
        onExit: this.onMergingExit
      },
      completed: {
        type: 'final',
        onEntry: this.onCompletedEntry
      },
      error: {
        type: 'final',
        onEntry: this.onErrorEntry
      }
    };
    
    this.currentState = 'waiting';
    this.context = {};
    this.listeners = [];
  }
  
  /**
   * 状态转换
   * @param {string} event - 事件名称
   * @param {object} payload - 事件数据
   * @returns {boolean} 转换是否成功
   */
  transition(event, payload = {}) {
    const currentStateDef = this.states[this.currentState];
    const nextState = currentStateDef.on?.[event];
    
    if (!nextState) {
      console.warn(`[StateMachine] 无效转换: ${this.currentState} -${event}-> ?`);
      return false;
    }
    
    const fromState = this.currentState;
    
    // 执行 exit 钩子
    currentStateDef.onExit?.(this.context);
    
    // 更新状态
    this.currentState = nextState;
    this.context = { ...this.context, ...payload };
    
    // 执行 entry 钩子
    this.states[nextState].onEntry?.(this.context);
    
    // 通知监听器
    this.notifyListeners(fromState, nextState, event);
    
    return true;
  }
  
  /**
   * 检查是否可以执行转换
   * @param {string} event - 事件名称
   * @returns {boolean} 是否可以转换
   */
  canTransition(event) {
    const currentStateDef = this.states[this.currentState];
    return !!currentStateDef.on?.[event];
  }
  
  /**
   * 获取当前状态
   * @returns {string} 当前状态
   */
  getState() {
    return this.currentState;
  }
  
  /**
   * 检查是否处于某状态
   * @param {string} state - 状态名称
   * @returns {boolean} 是否处于该状态
   */
  isInState(state) {
    return this.currentState === state;
  }
  
  /**
   * 添加状态变更监听器
   * @param {function} listener - 监听器函数 (fromState, toState, event) => void
   */
  addListener(listener) {
    this.listeners.push(listener);
  }
  
  notifyListeners(fromState, toState, event) {
    this.listeners.forEach(listener => {
      try {
        listener(fromState, toState, event);
      } catch (error) {
        console.error('[StateMachine] 监听器错误:', error);
      }
    });
  }
  
  // ========== 状态钩子 ==========
  
  onWaitingEntry(ctx) {
    ctx.status = 'waiting';
    ctx.error = null;
  }
  
  onCalculatingEntry(ctx) {
    ctx.status = 'calculating';
    ctx.startTime = Date.now();
  }
  
  onCalculatingExit(ctx) {
    // 离开 calculating 时中断MD5
    ctx.md5Store?.cancelMD5?.(ctx.uploadId);
  }
  
  onUploadingEntry(ctx) {
    ctx.status = 'uploading';
    ctx.canAbort = true;
  }
  
  onUploadingExit(ctx) {
    // 离开 uploading 时清理资源
    ctx.abortController?.abort();
    ctx.abortController = null;
    ctx.abortToken = null;
    ctx.canAbort = false;
  }
  
  onPausedEntry(ctx) {
    ctx.status = 'paused';
    ctx.error = '已暂停';
    ctx.canAbort = false;
  }
  
  onPausedExit(ctx) {
    ctx.error = null;
  }
  
  onMergingEntry(ctx) {
    ctx.status = 'merging';
  }
  
  onMergingExit(ctx) {
    // 合并完成清理
  }
  
  onCompletedEntry(ctx) {
    ctx.status = 'completed';
    ctx.progress = 100;
    ctx.completedAt = Date.now();
    // 清理资源
    ctx.file = null;
    ctx.abortController = null;
    ctx.abortToken = null;
  }
  
  onErrorEntry(ctx) {
    ctx.status = 'error';
    ctx.canAbort = false;
    // 清理资源
    ctx.abortController = null;
    ctx.abortToken = null;
  }
}
```

#### 5.2.2 集成到现有代码

**文件**: `index.js`

**修改**:

```javascript
import { UploadStateMachine } from './uploadStateMachine';

class UploadStore {
  async uploadFileToFolder(file, folderId, isPublic) {
    // 创建状态机实例
    const stateMachine = new UploadStateMachine();
    
    // 添加上下文
    stateMachine.context = {
      uploadId,
      file,
      folderId,
      isPublic,
      md5Store: this.rootStore.md5Store,
      queueStore: this.queueStore
    };
    
    // 添加状态变更日志
    stateMachine.addListener((from, to, event) => {
      console.log(`[Upload] ${file.name}: ${from} → ${to} (${event})`);
    });
    
    try {
      // 使用状态机驱动流程
      stateMachine.transition('START');
      
      // MD5计算
      const fileHash = await this.rootStore.md5Store.calculateFileMD5(file, uploadId);
      
      // 检查是否被暂停
      if (stateMachine.isInState('paused')) {
        return;  // 正常返回，不抛出错误
      }
      
      stateMachine.transition('COMPLETE', { fileHash });
      
      // 上传逻辑...
      
    } catch (error) {
      // 检查是否是暂停错误
      if (stateMachine.isInState('paused')) {
        return;  // 正常返回
      }
      
      stateMachine.transition('ERROR', { error });
      throw error;  // 真正的错误才抛出
    }
  }
}
```

### 阶段3：MD5可中断化（状态机稳定后）

#### 5.3.1 Promise.race 方案

**文件**: `md5.js`

**修改**:
```javascript
class MD5Store {
  _cancelListeners = new Map();

  calculateFileMD5(file, uploadId) {
    // 创建可取消的 Promise
    const cancelPromise = new Promise((_, reject) => {
      this._cancelListeners.set(uploadId, reject);
    });
    
    const md5Promise = this.calculateFileMD5Internal(file, uploadId);
    
    return Promise.race([md5Promise, cancelPromise]).finally(() => {
      this._cancelListeners.delete(uploadId);
    });
  }
  
  // 外部调用取消
  cancelMD5(uploadId) {
    const reject = this._cancelListeners.get(uploadId);
    if (reject) {
      reject(new Error('用户暂停'));
      this._cancelListeners.delete(uploadId);
    }
  }
  
  // 原有的计算逻辑
  async calculateFileMD5Internal(file, uploadId) {
    // ... Web Worker 计算逻辑
  }
}
```
