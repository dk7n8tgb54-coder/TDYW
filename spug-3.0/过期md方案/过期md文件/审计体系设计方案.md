# 审计体系设计方案

## 1. 概述

### 1.1 设计目标

构建一个高性能、高可靠、可扩展的分布式审计系统，满足以下核心需求：

| 维度 | 目标 | 指标 |
|------|------|------|
| 完整性 | 100% 记录关键操作 | 零丢失保障 |
| 实时性 | 秒级审计查询 | < 3s 响应时间 |
| 合规性 | 满足等保/ISO27001要求 | 6年数据保留 |
| 性能 | 不影响主业务流程 | 延迟增加 < 5% |

### 1.2 审计范围

### 2.2 核心组件

| 组件 | 技术选型 | 职责 |
|------|----------|------|
| 审计SDK | Python Decorator / Middleware | 业务系统无侵入接入 |

### 5.1 Python装饰器实现

```python
# audit/decorators.py
import functools
import uuid
import hashlib
import json
from datetime import datetime, timedelta
from typing import Optional, Callable, List
from dataclasses import dataclass, asdict

from django.http import HttpRequest
from django.conf import settings

from .client import AuditClient
from .risk import RiskEvaluator


@dataclass
class AuditContext:
    """审计上下文"""
    event_type: str
    resource_type: str
    resource_id_extractor: Optional[Callable] = None
    resource_name_extractor: Optional[Callable] = None
    capture_request_body: bool = False
    capture_response_body: bool = False
    sensitive_fields: Optional[List[str]] = None
    risk_check: bool = True
    

def audit_log(context: AuditContext):
    """
    审计日志装饰器
    
    Usage:
        @audit_log(AuditContext(
            event_type='DATA_MODIFY',
            resource_type='Document',
            resource_id_extractor=lambda req, res: res.data.get('id'),
            sensitive_fields=['password', 'token']
        ))
        def update_document(request):
            ...
    """
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            # 获取请求对象
            request = _get_request_from_args(args, kwargs)
            
            # 生成请求ID
            request_id = str(uuid.uuid4())
            request.audit_request_id = request_id
            
            # 记录开始时间
            start_time = datetime.now()
            
            # 执行原函数
            try:
                response = func(*args, **kwargs)
                success = True
                error_code = None
                error_message = None
            except Exception as e:
                success = False
                error_code = type(e).__name__
                error_message = str(e)
                response = None
                raise
            finally:
                # 计算执行时间
                duration_ms = int((datetime.now() - start_time).total_seconds() * 1000)
                
                # 构建审计事件
                event = _build_audit_event(
                    request=request,
                    response=response,
                    context=context,
                    request_id=request_id,
                    start_time=start_time,
                    duration_ms=duration_ms,
                    success=success,
                    error_code=error_code,
                    error_message=error_message
                )
                
                # 异步发送审计事件
                AuditClient.send_async(event)
                
            return response
        return wrapper
    return decorator


def audit_model_change(
    event_type: str,
    resource_type: str,
    id_field: str = 'id',
    name_field: str = 'name',
    track_fields: Optional[List[str]] = None
):
    """
    Model变更审计装饰器
    
    Usage:
        @audit_model_change(
            event_type='DATA_MODIFY',
            resource_type='Schedule',
            track_fields=['staff_id', 'shift_id', 'schedule_date']
        )
        def update_schedule(request, schedule_id):
            schedule = Schedule.objects.get(id=schedule_id)
            # ... 修改逻辑
            schedule.save()
            return schedule
    """
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            request = _get_request_from_args(args, kwargs)
            
            # 获取变更前的数据（如果有ID）
            resource_id = kwargs.get(id_field) or args[0] if args else None
            before_data = None
            
            if resource_id and hasattr(args[0], 'objects'):
                try:
                    instance = args[0].objects.get(pk=resource_id)
                    before_data = _extract_model_data(instance, track_fields)
                except args[0].DoesNotExist:
                    pass
            
            start_time = datetime.now()
            
            try:
                result = func(*args, **kwargs)
                success = True
                error = None
            except Exception as e:
                success = False
                error = str(e)
                raise
            finally:
                duration_ms = int((datetime.now() - start_time).total_seconds() * 1000)
                
                # 获取变更后的数据
                after_data = None
                if success and result:
                    after_data = _extract_model_data(result, track_fields)
                
                # 计算变更字段
                changed_fields = _compute_changed_fields(before_data, after_data)
                
                # 构建审计事件
                event = {
                    'event_id': f"aud_{datetime.now().strftime('%Y%m%d%H%M%S')}_{uuid.uuid4().hex[:8]}",
                    'event_time': datetime.now().isoformat(),
                    'event_type': event_type,
                    'actor': _extract_actor_info(request),
                    'resource': {
                        'resource_type': resource_type,
                        'resource_id': str(resource_id) if resource_id else None,
                        'resource_name': getattr(result, name_field, None) if result else None,
                    },
                    'action': {
                        'success': success,
                        'duration_ms': duration_ms,
                        'error_message': error
                    },
                    'context': {
                        'before_value': before_data,
                        'after_value': after_data,
                        'changed_fields': changed_fields
                    }
                }
                
                AuditClient.send_async(event)
            
            return result
        return wrapper
    return decorator


# ============ 内部辅助函数 ============

def _get_request_from_args(args, kwargs) -> HttpRequest:
    """从参数中提取request对象"""
    for arg in args:
        if isinstance(arg, HttpRequest):
            return arg
    return kwargs.get('request')


def _build_audit_event(request, response, context, **kwargs) -> dict:
    """构建审计事件"""
    event = {
        'event_id': f"aud_{datetime.now().strftime('%Y%m%d%H%M%S')}_{uuid.uuid4().hex[:8]}",
        'event_time': kwargs['start_time'].isoformat(),
        'event_type': context.event_type,
        'actor': _extract_actor_info(request),
        'resource': {
            'module': request.resolver_match.app_name if hasattr(request, 'resolver_match') else None,
            'resource_type': context.resource_type,
            'resource_id': context.resource_id_extractor(request, response) if context.resource_id_extractor else None,
            'resource_name': context.resource_name_extractor(request, response) if context.resource_name_extractor else None,
        },
        'action': {
            'success': kwargs['success'],
            'duration_ms': kwargs['duration_ms'],
            'request_id': kwargs['request_id'],
            'error_code': kwargs.get('error_code'),
            'error_message': kwargs.get('error_message')
        }
    }
    
    # 风险评估
    if context.risk_check:
        risk_result = RiskEvaluator.evaluate(event)
        event['risk'] = risk_result
    
    return event


def _extract_actor_info(request: HttpRequest) -> dict:
    """提取操作人信息"""
    user = getattr(request, 'user', None)
    return {
        'user_id': str(user.id) if user and hasattr(user, 'id') else 'anonymous',
        'username': getattr(user, 'username', 'anonymous'),
        'real_name': getattr(user, 'real_name', ''),
        'role': getattr(user, 'role', ''),
        'ip_address': _get_client_ip(request),
        'user_agent': request.META.get('HTTP_USER_AGENT', ''),
        'session_id': request.session.session_key if hasattr(request, 'session') else None
    }


def _get_client_ip(request: HttpRequest) -> str:
    """获取客户端真实IP"""
    x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
    if x_forwarded_for:
        return x_forwarded_for.split(',')[0].strip()
    return request.META.get('REMOTE_ADDR', '')


def _extract_model_data(instance, fields: Optional[List[str]] = None) -> dict:
    """提取模型数据"""
    if not instance:
        return None
    if fields:
        return {f: getattr(instance, f, None) for f in fields}
    return {f.name: getattr(instance, f.name) for f in instance._meta.fields}


def _compute_changed_fields(before: dict, after: dict) -> List[str]:
    """计算变更字段"""
    if not before or not after:
        return []
    return [k for k in before.keys() if k in after and before[k] != after[k]]

# audit/middleware.py
import time
import uuid
from typing import Optional

from django.http import HttpRequest, HttpResponse
from django.utils.deprecation import MiddlewareMixin

from .client import AuditClient
from .filters import SensitiveDataFilter


class AuditMiddleware(MiddlewareMixin):
    """
    全局审计中间件
    自动记录所有API请求
    """
    
    # 排除的URL路径
    EXCLUDED_PATHS = [
        '/health/',
        '/metrics/',
        '/static/',
        '/media/',
        '/admin/jsi18n/',
    ]
    
    # 排除的Content-Type
    EXCLUDED_CONTENT_TYPES = [
        'application/octet-stream',
        'multipart/form-data',
    ]
    
    def process_request(self, request: HttpRequest):
        """请求处理前"""
        # 检查是否需要审计
        if self._should_skip(request):
            return None
        
        # 记录请求开始时间
        request.audit_start_time = time.time()
        request.audit_request_id = str(uuid.uuid4())
        
        # 保存请求体（用于后续审计）
        if request.method in ['POST', 'PUT', 'PATCH']:
            request.audit_body = self._capture_request_body(request)
        
        return None
    
    def process_response(self, request: HttpRequest, response: HttpResponse) -> HttpResponse:
        """响应处理后"""
        if self._should_skip(request) or not hasattr(request, 'audit_start_time'):
            return response
        
        # 计算执行时间
        duration_ms = int((time.time() - request.audit_start_time) * 1000)
        
        # 构建审计事件
        event = self._build_event(request, response, duration_ms)
        
        # 异步发送（不阻塞响应）
        AuditClient.send_async(event)
        
        # 添加审计ID到响应头
        response['X-Audit-ID'] = request.audit_request_id
        
        return response
    
    def _should_skip(self, request: HttpRequest) -> bool:
        """判断是否需要跳过审计"""
        path = request.path_info
        
        # 检查排除路径
        for excluded in self.EXCLUDED_PATHS:
            if path.startswith(excluded):
                return True
        
        # 只审计API请求
        if not path.startswith('/api/'):
            return True
        
        return False
    
    def _capture_request_body(self, request: HttpRequest) -> Optional[dict]:
        """捕获请求体"""
        content_type = request.content_type or ''
        
        if 'application/json' in content_type:
            try:
                body = request.body.decode('utf-8')
                import json
                data = json.loads(body)
                return SensitiveDataFilter.filter(data)
            except:
                return None
        
        return None
    
    def _build_event(self, request: HttpRequest, response: HttpResponse, duration_ms: int) -> dict:
        """构建审计事件"""
        user = getattr(request, 'user', None)
        
        event = {
            'event_id': f"aud_{request.audit_request_id}",
            'event_time': time.strftime('%Y-%m-%dT%H:%M:%S'),
            'event_type': self._determine_event_type(request),
            'actor': {
                'user_id': str(user.id) if user and user.is_authenticated else 'anonymous',
                'username': getattr(user, 'username', 'anonymous'),
                'ip_address': self._get_client_ip(request),
                'user_agent': request.META.get('HTTP_USER_AGENT', '')[:512],
            },
            'resource': {
                'module': self._extract_module(request),
                'resource_type': 'API_ENDPOINT',
                'resource_id': request.path_info,
                'operation': request.method,
            },
            'action': {
                'success': response.status_code < 400,
                'status_code': response.status_code,
                'duration_ms': duration_ms,
                'request_id': request.audit_request_id,
            },
            'context': {
                'query_params': dict(request.GET),
                'request_body': getattr(request, 'audit_body', None),
            }
        }
        
        return event
    
    def _determine_event_type(self, request: HttpRequest) -> str:
        """确定事件类型"""
        method = request.method
        if method == 'GET':
            return 'DATA_QUERY'
        elif method == 'POST':
            return 'DATA_CREATE'
        elif method in ['PUT', 'PATCH']:
            return 'DATA_MODIFY'
        elif method == 'DELETE':
            return 'DATA_DELETE'
        return 'SYSTEM'
    
    def _extract_module(self, request: HttpRequest) -> str:
        """提取模块名称"""
        path = request.path_info
        parts = path.strip('/').split('/')
        if len(parts) >= 2:
            return parts[1]  # /api/document/... -> document
        return 'unknown'
    
    def _get_client_ip(self, request: HttpRequest) -> str:
        """获取客户端IP"""
        x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
        if x_forwarded_for:
            return x_forwarded_for.split(',')[0].strip()
        return request.META.get('REMOTE_ADDR', '')

from typing import Dict, List
from dataclasses import dataclass
from datetime import datetime, time


@dataclass
class RiskFactor:
    name: str
    weight: int
    condition: callable


class RiskEvaluator:
    """风险评估引擎"""
    
    # 风险因子配置
    RISK_FACTORS = [
        # 敏感操作
        RiskFactor('批量删除', 30, lambda e: _is_batch_delete(e)),
        RiskFactor('权限提升', 40, lambda e: _is_permission_escalation(e)),
        RiskFactor('数据导出', 20, lambda e: _is_data_export(e)),
        
        # 异常行为
        RiskFactor('非工作时间操作', 15, lambda e: _is_off_hours(e)),
        RiskFactor('异地登录', 25, lambda e: _is_abnormal_location(e)),
        RiskFactor('高频操作', 20, lambda e: _is_high_frequency(e)),
        
        # 数据敏感
        RiskFactor('机密数据访问', 35, lambda e: _is_confidential_access(e)),
        RiskFactor('跨租户访问', 50, lambda e: _is_cross_tenant(e)),
        
        # 失败异常
        RiskFactor('多次认证失败', 30, lambda e: _has_auth_failures(e)),
        RiskFactor('越权尝试', 45, lambda e: _is_unauthorized_attempt(e)),
    ]
    
    @classmethod
    def evaluate(cls, event: dict) -> dict:
        """评估事件风险"""
        score = 0
        factors = []
        
        for factor in cls.RISK_FACTORS:
            if factor.condition(event):
                score += factor.weight
                factors.append(factor.name)
        
        # 确定风险等级
        if score >= 80:
            level = 'CRITICAL'
        elif score >= 50:
            level = 'HIGH'
        elif score >= 25:
            level = 'MEDIUM'
        else:
            level = 'LOW'
        
        return {
            'risk_score': min(score, 100),
            'risk_level': level,
            'risk_factors': factors
        }

    def calculate_hash(cls, event: dict) -> str:
        """
        计算审计事件哈希值
        使用HMAC-SHA256确保防篡改
        """
        # 提取不可变字段
        data = {
            'event_id': event.get('event_id'),
            'event_time': event.get('event_time'),
            'event_type': event.get('event_type'),
            'user_id': event.get('actor', {}).get('user_id'),
            'resource_id': event.get('resource', {}).get('resource_id'),
            'operation': event.get('resource', {}).get('operation'),
            'context_before': event.get('context', {}).get('before_value'),
            'context_after': event.get('context', {}).get('after_value'),
        }
        
        # 序列化为规范字符串
        canonical_string = cls._canonicalize(data)
        
        # 计算HMAC
        signature = hmac.new(
            cls.SECRET_KEY,
            canonical_string.encode(),
            hashlib.sha256
        ).hexdigest()
        
        return signature
    
    @classmethod
    def verify_hash(cls, event: dict) -> bool:
        """验证审计事件完整性"""
        stored_hash = event.get('tamper_hash')
        if not stored_hash:
            return False
        
        calculated_hash = cls.calculate_hash(event)
        return hmac.compare_digest(stored_hash, calculated_hash)
    
    @classmethod
    def _canonicalize(cls, data: dict) -> str:
        """将数据规范化为字符串"""
        import json
        # 按键排序，确保一致性
        return json.dumps(data, sort_keys=True, separators=(',', ':'))


class MerkleTree:
    """默克尔树 - 批量完整性校验"""
    
    def __init__(self, leaves: list):
        self.leaves = [self._hash(leaf) for leaf in leaves]
        self.tree = self._build_tree()
    
    def _hash(self, data: str) -> str:
        return hashlib.sha256(data.encode()).hexdigest()
    
    def _build_tree(self) -> list:
        """构建默克尔树"""
        tree = [self.leaves]
        level = self.leaves
        
        while len(level) > 1:
            next_level = []
            for i in range(0, len(level), 2):
                left = level[i]
                right = level[i + 1] if i + 1 < len(level) else left
                next_level.append(self._hash(left + right))
            tree.append(next_level)
            level = next_level
        
        return tree
    
    def get_root(self) -> str:
        """获取根哈希"""
        return self.tree[-1][0] if self.tree else ''
    
    def get_proof(self, index: int) -> list:
        """获取验证路径"""
        proof = []
        for level in self.tree[:-1]:
            sibling_index = index ^ 1
            if sibling_index < len(level):
                proof.append(level[sibling_index])
            index //= 2
        return proof
```

# audit/filters.py
import re
from typing import Any, Dict, List, Union


class SensitiveDataFilter:
    """敏感数据脱敏过滤器"""
    
    # 敏感字段模式
    SENSITIVE_PATTERNS = {
        'password': r'(?i)(password|passwd|pwd)',
        'token': r'(?i)(token|secret|key|api_key)',
        'credit_card': r'\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b',
        'phone': r'\b1[3-9]\d{9}\b',
        'id_card': r'\b\d{17}[\dXx]\b',
        'email': r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
    }
    
    # 脱敏替换规则
    MASK_RULES = {
        'password': '******',
        'token': lambda x: x[:4] + '****' + x[-4:] if len(x) > 8 else '****',
        'credit_card': lambda x: x[:4] + ' **** **** ' + x[-4:],
        'phone': lambda x: x[:3] + '****' + x[-4:],
        'id_card': lambda x: x[:6] + '********' + x[-4:],
        'email': lambda x: x.split('@')[0][:2] + '***@' + x.split('@')[1],
    }
    
    @classmethod
    def filter(cls, data: Any, sensitive_fields: List[str] = None) -> Any:
        """
        脱敏处理
        
        Args:
            data: 原始数据
            sensitive_fields: 额外指定的敏感字段列表
        
        Returns:
            脱敏后的数据
        """
        if data is None:
            return None
        
        if isinstance(data, dict):
            return cls._filter_dict(data, sensitive_fields)
        elif isinstance(data, list):
            return [cls.filter(item, sensitive_fields) for item in data]
        elif isinstance(data, str):
            return cls._filter_string(data)
        else:
            return data
    
    @classmethod
    def _filter_dict(cls, data: dict, sensitive_fields: List[str] = None) -> dict:
        """处理字典类型"""
        result = {}
        for key, value in data.items():
            # 检查是否为敏感字段
            field_type = cls._detect_field_type(key, sensitive_fields)
            
            if field_type:
                result[key] = cls._mask_value(value, field_type)
            elif isinstance(value, (dict, list)):
                result[key] = cls.filter(value, sensitive_fields)
            else:
                result[key] = value
        
        return result
    
    @classmethod
    def _filter_string(cls, text: str) -> str:
        """处理字符串中的敏感信息"""
        for field_type, pattern in cls.SENSITIVE_PATTERNS.items():
            if field_type in ['credit_card', 'phone', 'id_card', 'email']:
                text = re.sub(pattern, lambda m: cls._mask_value(m.group(), field_type), text)
        return text
    
    @classmethod
    def _detect_field_type(cls, field_name: str, extra_fields: List[str] = None) -> str:
        """检测字段类型"""
        # 检查额外指定的敏感字段
        if extra_fields and field_name in extra_fields:
            return 'password'
        
        # 检查预定义模式
        for field_type, pattern in cls.SENSITIVE_PATTERNS.items():
            if re.search(pattern, field_name):
                return field_type
        
        return None
    
    @classmethod
    def _mask_value(cls, value: Any, field_type: str) -> str:
        """对值进行脱敏"""
        if not isinstance(value, str):
            value = str(value)
        
        rule = cls.MASK_RULES.get(field_type, '****')
        
        if callable(rule):
            return rule(value)
        return rule

import asyncio
from concurrent.futures import ThreadPoolExecutor
from typing import List
import orjson

from kafka import KafkaProducer
from kafka.errors import KafkaError


class AuditBatchWriter:
    """批量写入优化器"""
    
    def __init__(self, batch_size: int = 1000, flush_interval: float = 5.0):
        self.batch_size = batch_size
        self.flush_interval = flush_interval
        self.buffer = []
        self.lock = asyncio.Lock()
        self.executor = ThreadPoolExecutor(max_workers=4)
        
        # Kafka生产者配置
        self.producer = KafkaProducer(
            bootstrap_servers=['kafka-1:9092', 'kafka-2:9092'],
            value_serializer=lambda v: orjson.dumps(v),
            compression_type='lz4',
            batch_size=16384,
            linger_ms=100,
            acks='1',  # 审计日志可接受少量丢失，优先性能
            retries=3,
            max_in_flight_requests_per_connection=5,
        )
        
        # 启动定时刷新任务
        asyncio.create_task(self._scheduled_flush())
    
    async def write(self, event: dict):
        """异步写入事件"""
        async with self.lock:
            self.buffer.append(event)
            
            if len(self.buffer) >= self.batch_size:
                await self._flush()
    
    async def _flush(self):
        """批量刷新到Kafka"""
        if not self.buffer:
            return
        
        events = self.buffer[:]
        self.buffer = []
        
        # 在线程池中执行同步Kafka写入
        loop = asyncio.get_event_loop()
        await loop.run_in_executor(
            self.executor,
            self._send_to_kafka,
            events
        )
    
    def _send_to_kafka(self, events: List[dict]):
        """同步发送到Kafka"""
        topic = 'audit-events'
        
        futures = []
        for event in events:
            # 按租户分区，保证顺序
            partition = hash(event.get('tenant_id', 'default')) % 12
            future = self.producer.send(
                topic,
                value=event,
                partition=partition,
                key=event.get('user_id', '').encode()
            )
            futures.append(future)
        
        # 等待确认（带超时）
        for future in futures:
            try:
                future.get(timeout=10)
            except KafkaError as e:
                logger.error(f"Failed to send audit event: {e}")
    
    async def _scheduled_flush(self):
        """定时刷新任务"""
        while True:
            await asyncio.sleep(self.flush_interval)
            async with self.lock:
                await self._flush()


class AuditAsyncClient:
    """异步审计客户端"""
    
    _instance = None
    _writer = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._writer = AuditBatchWriter()
        return cls._instance
    
    @classmethod
    def send_async(cls, event: dict):
        """异步发送审计事件（非阻塞）"""
        instance = cls()
        asyncio.create_task(instance._writer.write(event))

from django.core.cache import cache

class AuditQueryCache:
