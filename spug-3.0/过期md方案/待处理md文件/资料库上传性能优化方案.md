```javascript
function calculateChunkSize(fileSize) {
    // 小文件：不分片
    if (fileSize < 10 * 1024 * 1024) {
        return fileSize; // 10MB以下直接上传
    }
    // 中等文件：1MB分片
    if (fileSize < 100 * 1024 * 1024) {
        return 1 * 1024 * 1024;
    }
    // 大文件：5MB分片
    if (fileSize < 1024 * 1024 * 1024) {
        return 5 * 1024 * 1024;
    }
    // 超大文件：10MB分片
    return 10 * 1024 * 1024;
}

// 并发上传控制
class ConcurrentUploader {
    constructor(maxConcurrency = 5) {
        this.maxConcurrency = maxConcurrency;
        this.activeUploads = 0;
        this.queue = [];
    }
    
    async upload(chunks) {
        const promises = chunks.map(chunk => this._uploadWithLimit(chunk));
        return Promise.all(promises);
    }
    
    async _uploadWithLimit(chunk) {
        if (this.activeUploads >= this.maxConcurrency) {
            await this._waitForSlot();
        }
        this.activeUploads++;
        try {
            return await this._doUpload(chunk);
        } finally {
            this.activeUploads--;
            this._processQueue();
        }
    }
}
```

```python
# views/upload/chunk_async.py
from celery import shared_task
from django.core.cache import cache

@shared_task(bind=True, max_retries=3)
def save_chunk_async(self, file_hash, chunk_index, chunk_data, user_id):
    """异步保存分片"""
    try:
        chunk_path = get_chunk_path(file_hash, chunk_index, user_id)
        
        # 使用流式写入
        with open(chunk_path, 'wb') as f:
            for chunk in chunk_data.chunks():
                f.write(chunk)
        
        # 更新进度缓存
        cache_key = f"upload_progress:{file_hash}:{user_id}"
        progress = cache.get(cache_key, {'uploaded_chunks': []})
        progress['uploaded_chunks'].append(chunk_index)
        cache.set(cache_key, progress, 3600)
        
        return {'status': 'success', 'chunk_index': chunk_index}
    
    except Exception as e:
        logger.error(f'保存分片失败: {e}')
        raise self.retry(exc=e, countdown=5)


class AsyncChunkUploadView(View):
```

```python
# libs/chunk_cache.py
import redis
from django.conf import settings

class ChunkProgressCache:
    """分片进度缓存管理"""
    
    def __init__(self):
        self.client = redis.Redis.from_url(settings.REDIS_URL)
    
    def set_chunk_uploaded(self, file_hash, chunk_index, ttl=3600):
        """标记分片已上传"""
        key = f"chunks:{file_hash}"
        self.client.sadd(key, chunk_index)
        self.client.expire(key, ttl)
    
    def get_uploaded_chunks(self, file_hash):
        """获取已上传分片列表"""
        key = f"chunks:{file_hash}"
        return [int(x) for x in self.client.smembers(key)]
    
    def is_chunk_uploaded(self, file_hash, chunk_index):
        """检查分片是否已上传"""
        key = f"chunks:{file_hash}"
        return self.client.sismember(key, chunk_index)
    
    def delete_progress(self, file_hash):
        """删除进度记录"""
        key = f"chunks:{file_hash}"
        self.client.delete(key)

# 使用布隆过滤器防止缓存穿透
from pybloom_live import ScalableBloomFilter

class ChunkBloomFilter:
    def __init__(self):
        self.bf = ScalableBloomFilter(mode=ScalableBloomFilter.SMALL_SET_GROWTH)
    
    def add(self, file_hash, chunk_index):
        self.bf.add(f"{file_hash}:{chunk_index}")
    
    def __contains__(self, item):
        return item in self.bf
```

#### 4.2.4 流式合并优化

```python
# tasks/merge_streaming.py
import mmap
import os
from celery import shared_task

@shared_task(bind=True, soft_time_limit=1800)
def merge_chunks_streaming(self, file_hash, total_chunks, output_path):
    """
    流式合并分片
    
    优化点：
    1. 使用内存映射减少内存占用
    2. 批量写入减少IO次数
    3. 进度实时上报
    """
    try:
        # 预分配文件空间
        total_size = sum(
            os.path.getsize(get_chunk_path(file_hash, i))
            for i in range(total_chunks)
        )
        
        with open(output_path, 'wb') as outfile:
            # 预分配空间
            outfile.truncate(total_size)
            
            # 使用内存映射
            with mmap.mmap(outfile.fileno(), 0, access=mmap.ACCESS_WRITE) as mm:
                offset = 0
                for i in range(total_chunks):
                    chunk_path = get_chunk_path(file_hash, i)
                    
                    with open(chunk_path, 'rb') as chunk_file:
                        chunk_data = chunk_file.read()
                        mm[offset:offset + len(chunk_data)] = chunk_data
                        offset += len(chunk_data)
                    
                    # 更新进度
                    if i % 10 == 0:
                        self.update_state(
                            state='PROGRESS',
                            meta={'progress': (i + 1) / total_chunks * 100}
                        )
        
        return {'status': 'success', 'output_path': output_path}
    
    except SoftTimeLimitExceeded:
        logger.error('合并任务超时')
        raise
```

#### 4.3.1 对象存储集成

```python
# libs/storage/oss.py
import oss2
from django.conf import settings

class OSSStorage:
    """阿里云OSS存储"""
    
    def __init__(self):
        self.auth = oss2.Auth(
            settings.OSS_ACCESS_KEY_ID,
            settings.OSS_ACCESS_KEY_SECRET
        )
        self.bucket = oss2.Bucket(
            self.auth,
            settings.OSS_ENDPOINT,
            settings.OSS_BUCKET_NAME
        )
    
    def upload_chunk(self, file_hash, chunk_index, chunk_data):
        """上传分片到OSS"""
        key = f"chunks/{file_hash}/{chunk_index}.part"
        self.bucket.put_object(key, chunk_data)
        return key
    
    def merge_chunks(self, file_hash, total_chunks, output_key):
        """调用OSS合并接口"""
        # 使用OSS的AppendUpload或MultipartUpload
        from oss2.models import PartInfo
        
        upload_id = self.bucket.init_multipart_upload(output_key).upload_id
        parts = []
        
        for i in range(total_chunks):
            key = f"chunks/{file_hash}/{i}.part"
            part = self.bucket.upload_part(
                output_key, upload_id, i + 1,
                self.bucket.get_object(key)
            )
            parts.append(PartInfo(i + 1, part.etag))
        
        self.bucket.complete_multipart_upload(output_key, upload_id, parts)
        return output_key


# 本地存储（小文件）+ OSS（大文件）混合策略
class HybridStorage:
    def __init__(self):
        self.local = LocalStorage()
        self.oss = OSSStorage()
        self.threshold = 100 * 1024 * 1024  # 100MB
    
    def store(self, file_data, file_size):
        if file_size < self.threshold:
            return self.local.store(file_data)
        else:
            return self.oss.store(file_data)
```

```python
# middleware/rate_limit.py
from django.core.cache import cache
from rest_framework.throttling import BaseThrottle

class UploadRateThrottle(BaseThrottle):
    """上传限流器"""
    
    def allow_request(self, request, view):
        user_id = request.user.id
        
        # 用户级限流：每分钟10个文件
        user_key = f"upload_limit:user:{user_id}"
        user_count = cache.get(user_key, 0)
        if user_count >= 10:
            return False
        cache.set(user_key, user_count + 1, 60)
        
        # IP级限流：每分钟100个请求
        ip_key = f"upload_limit:ip:{self.get_client_ip(request)}"
        ip_count = cache.get(ip_key, 0)
        if ip_count >= 100:
            return False
        cache.set(ip_key, ip_count + 1, 60)
        
        # 系统级限流：并发上传数
        system_key = "upload_limit:system"
        system_count = cache.get(system_key, 0)
        if system_count >= 50:  # 最多50个并发上传
            return False
        cache.set(system_key, system_count + 1, 300)
        
        return True
```

#### 4.4.2 熔断机制

```python
# libs/circuit_breaker.py
import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=60):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
    
    def call(self, func, *args, **kwargs):
        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
            else:
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise e
    
    def _on_success(self):
        self.failure_count = 0
        self.state = CircuitState.CLOSED
    
    def _on_failure(self):
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

# 使用熔断器保护上传接口
upload_circuit = CircuitBreaker(failure_threshold=10, recovery_timeout=120)

@auth('document.document.upload')
def post(self, request):
    try:
        return upload_circuit.call(self._handle_upload, request)
    except Exception as e:
        return json_response(error='上传服务暂不可用，请稍后重试')
```
